\chapter{Setup}

\section{The fields and the primes}

Let $p$ be a prime number and $f$ a positive integer. Set $q = p^f$.
Let $L = \QQ(\zeta_{p(q-1)})$ be the $p(q-1)$-th cyclotomic field. Inside $L$, we
consider the subfields $K = \QQ(\zeta_{q-1})$ and $F = \QQ(\zeta_p)$, and we fix a primitive
$p$-th root of unity $\zeta_p \in \cO_F$. We fix a maximal ideal $P$ of $\cO_K$ lying above $p$
and a maximal ideal $\fP$ of $\cO_L$ lying above $P$. Since the order of $p$ modulo $q - 1$ is
$f$, the residue field $\cO_K/P$ has $q$ elements.

\begin{remark}
  In the formalization, most results of Chapters 1 to 3 are proved for an arbitrary integer
  $n \geq 1$ in place of $q - 1$, with $K = \QQ(\zeta_n)$, any number field $L$ containing $K$
  and $F$, and the hypothesis that $P$ has Teichmüller lifts of order $n$
  (\ref{def:hasTeichmullerLift}). Some of them then carry hypotheses such as $1 < n$,
  $e(\fP \mid P) \geq 2$ or $\zeta_p - 1 \notin \fP^2$, which all hold in the setting above.
  The case $n = q - 1$ is obtained with \ref{lem:hasTeichmullerLift}.
\end{remark}

\section{The Teichmüller character}

\begin{definition}
  \label{def:hasTeichmullerLift}
  \lean{Ideal.HasTeichmullerLift}
  \leanok
  Let $R$ be a commutative ring, $I$ an ideal of $R$ and $n$ a positive integer. The ideal $I$
  \emph{has Teichmüller lifts of order $n$} if the reduction map
  $\mu_n(R) \to (R/I)^\times$ is bijective.
\end{definition}

\begin{lemma}
  \label{lem:hasTeichmullerLift}
  \uses{def:hasTeichmullerLift}
  \lean{Stickelberger.hasTeichmullerLift}
  \leanok
  The ideal $P$ has Teichmüller lifts of order $q - 1$.
\end{lemma}

\begin{proof}
  \leanok
  The ring $\cO_K$ contains $q - 1$ distinct $(q-1)$-th roots of unity, and the integer $q - 1$
  is not in $P$. Hence these roots stay distinct modulo $P$, so the reduction map is injective,
  and it is bijective since $(\cO_K/P)^\times$ has $q - 1$ elements.
\end{proof}

\begin{definition}
  \label{def:teichmuller}
  \uses{def:hasTeichmullerLift}
  \lean{MulChar.teichmuller}
  \leanok
  Assume that $P$ has Teichmüller lifts of order $q - 1$ (\ref{lem:hasTeichmullerLift}).
  The \emph{Teichmüller character} $\om : \cO_K/P \to \cO_K$ is the multiplicative character
  whose restriction to $(\cO_K/P)^\times$ is the inverse of the reduction map
  $\mu_{q-1}(\cO_K) \xrightarrow{\sim} (\cO_K/P)^\times$, extended by $\om(0) = 0$.
\end{definition}

\begin{lemma}
  \label{lem:mk_teichmuller}
  \uses{def:teichmuller}
  \lean{MulChar.mk_teichmuller_apply}
  \leanok
  For all $x \in \cO_K/P$, we have $\om(x) \equiv x \pmod{P}$.
\end{lemma}

\begin{proof}
  \leanok
  This is the definition for $x \neq 0$, and both sides vanish for $x = 0$.
\end{proof}

\begin{lemma}
  \label{lem:orderOf_teichmuller}
  \uses{def:teichmuller}
  \lean{MulChar.orderOf_teichmuller}
  \leanok
  The order of $\om$ as a multiplicative character is $q - 1$.
\end{lemma}

\begin{proof}
  \leanok
  The values of $\om$ on units are $(q-1)$-th roots of unity, so $\om^{q-1} = 1$. The value of
  $\om$ at the reduction of a primitive $(q-1)$-th root of unity is that root, so no smaller
  power of $\om$ is trivial.
\end{proof}

\begin{lemma}
  \label{lem:teichmuller_sum}
  \uses{lem:orderOf_teichmuller}
  \lean{Stickelberger.ringHomComp_teichmuller_zpow_ne_one}
  \leanok
  Let $a \in \ZZ$ with $(q-1) \nmid a$. The character $\om^{-a}$, viewed with values in $\cO_L$,
  is nontrivial. In particular,
  \[
    \sum_{x \in \cO_K/P} \om(x)^{-a} = 0.
  \]
\end{lemma}

\begin{proof}
  \leanok
  The character $\om^{-a}$ is nontrivial by \ref{lem:orderOf_teichmuller}, and stays nontrivial
  after composing with the injection $\cO_K \to \cO_L$. The vanishing of the sum is then
  \texttt{MulChar.sum\_eq\_zero\_of\_ne\_one}.
\end{proof}

\begin{lemma}
  \label{lem:map_ringHomComp_teichmuller_zpow_apply}
  \uses{def:teichmuller}
  \lean{MulChar.map_ringHomComp_teichmuller_zpow_apply}
  \leanok
  Let $R$ be a ring containing $\cO_K$ and let $\sigma$ be a ring endomorphism of $R$ such that
  $\sigma(\zeta_{q-1}) = \zeta_{q-1}^m$ for some $m \in \mathbb{N}$. Then
  $\sigma(\om(x)^a) = \om(x)^{am}$ for all $x \in \cO_K/P$ and $a \in \ZZ$.
\end{lemma}

\begin{proof}
  \leanok
  For $x \neq 0$, the value $\om(x)^a$ is a power $\zeta_{q-1}^k$, so
  $\sigma(\om(x)^a) = \zeta_{q-1}^{mk} = \om(x)^{am}$. Both sides vanish for $x = 0$.
\end{proof}

\section{The additive character}

\begin{definition}
  \label{def:addchar}
  \lean{AddChar.traceChar}
  \leanok
  The \emph{additive character} $\psi : \cO_K/P \to \cO_F$ is defined by
  \[
    \psi(x) = \zeta_p^{\tr_{(\cO_K/P)/(\ZZ/p\ZZ)}(x)}.
  \]
  We also view it as a character with values in $\cO_L$.
\end{definition}

\begin{lemma}
  \label{lem:addCharTrace_ne_one}
  \uses{def:addchar}
  \lean{AddChar.traceChar_ne_one}
  \leanok
  The character $\psi$ is nontrivial, i.e., $\psi \neq 1$.
\end{lemma}

\begin{proof}
  \leanok
  Since the extension $\cO_K/P$ over $\ZZ/p\ZZ$ is separable, the trace
  $\tr : \cO_K/P \to \ZZ/p\ZZ$ is nonzero, so there exists $x$ with
  $\tr(x) \neq 0$, hence $\psi(x) = \zeta_p^{\tr(x)} \neq 1$.
\end{proof}

\begin{lemma}
  \label{lem:addCharTrace_isPrimitive}
  \uses{lem:addCharTrace_ne_one}
  \lean{AddChar.isPrimitive_traceChar}
  \leanok
  The character $\psi$ is primitive.
\end{lemma}

\begin{proof}
  \leanok
  Since $\cO_K/P$ is a field, any nontrivial additive character is primitive.
  The character $\psi$ is nontrivial by \ref{lem:addCharTrace_ne_one}.
\end{proof}

\begin{lemma}
  \label{lem:addCharTrace_frob_apply}
  \uses{def:addchar}
  \lean{AddChar.traceChar_apply_pow}
  \leanok
  For all $x \in \cO_K/P$, $\psi(x^p) = \psi(x)$.
\end{lemma}

\begin{proof}
  \leanok
  The Frobenius map $x \mapsto x^p$ is a $(\ZZ/p\ZZ)$-algebra automorphism of $\cO_K/P$, so
  it preserves the trace.
\end{proof}

\begin{lemma}
  \label{lem:addCharTrace_mk_sq_eq}
  \uses{def:addchar}
  \lean{AddChar.mk_traceChar_apply_eq_one_add_smul}
  \leanok
  Let $\fQ$ be an ideal of $\cO_L$ with $\zeta_p - 1 \in \fQ$. Then
  \[
    \psi(x) \equiv 1 + \tr(x) \cdot (\zeta_p - 1) \pmod{\fQ^2}
  \]
  for all $x \in \cO_K/P$.
\end{lemma}

\begin{proof}
  \leanok
  Write $\tr(x) = a$ with $a \in \mathbb{N}$. By the binomial theorem,
  $\psi(x) = (1 + (\zeta_p - 1))^a \equiv 1 + a(\zeta_p - 1) \pmod{\fQ^2}$.
\end{proof}

\begin{lemma}
  \label{lem:addCharTrace_mk_eq_one}
  \uses{lem:addCharTrace_mk_sq_eq}
  \lean{AddChar.mk_traceChar_apply_eq_one}
  \leanok
  Let $\fQ$ be an ideal of $\cO_L$ with $\zeta_p - 1 \in \fQ$.
  Then $\psi(x) \equiv 1 \pmod{\fQ}$ for all $x \in \cO_K/P$.
\end{lemma}

\begin{proof}
  \leanok
  This follows from \ref{lem:addCharTrace_mk_sq_eq}, since $\tr(x) \cdot (\zeta_p - 1) \in \fQ$.
\end{proof}

\section{Primes above $p$}

\begin{lemma}
  \label{lem:zeta_sub_one_mem}
  \lean{Stickelberger.zeta_sub_one_mem}
  \leanok
  For every prime ideal $\fQ$ of $\cO_L$ above $p$, we have $\zeta_p - 1 \in \fQ$.
\end{lemma}

\begin{proof}
  \leanok
  The ideal $(\zeta_p - 1)$ is the only prime of $\cO_F$ above $p$, so it is
  $\fQ \cap \cO_F$.
\end{proof}

\begin{lemma}
  \label{lem:ramificationIdx_under_eq_one}
  \uses{lem:ramificationIdx_eq_p_sub_one}
  \lean{Stickelberger.ramificationIdx_under_eq_one}
  \leanok
  Every prime ideal of $\cO_L$ above $p$ is unramified over $\cO_F$.
\end{lemma}

\begin{proof}
  \leanok
  Let $\fQ$ be such a prime. By \ref{lem:ramificationIdx_eq_p_sub_one}, $e(\fQ \mid p) = p - 1$,
  and this is also the ramification index of $p$ in $F$. By multiplicativity of ramification
  indices in the tower $L/F/\QQ$, the ramification index of $\fQ$ over $\cO_F$ is $1$.
\end{proof}

\chapter{The Gauss sum}
\label{chap:gauss_sum}

Throughout this chapter and the next two, we assume that $p$ is an odd prime.

\section{Gauss and Jacobi sums}

\begin{definition}
  \label{def:gauss_sum_s}
  \uses{def:teichmuller, def:addchar}
  \lean{Stickelberger.gaussSum}
  \leanok
  For $a \in \ZZ$, the \emph{Gauss sum} is
  \[
    g_a = g(\om^{-a}, \psi) = \sum_{x \in \cO_K/P} \om(x)^{-a} \cdot \psi(x) \in \cO_L.
  \]
\end{definition}

\begin{definition}
  \label{def:jacobi_sum}
  \uses{def:teichmuller}
  \lean{Stickelberger.jacobiSum}
  \leanok
  For $a, b \in \ZZ$, the \emph{Jacobi sum} is
  \[
    J(a, b) = J(\om^{-a}, \om^{-b}) = \sum_{x + y = 1} \om(x)^{-a} \om(y)^{-b} \in \cO_K.
  \]
\end{definition}

\begin{lemma}
  \label{lem:GaussSum_periodic}
  \uses{def:gauss_sum_s, lem:orderOf_teichmuller}
  \lean{Stickelberger.gaussSum_add_of_dvd}
  \leanok
  If $(q-1) \mid k$, then for all $a \in \ZZ$, $g_{a+k} = g_a$.
\end{lemma}

\begin{proof}
  \leanok
  Since $\om^{q-1} = 1$ by \ref{lem:orderOf_teichmuller}, we have $\om^{-k} = 1$, so
  $\om^{-(a+k)} = \om^{-a}$.
\end{proof}

\begin{lemma}
  \label{lem:gaussSum_of_dvd}
  \uses{lem:GaussSum_periodic, lem:gaussSum_one_left, lem:addCharTrace_ne_one}
  \lean{Stickelberger.gaussSum_eq_neg_one_of_dvd}
  \leanok
  If $(q-1) \mid a$, then $g_a = -1$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:GaussSum_periodic}, $g_a = g_0 = g(1, \psi)$, which is $-1$ by
  \ref{lem:gaussSum_one_left} since $\psi$ is nontrivial by \ref{lem:addCharTrace_ne_one}.
\end{proof}

\begin{lemma}
  \label{lem:GaussSum_frob}
  \uses{def:gauss_sum_s, lem:addCharTrace_frob_apply}
  \lean{Stickelberger.gaussSum_frob}
  \leanok
  For all $a \in \ZZ$, $g_{pa} = g_a$.
\end{lemma}

\begin{proof}
  \leanok
  Substitute $x \mapsto x^p$, a bijection of $\cO_K/P$, in the sum defining $g_a$. We have
  $\om(x^p)^{-a} = \om(x)^{-pa}$ and $\psi(x^p) = \psi(x)$ by
  \ref{lem:addCharTrace_frob_apply}.
\end{proof}

\begin{lemma}
  \label{lem:GaussSum_mul_GaussSum}
  \uses{def:gauss_sum_s, def:jacobi_sum, lem:teichmuller_sum}
  \lean{Stickelberger.gaussSum_mul_gaussSum}
  \leanok
  If $(q-1) \nmid (a+b)$, then
  \[
    g_a \cdot g_b = g_{a+b} \cdot J(a, b).
  \]
\end{lemma}

\begin{proof}
  \leanok
  The character $\om^{-(a+b)}$ is nontrivial by \ref{lem:teichmuller_sum}, so this is
  \texttt{jacobiSum\_mul\_nontrivial} applied to $\om^{-a}$ and $\om^{-b}$.
\end{proof}

\begin{lemma}
  \label{lem:GaussSum_mul_GaussSum_neg}
  \uses{def:gauss_sum_s, lem:addCharTrace_isPrimitive, lem:teichmuller_sum}
  \lean{Stickelberger.gaussSum_mul_gaussSum_neg}
  \leanok
  If $(q-1) \nmid a$, then
  \[
    g_a \cdot g_{-a} = \om^{-a}(-1) \cdot p^f.
  \]
\end{lemma}

\begin{proof}
  \leanok
  The character $\om^{-a}$ is nontrivial by \ref{lem:teichmuller_sum} and $\psi$ is primitive
  by \ref{lem:addCharTrace_isPrimitive}. By \texttt{gaussSum\_mul\_gaussSum\_eq\_card},
  $g(\om^{-a}, \psi) \cdot g(\om^{a}, \psi^{-1}) = |\cO_K/P| = p^f$, and the change of
  variable $x \mapsto -x$ gives $g(\om^{a}, \psi^{-1}) = \om^{a}(-1) \cdot g_{-a}$.
\end{proof}

\section{Congruences}

\begin{lemma}
  \label{lem:gaussSum_mem_span_zeta_sub_one}
  \uses{def:gauss_sum_s, lem:addCharTrace_mk_eq_one, lem:teichmuller_sum}
  \lean{Stickelberger.gaussSum_mem_span_zeta_sub_one}
  \leanok
  If $(q-1) \nmid a$, then $g_a \in (\zeta_p - 1)\cO_L$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:addCharTrace_mk_eq_one}, $\psi(x) \equiv 1$ modulo $(\zeta_p - 1)$, so
  $g_a \equiv \sum_x \om(x)^{-a}$, which is $0$ by \ref{lem:teichmuller_sum}.
\end{proof}

\begin{lemma}
  \label{lem:GaussSum_mem}
  \uses{lem:gaussSum_mem_span_zeta_sub_one, lem:zeta_sub_one_mem}
  \lean{Stickelberger.gaussSum_mem}
  \leanok
  If $(q-1) \nmid a$, then $g_a \in \fP$.
\end{lemma}

\begin{proof}
  \leanok
  Combine \ref{lem:gaussSum_mem_span_zeta_sub_one} and \ref{lem:zeta_sub_one_mem}.
\end{proof}

\begin{lemma}
  \label{lem:not_isUnit_gaussSum}
  \uses{lem:gaussSum_mem_span_zeta_sub_one}
  \lean{Stickelberger.not_isUnit_gaussSum}
  \leanok
  If $(q-1) \nmid a$, then $g_a$ is not a unit of $\cO_L$.
\end{lemma}

\begin{proof}
  \leanok
  The element $\zeta_p - 1$ is not a unit of $\cO_L$, so this follows from
  \ref{lem:gaussSum_mem_span_zeta_sub_one}.
\end{proof}

\begin{lemma}
  \label{lem:norm_GaussSum}
  \uses{lem:GaussSum_mul_GaussSum_neg, lem:not_isUnit_gaussSum}
  \lean{Stickelberger.exists_natAbs_norm_gaussSum_eq_pow}
  \leanok
  If $(q-1) \nmid a$, then there exists $k \geq 1$ such that
  $|N_{L/\QQ}(g_a)| = p^k$.
\end{lemma}

\begin{proof}
  \leanok
  Taking norms in \ref{lem:GaussSum_mul_GaussSum_neg}, the integer $|N_{L/\QQ}(g_a)|$ divides
  a power of $p$, so it is a power $p^k$. Since $g_a$ is not a unit by
  \ref{lem:not_isUnit_gaussSum}, we have $k \geq 1$.
\end{proof}

\begin{lemma}
  \label{lem:gaussSum_ne_zero}
  \uses{lem:gaussSum_of_dvd, lem:norm_GaussSum, lem:GaussSum_mul_GaussSum}
  \lean{Stickelberger.gaussSum_ne_zero, Stickelberger.jacobiSum_ne_zero}
  \leanok
  For all $a \in \ZZ$, we have $g_a \neq 0$. If $(q-1) \nmid (a+b)$, then $J(a, b) \neq 0$.
\end{lemma}

\begin{proof}
  \leanok
  If $(q-1) \mid a$, then $g_a = -1$ by \ref{lem:gaussSum_of_dvd}. Otherwise its norm is
  nonzero by \ref{lem:norm_GaussSum}. The second claim follows from the first and
  \ref{lem:GaussSum_mul_GaussSum}.
\end{proof}

\begin{lemma}
  \label{lem:mk_sq_gausssum_eq}
  \uses{lem:addCharTrace_mk_sq_eq, lem:teichmuller_sum, lem:mk_teichmuller,
        lem:zeta_sub_one_mem, lem:ramificationIdx_eq_p_sub_one}
  \lean{Stickelberger.mk_gaussSum_one_eq_neg_zeta_sub_one}
  \leanok
  We have
  \[
    g_1 \equiv -(\zeta_p - 1) \pmod{\fP^2}.
  \]
\end{lemma}

\begin{proof}
  \leanok
  The extension $K/\QQ$ is unramified at $p$, so $e(\fP \mid P) = p - 1 \geq 2$ by
  \ref{lem:ramificationIdx_eq_p_sub_one}, hence $P \subseteq \fP^2$. Since
  $\zeta_p - 1 \in \fP$ by \ref{lem:zeta_sub_one_mem}, \ref{lem:addCharTrace_mk_sq_eq} gives
  \[
    g_1 \equiv \sum_{x} \om(x)^{-1} + (\zeta_p-1) \sum_{x} \om(x)^{-1} \tr(x) \pmod{\fP^2}.
  \]
  The first sum vanishes by \ref{lem:teichmuller_sum} since $q - 1 \geq 2$. For the second, we
  have $\om(x) \equiv x \pmod{P}$ by \ref{lem:mk_teichmuller} and
  $\tr(x) = \sum_{i<f} x^{p^i}$, and $\sum_{x \in \cO_K/P} x^{-1} x^{p^i}$ is $-1$ for $i = 0$
  and $0$ for $0 < i < f$. Hence the second sum is $-1$ modulo $\fP^2$.
\end{proof}

\section{Galois action}

\begin{lemma}
  \label{lem:galFEquiv}
  \lean{Stickelberger.galFEquiv}
  \leanok
  Restriction to $K$ induces an isomorphism $\Gal(L/F) \cong \Gal(K/\QQ) \cong
  (\ZZ/(q-1)\ZZ)^\times$. For $\sigma \in \Gal(L/F)$, we denote by
  $\fn(\sigma) \in \{0, \ldots, q-2\}$ the representative of its image, so that
  $\sigma(\zeta_{q-1}) = \zeta_{q-1}^{\fn(\sigma)}$.
\end{lemma}

\begin{proof}
  \leanok
  Since $\gcd(q-1, p) = 1$, the fields $K$ and $F$ are linearly disjoint over $\QQ$ and
  $L = KF$.
\end{proof}

\begin{lemma}
  \label{lem:gal_gaussSum_eq_gaussSum}
  \uses{def:gauss_sum_s, lem:galFEquiv, lem:map_ringHomComp_teichmuller_zpow_apply}
  \lean{Stickelberger.smul_gaussSum_eq_mul_galFEquiv}
  \leanok
  For all $\sigma \in \Gal(L/F)$ and $a \in \ZZ$,
  \[
    \sigma(g_a) = g_{a \cdot \fn(\sigma)}.
  \]
\end{lemma}

\begin{proof}
  \leanok
  The automorphism $\sigma$ fixes $\zeta_p$, hence the values of $\psi$, and
  $\sigma(\om(x)^{-a}) = \om(x)^{-a \fn(\sigma)}$ by
  \ref{lem:map_ringHomComp_teichmuller_zpow_apply}.
\end{proof}

\begin{lemma}
  \label{lem:gauss_sum_galois_action_K}
  \uses{def:gauss_sum_s, def:addchar, lem:map_ringHomComp_teichmuller_zpow_apply}
  \lean{Stickelberger.smul_gaussSum_eq_mul_gaussSum}
  \leanok
  Let $\tau \in \Gal(L/K)$ and $e \in (\ZZ/p\ZZ)^\times$ with $\tau(\zeta_p) = \zeta_p^e$.
  Then for all $a \in \ZZ$,
  \[
    \tau(g_a) = \om(e)^{a} \cdot g_a.
  \]
\end{lemma}

\begin{proof}
  \leanok
  The automorphism $\tau$ fixes the values of $\om$ and $\tau(\psi(x)) = \psi(ex)$. The
  substitution $y = ex$ in the sum gives
  $\tau(g_a) = \sum_y \om(y/e)^{-a}\psi(y) = \om(e)^{a} g_a$.
\end{proof}

\chapter{The valuation function $s$}

\section{General properties}

\begin{definition}
  \label{def:s}
  \uses{def:gauss_sum_s}
  \lean{Stickelberger.valGauss}
  \leanok
  For $a \in \ZZ$ and a maximal ideal $\fQ$ of $\cO_L$, let $s_{\fQ}(a) \in \mathbb{N}$ be the
  multiplicity of $\fQ$ in the ideal $(g_a)$. We write $s(a) = s_{\fP}(a) = v_{\fP}(g_a)$.
\end{definition}

\begin{lemma}
  \label{lem:valGauss_periodic}
  \uses{def:s, lem:GaussSum_periodic}
  \lean{Stickelberger.valGauss_congr}
  \leanok
  If $a \equiv b \pmod{q-1}$, then $s(a) = s(b)$.
\end{lemma}

\begin{proof}
  \leanok
  Indeed $g_a = g_b$ by \ref{lem:GaussSum_periodic}.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_eq_zero}
  \uses{def:s, lem:gaussSum_of_dvd}
  \lean{Stickelberger.valGauss_eq_zero_of_dvd}
  \leanok
  If $(q - 1) \mid a$, then $s(a) = 0$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:gaussSum_of_dvd}, $g_a = -1$ is a unit, so $(g_a) = \cO_L$.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_add_valGauss}
  \uses{def:s, lem:GaussSum_mul_GaussSum, lem:gaussSum_ne_zero}
  \lean{Stickelberger.valGauss_add_valGauss}
  \leanok
  If $(q-1) \nmid (a+b)$, then $s(a) + s(b) = s(a+b) + v_{\fP}(J(a, b))$.
\end{lemma}

\begin{proof}
  \leanok
  Take valuations in \ref{lem:GaussSum_mul_GaussSum}. All the elements involved are nonzero by
  \ref{lem:gaussSum_ne_zero}, so the valuations are finite.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_subadditive}
  \uses{lem:valGauss_add_valGauss, lem:valGauss_eq_zero}
  \lean{Stickelberger.valGauss_add_le}
  \leanok
  For all $a, b \in \ZZ$, $s(a + b) \leq s(a) + s(b)$.
\end{lemma}

\begin{proof}
  \leanok
  If $(q-1) \mid (a+b)$, then $s(a+b) = 0$ by \ref{lem:valGauss_eq_zero}. Otherwise use
  \ref{lem:valGauss_add_valGauss}.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_add_valGauss_neg_self}
  \uses{def:s, lem:GaussSum_mul_GaussSum_neg, lem:gaussSum_ne_zero}
  \lean{Stickelberger.valGauss_add_valGauss_neg}
  \leanok
  If $(q-1) \nmid a$, then $s(a) + s(-a) = f \cdot e(\fP \mid p)$.
\end{lemma}

\begin{proof}
  \leanok
  Take valuations in \ref{lem:GaussSum_mul_GaussSum_neg}: the factor $\om^{-a}(-1)$ is a root of
  unity, hence a unit, and $v_{\fP}(p^f) = f \cdot e(\fP \mid p)$.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_add_valGauss_sub_self}
  \uses{lem:valGauss_add_valGauss_neg_self, lem:valGauss_periodic, lem:valGauss_eq_zero}
  \lean{Stickelberger.valGauss_add_valGauss_sub}
  \leanok
  Assume $(q-1) \mid k$. Then
  \[
    s(a) + s(k - a) = \begin{cases} 0 & \text{if } (q-1) \mid a, \\
    f \cdot e(\fP \mid p) & \text{otherwise.} \end{cases}
  \]
\end{lemma}

\begin{proof}
  \leanok
  We have $s(k - a) = s(-a)$ by \ref{lem:valGauss_periodic}. Conclude with
  \ref{lem:valGauss_eq_zero} and \ref{lem:valGauss_add_valGauss_neg_self}.
\end{proof}

\begin{lemma}
  \label{lem:s_sum}
  \uses{lem:valGauss_add_valGauss_sub_self, lem:valGauss_eq_zero}
  \lean{Stickelberger.two_mul_sum_valGauss}
  \leanok
  We have
  \[
    2 \sum_{a=0}^{q-2} s(a) = (q-2) \cdot f \cdot e(\fP \mid p).
  \]
\end{lemma}

\begin{proof}
  \leanok
  Pair the term $a$ with $q - 1 - a$. By \ref{lem:valGauss_add_valGauss_sub_self} with
  $k = q - 1$, each of the $q - 2$ values $1 \leq a \leq q-2$ gives
  $s(a) + s(q-1-a) = f \cdot e(\fP \mid p)$, and $s(0) = s(q-1) = 0$ by
  \ref{lem:valGauss_eq_zero}.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_eq_zero_of_not_liesOver}
  \uses{def:s, lem:norm_GaussSum, lem:valGauss_eq_zero}
  \lean{Stickelberger.valGauss_eq_zero_of_not_liesOver}
  \leanok
  Let $\fQ$ be a maximal ideal of $\cO_L$ not lying above $p$. Then $s_{\fQ}(a) = 0$ for all
  $a \in \ZZ$.
\end{lemma}

\begin{proof}
  \leanok
  If $(q-1) \mid a$, use \ref{lem:valGauss_eq_zero}. Otherwise, if $g_a \in \fQ$, then the norm
  of $\fQ$ divides $N_{L/\QQ}(g_a)$, which is a power of $p$ by \ref{lem:norm_GaussSum}, so
  $\fQ$ lies above $p$.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_smul}
  \uses{def:s, lem:gal_gaussSum_eq_gaussSum}
  \lean{Stickelberger.valGauss_smul}
  \leanok
  For all $\sigma \in \Gal(L/F)$ and $a \in \ZZ$,
  $s_{\sigma(\fP)}(a) = s_{\fP}(a \cdot \fn(\sigma^{-1}))$.
\end{lemma}

\begin{proof}
  \leanok
  The multiplicity of $\sigma(\fP)$ in $(g_a)$ is the multiplicity of $\fP$ in
  $(\sigma^{-1}(g_a))$, and $\sigma^{-1}(g_a) = g_{a \fn(\sigma^{-1})}$ by
  \ref{lem:gal_gaussSum_eq_gaussSum}.
\end{proof}

\section{Upper bound by the sum of digits}

\begin{lemma}
  \label{lem:valGauss_one}
  \uses{lem:GaussSum_mem, lem:mk_sq_gausssum_eq, lem:zeta_sub_one_not_mem_sq}
  \lean{Stickelberger.valGauss_one}
  \leanok
  We have $s(1) = 1$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:GaussSum_mem}, $g_1 \in \fP$, so $s(1) \geq 1$.
  By \ref{lem:mk_sq_gausssum_eq}, $g_1 \equiv -(\zeta_p - 1) \pmod{\fP^2}$, and
  $\zeta_p - 1 \notin \fP^2$ by \ref{lem:zeta_sub_one_not_mem_sq}, so $s(1) \leq 1$.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_le_self}
  \uses{lem:valGauss_subadditive, lem:valGauss_one, lem:valGauss_eq_zero}
  \lean{Stickelberger.valGauss_le_self}
  \leanok
  For all $a \in \mathbb{N}$, $s(a) \leq a$.
\end{lemma}

\begin{proof}
  \leanok
  By induction on $a$, using $s(0) = 0$ (\ref{lem:valGauss_eq_zero}),
  $s(a+1) \leq s(a) + s(1)$ (\ref{lem:valGauss_subadditive}) and $s(1) = 1$
  (\ref{lem:valGauss_one}).
\end{proof}

\begin{lemma}
  \label{lem:valGauss_le_sum_digits}
  \uses{lem:valGauss_subadditive, lem:valGauss_le_self, lem:GaussSum_frob}
  \lean{Stickelberger.valGauss_le_sum_digits}
  \leanok
  Let $a \in \mathbb{N}$ and let
  $a = a_0 + a_1 p + \cdots + a_{t-1} p^{t-1}$ be its $p$-adic expansion.
  Then
  \[
    s(a) \leq a_0 + a_1 + \cdots + a_{t-1}.
  \]
\end{lemma}

\begin{proof}
  \leanok
  By induction on the list of digits. Write $a = a_0 + p a'$. Then
  $s(a) \leq s(a_0) + s(pa') \leq a_0 + s(a')$ by \ref{lem:valGauss_subadditive},
  \ref{lem:valGauss_le_self}, and $g_{pa'} = g_{a'}$ (\ref{lem:GaussSum_frob}).
\end{proof}

\section{The digit formula}

\begin{proposition}
  \label{prop:valGauss_eq_sum_digits}
  \uses{lem:valGauss_le_sum_digits, lem:s_sum, lem:ramificationIdx_eq_p_sub_one,
        lem:zeta_sub_one_not_mem_sq}
  \lean{Stickelberger.valGauss_eq_sum_digits}
  \leanok
  Let $0 \leq a \leq q - 2$ and let
  $a = a_0 + a_1 p + \cdots + a_{f-1} p^{f-1}$, $0 \leq a_i \leq p - 1$,
  be the $p$-adic expansion of $a$. Then
  \[
    s(a) = a_0 + a_1 + \cdots + a_{f-1}.
  \]
\end{proposition}

\begin{proof}
  \leanok
  By \ref{lem:ramificationIdx_eq_p_sub_one}, $e(\fP \mid p) = p - 1$, so
  \ref{lem:s_sum} gives $\sum_{a=0}^{q-2} s(a) = (q-2)f(p-1)/2$. A direct count shows that
  the sum of the digit sums of $0, \ldots, q-2$ has the same value. Since each term satisfies
  $s(a) \leq a_0 + \cdots + a_{f-1}$ by \ref{lem:valGauss_le_sum_digits}, all these
  inequalities are equalities.
\end{proof}

\begin{lemma}
  \label{lem:sub_one_mul_sum_mod_eq}
  \lean{Nat.sub_one_mul_sum_mul_pow_mod_eq}
  \leanok
  Let $b > 1$, $l \geq 0$ and $0 \leq x < b^l - 1$. Then
  \[
    (b - 1) \sum_{i=0}^{l-1} (x b^i \bmod (b^l - 1)) = (b^l - 1) \cdot S_b(x),
  \]
  where $S_b(x)$ is the sum of the base $b$ digits of $x$.
\end{lemma}

\begin{proof}
  \leanok
  Modulo $b^l - 1$, multiplication by $b$ cyclically rotates the $l$ base $b$ digits of $x$.
  Hence each digit of $x$ occurs exactly once in each position in the sum, which is therefore
  $S_b(x)(1 + b + \cdots + b^{l-1})$.
\end{proof}

\begin{lemma}
  \label{lem:s_mod}
  \uses{prop:valGauss_eq_sum_digits, lem:sub_one_mul_sum_mod_eq}
  \lean{Stickelberger.mul_valGauss_eq_mul_sum}
  \leanok
  Let $m, d$ be positive integers with $q - 1 = md$, and let $a \in \mathbb{N}$ with
  $da \leq q - 2$. Then
  \[
    m \cdot s(da) = (p-1) \sum_{i=0}^{f-1} (ap^i \bmod m).
  \]
\end{lemma}

\begin{proof}
  \leanok
  Multiply both sides by $d$. Since $d (ap^i \bmod m) = dap^i \bmod (q-1)$, this follows from
  \ref{prop:valGauss_eq_sum_digits} and \ref{lem:sub_one_mul_sum_mod_eq} applied to $b = p$,
  $l = f$ and $x = da$.
\end{proof}

\chapter{Ideal factorization of Gauss sums}

\section{Setup for Stickelberger's theorem}

Let $m$ be a positive integer with $p \nmid m$, and assume that $f$ is the multiplicative order
of $p$ modulo $m$. Then $m \mid q - 1$, and we set $d = (q-1)/m$. Let $k = \QQ(\zeta_m)$, a
subfield of $L$, and let $\fp$ be a maximal ideal of $\cO_k$ lying above $p$. We assume that
$\fP$ lies above $\fp$.

Let $G = \Gal(k/\QQ) \cong (\ZZ/m\ZZ)^\times$.
For $a \in (\ZZ/m\ZZ)^\times$, let $\sigma_a \in G$ denote the element with
$\sigma_a(\zeta_m) = \zeta_m^a$. The \emph{Stickelberger element} is
\[
  \Theta = \Theta(k) = \sum_{a \in (\ZZ/m\ZZ)^\times} a \cdot \sigma_a^{-1} \in \ZZ[G],
\]
where $a$ is identified with its representative in $\{0, \ldots, m-1\}$. For an ideal
$\mathfrak{a}$ of $\cO_k$, we write
$\mathfrak{a}^{\Theta} = \prod_{a \in (\ZZ/m\ZZ)^\times} \sigma_a^{-1}(\mathfrak{a})^{a}$.
\begin{remark}
  With the notation of Washington, $\Theta = m\theta$ where
  $\theta = \sum_{(a,m)=1} \{a/m\} \sigma_a^{-1} \in \QQ[G]$.
\end{remark}

\section{Descent of $g_d^m$ to $k$}

\begin{lemma}
  \label{lem:gauss_sum_power_fixed_2}
  \uses{lem:gauss_sum_galois_action_K, lem:orderOf_teichmuller}
  \lean{Stickelberger.smul_gaussSum_pow_eq_self}
  \leanok
  For any $\tau \in \Gal(L/K)$, we have $\tau(g_d^m) = g_d^m$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:gauss_sum_galois_action_K}, $\tau(g_d) = \om(e)^d g_d$ for some $e$, and
  $\om(e)^{dm} = \om(e)^{q-1} = 1$ by \ref{lem:orderOf_teichmuller}.
\end{proof}

\begin{lemma}
  \label{lem:gauss_sum_power_fixed_1}
  \uses{lem:gal_gaussSum_eq_gaussSum, lem:GaussSum_periodic}
  \lean{Stickelberger.smul_gaussSum_eq_self}
  \leanok
  Let $E = \QQ(\zeta_{mp}) \subseteq L$. For any $\tau \in \Gal(L/E)$, we have
  $\tau(g_d) = g_d$.
\end{lemma}

\begin{proof}
  \leanok
  The automorphism $\tau$ fixes $F$ and $\zeta_m$, so $\fn(\tau) \equiv 1 \pmod{m}$, hence
  $d \fn(\tau) \equiv d \pmod{q-1}$. Conclude with \ref{lem:gal_gaussSum_eq_gaussSum} and
  \ref{lem:GaussSum_periodic}.
\end{proof}

\begin{lemma}
  \label{lem:gauss_sum_power_in_cyclotomic}
  \uses{lem:gauss_sum_power_fixed_1, lem:gauss_sum_power_fixed_2}
  \lean{Stickelberger.exists_gaussSum_pow_eq_algebraMap}
  \leanok
  There exists $\Gamma \in \cO_k$ with $g_d^m = \Gamma$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:gauss_sum_power_fixed_2} and \ref{lem:gauss_sum_power_fixed_1}, the element
  $g_d^m$ lies in $K \cap E = \QQ(\zeta_{\gcd(q-1, mp)}) = k$, and it is integral.
\end{proof}

\section{Factorization in $L$}

\begin{lemma}
  \label{lem:gaussSum_factorization_L}
  \uses{def:s, lem:gaussSum_ne_zero, lem:valGauss_eq_zero_of_not_liesOver}
  \lean{Stickelberger.span_singleton_gaussSum_eq_prod}
  \leanok
  For all $a \in \ZZ$, we have the equality of ideals of $\cO_L$
  \[
    (g_a) = \prod_{\fQ \mid p} \fQ^{s_{\fQ}(a)},
  \]
  where the product runs over the primes of $\cO_L$ above $p$.
\end{lemma}

\begin{proof}
  \leanok

\end{proof}

\section{Primes of $k$ above $p$}

\begin{lemma}
  \label{lem:stabilizer_tP0}
  \lean{IsCyclotomicExtension.Rat.galEquivZMod_stabilizer}
  \leanok
  The stabilizer of $\fp$ in $G$ is the subgroup $\langle \sigma_p \rangle$, which has
  order $f$.
\end{lemma}

\begin{proof}
  \leanok
  The stabilizer of $\fp$ is its decomposition group, which acts on the residue field through
  powers of the Frobenius, and $\sigma_p$ induces the Frobenius. The two groups have order $f$.
\end{proof}

\begin{definition}
  \label{def:equivUnits}
  \uses{lem:stabilizer_tP0}
  \lean{Stickelberger.equivUnits}
  \leanok
  For each prime $\fq$ of $\cO_k$ above $p$, choose $a_{\fq} \in (\ZZ/m\ZZ)^\times$ with
  $\sigma_{a_{\fq}}^{-1}(\fp) = \fq$. The map
  \[
    \{\fq \mid p\} \times \{0, \ldots, f-1\} \to (\ZZ/m\ZZ)^\times, \qquad
    (\fq, j) \mapsto a_{\fq} p^j
  \]
  is a bijection, and the image $a$ of $(\fq, j)$ satisfies $\sigma_a^{-1}(\fp) = \fq$.
\end{definition}

\begin{proof}
  \leanok
  The group $G$ acts transitively on the primes above $p$, so $a_{\fq}$ exists. The bijectivity
  and the last claim follow from \ref{lem:stabilizer_tP0}.
\end{proof}

\begin{lemma}
  \label{lem:valuation_stickelberger}
  \uses{def:equivUnits}
  \lean{Stickelberger.sum_equivUnits}
  \leanok
  Let $\fq$ be a prime of $\cO_k$ above $p$ and $a \in (\ZZ/m\ZZ)^\times$ with
  $\sigma_a^{-1}(\fp) = \fq$. Then the sum of the (representatives of the) images of
  $(\fq, j)$, $0 \leq j < f$, under the bijection of \ref{def:equivUnits} is
  $\sum_{j=0}^{f-1} (ap^j \bmod m)$. In particular,
  $v_{\fq}(\fp^{\Theta}) = \sum_{j=0}^{f-1} (ap^j \bmod m)$.
\end{lemma}

\begin{proof}
  \leanok
  The images of the $(\fq, j)$ are the $a p^j$ up to a cyclic shift of $j$ modulo $f$. The
  second claim follows since the exponents of $\fq$ in $\fp^{\Theta}$ are the $b$ with
  $\sigma_b^{-1}(\fp) = \fq$.
\end{proof}

\section{Valuations and the factorization}

\begin{lemma}
  \label{lem:ramificationIdx_eq_sub_one}
  \uses{lem:ramificationIdx_eq_p_sub_one}
  \lean{Stickelberger.ramificationIdx_eq_sub_one'}
  \leanok
  Let $\fq$ be a prime of $\cO_k$ above $p$ and $\fQ$ a prime of $\cO_L$ above $\fq$. Then
  $e(\fQ \mid \fq) = p - 1$.
\end{lemma}

\begin{proof}
  \leanok
  We have $e(\fQ \mid p) = p - 1$ by \ref{lem:ramificationIdx_eq_p_sub_one}, and $k/\QQ$ is
  unramified at $p$ since $p \nmid m$.
\end{proof}

\begin{lemma}
  \label{lem:valuation_gauss_sum_power}
  \uses{lem:s_mod, lem:valGauss_smul, lem:valGauss_periodic, lem:valuation_stickelberger,
        lem:galFEquiv}
  \lean{Stickelberger.mul_valGauss_eq}
  \leanok
  Let $\fq$ be a prime of $\cO_k$ above $p$ and $\fQ$ a prime of $\cO_L$ above $\fq$. Then
  \[
    m \cdot s_{\fQ}(d) = (p - 1) \sum_{j=0}^{f-1} b_j,
  \]
  where $b_j \in \{0, \ldots, m-1\}$ represents the image of $(\fq, j)$ under the bijection of
  \ref{def:equivUnits}.
\end{lemma}

\begin{proof}
  \leanok
  The group $\Gal(L/F)$ acts transitively on the primes of $\cO_L$ above $p$, since $F$ has a
  single prime above $p$. Choose $\tau \in \Gal(L/F)$ with $\tau(\fP) = \fQ$ and let
  $a \in (\ZZ/m\ZZ)^\times$ be the image of $\fn(\tau^{-1})$ (\ref{lem:galFEquiv}). The
  restriction of $\tau$ to $k$ is $\sigma_a^{-1}$, so $\sigma_a^{-1}(\fp) = \fq$. By
  \ref{lem:valGauss_smul} and \ref{lem:valGauss_periodic}, $s_{\fQ}(d) = s_{\fP}(da)$, and we
  conclude with \ref{lem:s_mod} and \ref{lem:valuation_stickelberger}.
\end{proof}

\begin{lemma}
  \label{lem:gauss_sum_factorization_cyclotomic}
  \uses{lem:gauss_sum_power_in_cyclotomic, lem:gaussSum_factorization_L, def:equivUnits,
        lem:ramificationIdx_eq_sub_one, lem:valuation_gauss_sum_power}
  \lean{Stickelberger.span_singleton_eq_prod_of_gaussSum_pow_eq}
  \leanok
  Let $\Gamma \in \cO_k$ with $g_d^m = \Gamma$ (\ref{lem:gauss_sum_power_in_cyclotomic}). Then
  \[
    (\Gamma) = \fp^{\Theta}
  \]
  as ideals of $\cO_k$.
\end{lemma}

\begin{proof}
  \leanok
  Since $\cO_L$ is faithfully flat over $\cO_k$, it suffices to compare the extensions of both
  sides to $\cO_L$. By \ref{lem:gaussSum_factorization_L}, the ideal $(\Gamma)\cO_L = (g_d)^m$
  is $\prod_{\fQ \mid p} \fQ^{m s_{\fQ}(d)}$. Regrouping the factors of $\fp^{\Theta}$ with
  \ref{def:equivUnits} and using $e(\fQ \mid \fq) = p - 1$
  (\ref{lem:ramificationIdx_eq_sub_one}), the extension of $\fp^{\Theta}$ is
  $\prod_{\fq \mid p} \prod_{\fQ \mid \fq} \fQ^{(p-1) \sum_j b_j}$, with the notation of
  \ref{lem:valuation_gauss_sum_power}. The exponents agree by
  \ref{lem:valuation_gauss_sum_power}.
\end{proof}

\section{Stickelberger's theorem}

\begin{lemma}
  \label{lem:stickelberger_in_L}
  \uses{lem:gauss_sum_power_in_cyclotomic, lem:gauss_sum_factorization_cyclotomic,
        lem:hasTeichmullerLift}
  \lean{Stickelberger.isPrincipal_prod_smul_pow_of_orderOf_eq}
  \leanok
  In the setting of this chapter, the ideal $\fp^{\Theta}$ of $\cO_k$ is principal.
\end{lemma}

\begin{proof}
  \leanok
  Choose a maximal ideal $\fP$ of $\cO_L$ above $\fp$ and let $P = \fP \cap \cO_K$. The ideal
  $P$ has Teichmüller lifts of order $q - 1$ by \ref{lem:hasTeichmullerLift}. Take $\Gamma$ as in
  \ref{lem:gauss_sum_power_in_cyclotomic}; then $\fp^{\Theta} = (\Gamma)$ by
  \ref{lem:gauss_sum_factorization_cyclotomic}.
\end{proof}

\begin{theorem}
  \label{thm:stickelberger}
  \uses{lem:stickelberger_in_L}
  \lean{Stickelberger.isPrincipal_prod_smul_pow_of_coprime}
  \leanok
  Let $k$ be a number field isomorphic to $\QQ(\zeta_m)$, let $p$ be an odd prime with
  $p \nmid m$ and let $\fp$ be a maximal ideal of $\cO_k$ above $p$. Then the ideal
  $\fp^{\Theta}$ is principal.
\end{theorem}

\begin{proof}
  \leanok
  Let $f$ be the order of $p$ modulo $m$ and $q = p^f$. The field
  $L = \QQ(\zeta_{p(q-1)})$ contains a subfield $k_0 = \QQ(\zeta_m)$, and an isomorphism
  $k_0 \cong k$ commutes with the action of the $\sigma_a$. Apply
  \ref{lem:stickelberger_in_L} to the preimage of $\fp$ in $\cO_{k_0}$ and transport the
  result to $k$.
\end{proof}

\begin{lemma}
  \label{lem:class_coprime_rep}
  \lean{ClassGroup.exists_mk0_eq_and_isCoprime}
  \leanok
  Let $R$ be a Dedekind domain and $J$ a nonzero ideal of $R$. Every ideal class of $R$
  contains an integral ideal coprime to $J$.
\end{lemma}

\begin{proof}
  \leanok
  Let $I_0$ be an integral ideal in the inverse class. By the Chinese remainder theorem, there is
  $x \in I_0$ such that $(x) I_0^{-1}$ is coprime to $J$, and this ideal lies in the given class.
\end{proof}

\begin{corollary}
  \label{cor:stickelberger_annihilates}
  \uses{thm:stickelberger, lem:class_coprime_rep}
  \lean{Stickelberger.stickelberger_theorem}
  \leanok
  Let $k$ be a number field isomorphic to $\QQ(\zeta_m)$. The Stickelberger element $\Theta$
  annihilates the ideal class group $\mathrm{Cl}(k)$: for every ideal class $c$,
  \[
    c^{\Theta} = \prod_{a \in (\ZZ/m\ZZ)^\times} \sigma_a^{-1}(c)^{a} = 1.
  \]
\end{corollary}

\begin{proof}
  \leanok
  By \ref{lem:class_coprime_rep}, choose an integral ideal $\mathfrak{b} \in c$ coprime to
  $2m$. Every prime factor of $\mathfrak{b}$ lies above an odd prime not dividing $m$, so
  \ref{thm:stickelberger} applied to each prime factor shows that $\mathfrak{b}^{\Theta}$ is
  principal. Hence $c^{\Theta} = [\mathfrak{b}^{\Theta}] = 1$.
\end{proof}

\begin{remark}
  Mathematically, coprimality of the representative to $m$ alone suffices for
  Stickelberger's theorem. We additionally require the norm to be odd (i.e.\
  coprimality to $2m$) only because the formalized form of \ref{thm:stickelberger}
  is proved for primes of odd residue characteristic.
\end{remark}

\chapter{The Kronecker-Weber theorem}

Throughout this chapter, all number fields are subfields of a fixed field $A$ of characteristic
$0$, and for every $n \geq 1$ we fix a primitive $n$-th root of unity $\zeta_n \in A$. Thus
$\QQ(\zeta_n)$ is a well-defined subfield of $A$, and $\QQ(\zeta_n)\QQ(\zeta_{n'}) =
\QQ(\zeta_{\operatorname{lcm}(n,n')})$. The compositum of two subfields $K, K'$ of $A$ is written
$KK'$.

\begin{definition}
  \label{def:unramified_outside}
  \lean{UnramifiedOutside}
  \leanok
  Let $p$ be a prime. A number field $K$ is \emph{unramified outside $p$} if every prime number
  $q \neq p$ is unramified in $K$.
\end{definition}

\begin{lemma}
  \label{lem:unramified_outside_sup}
  \lean{unramifiedOutside_sup}
  \leanok
  \uses{def:unramified_outside}
  If $K$ and $F$ are number fields unramified outside $p$, then so is $KF$.
\end{lemma}

\begin{proof}
  \leanok
  A prime $q$ that is unramified in $K$ and in $F$ is unramified in $KF$ (Mathlib PR \#36843).
\end{proof}

\section{Reduction to the cyclic case}

\begin{theorem}
  \label{thm:kronecker_weber}
  \lean{kronecker_weber}
  \leanok
  \uses{lem:kw_reduce_to_prime_power}
  Let $K$ be a number field such that $K/\QQ$ is abelian. Then there exists $n$ such that $K$
  embeds into the cyclotomic field $\QQ(\zeta_n)$.
\end{theorem}

\begin{proof}
  \leanok
  Take for $A$ an algebraic closure of $K$, and choose a primitive $n$-th root of unity in $A$ for
  every $n \geq 1$. The image of $K$ in $A$ is an abelian extension of $\QQ$, so by
  \ref{lem:kw_reduce_to_prime_power} it lies in $\QQ(\zeta_n)$ for some $n \geq 1$. The field
  $\QQ(\zeta_n)$ is isomorphic to the abstract cyclotomic field of level $n$.
\end{proof}

\begin{lemma}
  \label{lem:kw_abelian_cyclic_decomp}
  \lean{IsAbelianGalois.exists_isCyclic_primePow_iSup_eq_top}
  \leanok
  Every finite abelian extension $K/F$ is the compositum of finitely many subextensions
  $C_i/F$, each of which is cyclic of prime power degree.
\end{lemma}

\begin{proof}
  \leanok
  The group $\Gal(K/F)$ is a finite abelian group, hence a direct product of cyclic groups of prime
  power order. Each factor corresponds to a cyclic subextension of prime power degree, and $K$ is
  their compositum.
\end{proof}

\begin{lemma}
  \label{lem:kw_reduce_to_prime_power}
  \lean{kw_reduce_to_prime_power}
  \leanok
  \uses{lem:kw_abelian_cyclic_decomp, lem:kw_cyclic_primePow_le_cyclotomic}
  Let $K \subseteq A$ be a number field with $K/\QQ$ abelian. Then $K \subseteq \QQ(\zeta_n)$ for
  some $n \geq 1$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:kw_abelian_cyclic_decomp}, the field $K$ is the compositum of finitely many cyclic
  subextensions $C_i$ of prime power degree. By \ref{lem:kw_cyclic_primePow_le_cyclotomic},
  each $C_i$ lies in some $\QQ(\zeta_{n_i})$. By induction on the number of factors, $K$ lies in
  $\QQ(\zeta_n)$ with $n$ the least common multiple of the $n_i$.
\end{proof}

\begin{lemma}
  \label{lem:kw_cyclic_primePow_le_cyclotomic}
  \lean{kw_cyclic_primePow_le_cyclotomic}
  \leanok
  \uses{lem:kw_reduce_to_unramified_outside_p, prop:kw_odd_prime_power, prop:kw_2_power}
  Let $p$ be a prime and $K \subseteq A$ a cyclic extension of $\QQ$ of degree $p^m$. Then
  $K \subseteq \QQ(\zeta_n)$ for some $n \geq 1$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:kw_reduce_to_unramified_outside_p}, there are a cyclic extension $F/\QQ$ of degree
  $p^k$, unramified outside $p$, and an integer $n \geq 1$ with $K\QQ(\zeta_n) = F\QQ(\zeta_n)$.
  The field $F$ lies in some $\QQ(\zeta_c)$: take $c = 1$ if $k = 0$, use
  \ref{prop:kw_2_power} if $p = 2$, and \ref{prop:kw_odd_prime_power} if $p$ is odd and
  $k > 0$. Then
  \[
    K \subseteq K\QQ(\zeta_n) = F\QQ(\zeta_n) \subseteq \QQ(\zeta_c)\QQ(\zeta_n)
      = \QQ(\zeta_{\operatorname{lcm}(c,n)}).
  \]
\end{proof}

\section{Removing the ramification away from $p$}

\subsection{Tame ramification}

In this subsection, $G$ is a finite group acting on a Dedekind domain $S$ by ring automorphisms,
$R = S^G$ is the ring of invariants, and $Q$ is a nonzero maximal ideal of $S$ with finite residue
field $S/Q$. Let $\fp = Q \cap R$ and $N\fp = \#(R/\fp)$. The decomposition group $D(Q)$ is the
stabilizer of $Q$ in $G$, and the inertia group $I(Q) \subseteq D(Q)$ is the subgroup of elements
acting trivially on $S/Q$. We say that $Q$ is \emph{tame} if $\#I(Q)$ is not zero in $S/Q$.
Every element of $D(Q)$ preserves $Q$ and $Q^2$, hence acts on the cotangent space $Q/Q^2$, which
is a one-dimensional $S/Q$-vector space.

\begin{definition}
  \label{def:tame_character}
  \lean{Ideal.tameCharacter}
  \leanok
  An element $\tau \in I(Q)$ acts trivially on $S/Q$, so it acts on $Q/Q^2$ by an $S/Q$-linear
  automorphism. The \emph{tame character} is the determinant of this action,
  \[
    \theta : I(Q) \to (S/Q)^\times, \qquad \theta(\tau) = \det\big(\tau \mid Q/Q^2\big).
  \]
  It is a group homomorphism because the determinant is multiplicative.
\end{definition}

\begin{lemma}
  \label{lem:tame_character_smul}
  \lean{Ideal.cotangentInertiaAction_apply_eq_smul}
  \leanok
  \uses{def:tame_character}
  For $\tau \in I(Q)$ and $v \in Q/Q^2$, one has $\tau v = \theta(\tau) v$.
\end{lemma}

\begin{proof}
  \leanok
  A linear endomorphism of a one-dimensional vector space is multiplication by its determinant.
\end{proof}

\begin{lemma}[Rigidity]
  \label{lem:tame_character_rigidity}
  \lean{Ideal.eq_one_of_tameCharacter_eq_one}
  \leanok
  \uses{def:tame_character}
  Assume that $G$ acts faithfully on $S$. Let $\tau \in I(Q)$ be such that the order $m$ of $\tau$
  is not zero in $S/Q$ and $\theta(\tau) = 1$. Then $\tau = 1$.
\end{lemma}

\begin{proof}
  \leanok
  Since $\theta(\tau) = 1$ and $Q/Q^2$ is one-dimensional, $\tau$ acts trivially on $Q/Q^2$, that
  is, $\tau x - x \in Q^2$ for all $x \in Q$. From the identity
  \[
    \tau(xy) - xy = (\tau x - x)\,\tau y + x\,(\tau y - y)
  \]
  and induction on $n$, one gets $\tau x - x \in Q^{n+1}$ for every $x \in Q^n$.

  We now show by induction on $n$ that $\delta = \tau x - x$ lies in $Q^n$ for every $x \in S$.
  Assume that $\delta \in Q^n$. By the previous step, $\tau^k \delta \equiv \delta \pmod{Q^{n+1}}$
  for all $k$. The sum $\sum_{k=0}^{m-1} \tau^k \delta$ telescopes to $\tau^m x - x = 0$, so
  $m\delta \in Q^{n+1}$. As $m$ is a unit modulo $Q$, it is a unit modulo $Q^{n+1}$, hence
  $\delta \in Q^{n+1}$.

  By Krull's intersection theorem, $\bigcap_n Q^n = 0$, so $\tau x = x$ for every $x \in S$.
  The action being faithful, $\tau = 1$.
\end{proof}

\begin{lemma}
  \label{lem:tame_inertia_cyclic}
  \lean{Ideal.tameCharacter_injective, Ideal.isCyclic_inertia, Ideal.isCyclic_inertia'}
  \leanok
  \uses{def:tame_character, lem:tame_character_rigidity}
  Assume that $G$ acts faithfully on $S$ and that $Q$ is tame. Then $\theta$ is injective, and the
  inertia group $I(Q)$ is cyclic.
\end{lemma}

\begin{proof}
  \leanok
  The order of every $\tau \in I(Q)$ divides $\#I(Q)$, so it is not zero in $S/Q$. Hence $\theta$
  is injective by \ref{lem:tame_character_rigidity}. Thus $I(Q)$ embeds into the multiplicative
  group of the finite field $S/Q$, and a finite subgroup of the multiplicative group of a field is
  cyclic.
\end{proof}

\begin{lemma}
  \label{lem:tame_character_frob}
  \lean{Ideal.tameCharacter_conj}
  \leanok
  \uses{def:tame_character, lem:tame_character_smul}
  Let $\sigma \in D(Q)$ be an arithmetic Frobenius, that is, $\sigma(x) \equiv x^{N\fp}
  \pmod Q$ for all $x \in S$. Then $\theta(\sigma\tau\sigma^{-1}) = \theta(\tau)^{N\fp}$ for all
  $\tau \in I(Q)$.
\end{lemma}

\begin{proof}
  \leanok
  The inertia group is normal in $D(Q)$, so $\sigma\tau\sigma^{-1} \in I(Q)$. The action of
  $\sigma$ on $Q/Q^2$ is semilinear: $\sigma(cv) = \sigma(c)\sigma(v)$ for $c \in S/Q$. Using
  \ref{lem:tame_character_smul}, for every $v \in Q/Q^2$,
  \[
    \sigma\tau\sigma^{-1} v = \sigma\big(\theta(\tau)\,\sigma^{-1} v\big)
      = \sigma(\theta(\tau))\, v = \theta(\tau)^{N\fp}\, v.
  \]
\end{proof}

\begin{lemma}
  \label{lem:abelian_tame_ramificationIdx_dvd}
  \lean{Ideal.card_inertia_dvd_card_sub_one, Ideal.card_inertia_dvd_card_sub_one'}
  \leanok
  \uses{lem:tame_inertia_cyclic, lem:tame_character_frob}
  Assume moreover that $G$ is abelian and acts faithfully on $S$, and that $Q$ is tame. Then
  $\#I(Q)$ divides $N\fp - 1$.
\end{lemma}

\begin{proof}
  \leanok
  Let $\tau$ generate the cyclic group $I(Q)$ (\ref{lem:tame_inertia_cyclic}) and let $\sigma$ be
  an arithmetic Frobenius at $Q$, which exists since $R = S^G$. As $G$ is abelian,
  $\sigma\tau\sigma^{-1} = \tau$, so \ref{lem:tame_character_frob} gives $\theta(\tau)^{N\fp - 1}
  = 1$. Since $\theta$ is injective, $\tau^{N\fp - 1} = 1$ and the order $\#I(Q)$ of $\tau$ divides
  $N\fp - 1$.
\end{proof}

When $S = \cO_L$ for a Galois extension of number fields $L/K$ and $G = \Gal(L/K)$, the order of
$I(Q)$ is the ramification index $e(Q \mid \fp)$. In particular, if $Q$ lies above the prime
number $q$ and $q \nmid \#G$, then $Q$ is tame (\texttt{Ideal.card\_inertia\_notMem\_of\_not\_dvd}).

\subsection{Abhyankar's lemma and the inertia field}

\begin{lemma}
  \label{lem:card_dvd_lcm_cyclic}
  \lean{card_dvd_lcm_of_isCyclic_of_inf_eq_bot}
  \leanok
  Let $H_1, H_2$ be subgroups of a finite cyclic group $C$ with $H_1 \cap H_2 = 1$. Then $\#C$
  divides $\operatorname{lcm}([C : H_1], [C : H_2])$.
\end{lemma}

\begin{proof}
  \leanok
  The map $C \to C/H_1 \times C/H_2$ is injective, and the exponent of the target divides the
  least common multiple of the two orders. A cyclic group has order equal to its exponent.
\end{proof}

\begin{lemma}[Tame Abhyankar]
  \label{lem:ramificationIdx_sup_dvd_lcm}
  \lean{ramificationIdx_sup_dvd_lcm}
  \leanok
  \uses{lem:tame_inertia_cyclic, lem:card_dvd_lcm_cyclic}
  Let $K/\QQ$ be a finite Galois extension, $E, F$ two Galois subextensions with $EF = K$, and
  $\fP$ a prime of $K$ above a prime number $p$ with $p \nmid [K:\QQ]$. Then
  \[
    e(\fP\mid p) \ \big|\ \operatorname{lcm}\big(e(\fP\cap E\mid p),\ e(\fP\cap F\mid p)\big).
  \]
\end{lemma}

\begin{proof}
  \leanok
  Let $G = \Gal(K/\QQ)$ and $I = I(\fP) \subseteq G$. As $p \nmid \#G$, the prime $\fP$ is tame,
  so $I$ is cyclic by \ref{lem:tame_inertia_cyclic}, of order $e(\fP\mid p)$. Write $H_E =
  \Gal(K/E)$ and $H_F = \Gal(K/F)$. The group $I \cap H_E$ is the inertia group of $\fP$ in
  $K/E$, so the multiplicativity of ramification indices in $\QQ \subseteq E \subseteq K$ gives
  $[I : I\cap H_E] = e(\fP\cap E\mid p)$, and likewise for $F$. Since $EF = K$, we have $H_E \cap
  H_F = 1$, and \ref{lem:card_dvd_lcm_cyclic} applies to $I \cap H_E$ and $I \cap H_F$.
\end{proof}

\begin{lemma}[Tame inertia complement]
  \label{lem:inertia_isComplement_fixingSubgroup}
  \lean{inertia_isComplement_fixingSubgroup}
  \leanok
  \uses{lem:abelian_tame_ramificationIdx_dvd, lem:ramificationIdx_sup_dvd_lcm}
  Let $L/\QQ$ be a finite abelian extension, and let $K, E \subseteq L$ be subfields with $KE = L$,
  $[K:\QQ] = p^m$ and $E \cong \QQ(\zeta_q)$, where $q \neq p$ are primes and $q \nmid [L:\QQ]$.
  Let $\fQ$ be a prime of $L$ above $q$. Then the inertia group $I = I(\fQ) \subseteq \Gal(L/\QQ)$
  is a complement of $\Gal(L/E)$: one has $I \cap \Gal(L/E) = 1$ and $\#I \cdot \#\Gal(L/E) =
  [L:\QQ]$.
\end{lemma}

\begin{proof}
  \leanok
  We first show that $e(\fQ\mid q) = q - 1$. By \ref{lem:ramificationIdx_sup_dvd_lcm},
  $e(\fQ\mid q)$ divides the least common multiple of $e(\fQ\cap K\mid q)$ and
  $e(\fQ\cap E\mid q) = q-1$. The extension $K/\QQ$ is abelian and $q \nmid p^m$, so
  \ref{lem:abelian_tame_ramificationIdx_dvd} gives that $e(\fQ\cap K\mid q)$ divides $q - 1$.
  Hence $e(\fQ\mid q)$ divides $q - 1$. Conversely $q - 1 = e(\fQ\cap E\mid q)$ divides
  $e(\fQ\mid q)$.

  It follows that $e(\fQ \mid \fQ\cap E) = 1$. This ramification index is the order of
  $I \cap \Gal(L/E)$, the inertia group of $\fQ$ in $L/E$, so $I \cap \Gal(L/E) = 1$. Finally
  $\#I \cdot \#\Gal(L/E) = (q-1)[L:E] = [E:\QQ][L:E] = [L:\QQ]$.
\end{proof}

\begin{lemma}[Inertia field from a complement]
  \label{lem:inertia_field_from_complement}
  \lean{IsGaloisGroup.fixedPoints_sup_eq_top_of_isComplement}
  \leanok
  Let $L/K$ be a finite Galois extension with group $G$, let $E$ be an intermediate field, and let
  $I$ be a normal subgroup of $G$ that is a complement of $\Gal(L/E)$. Then the fixed field
  $F = L^{I}$ satisfies $FE = L$, and $G/I \cong \Gal(L/E)$.
\end{lemma}

\begin{proof}
  \leanok
  By the Galois correspondence, $FE$ is the fixed field of $I \cap \Gal(L/E) = 1$, that is, $L$.
  The isomorphism $G/I \cong \Gal(L/E)$ is induced by the inclusion of $\Gal(L/E)$ in $G$.
\end{proof}

\begin{lemma}
  \label{lem:kw_ramification_reduction}
  \lean{kw_ramification_reduction}
  \leanok
  \uses{lem:inertia_isComplement_fixingSubgroup, lem:inertia_field_from_complement}
  Let $K \subseteq A$ be a cyclic extension of $\QQ$ of degree $p^m$, and let $q \neq p$ be a prime
  that ramifies in $K$. Then there is a cyclic extension $F \subseteq A$ of $\QQ$ of $p$-power
  degree such that
  \begin{itemize}
    \item the prime $q$ is unramified in $F$;
    \item every prime that is unramified in $K$ is unramified in $F$;
    \item one has $K\QQ(\zeta_q) = F\QQ(\zeta_q)$.
  \end{itemize}
\end{lemma}

\begin{proof}
  \leanok
  Put $E = \QQ(\zeta_q)$ and $L = KE$. The extension $L/\QQ$ is abelian, and $[L:\QQ]$ divides
  $[K:\QQ][E:\QQ] = p^m(q-1)$, so $q \nmid [L:\QQ]$. Fix a prime $\fQ$ of $L$ above $q$, with
  inertia group $I \subseteq \Gal(L/\QQ)$. By \ref{lem:inertia_isComplement_fixingSubgroup}, the
  group $I$ is a complement of $\Gal(L/E)$. Let $F = L^{I}$ be the inertia field of $\fQ$. By
  \ref{lem:inertia_field_from_complement}, we get $FE = L = KE$ and
  \[
    \Gal(F/\QQ) \cong \Gal(L/\QQ)/I \cong \Gal(L/E).
  \]
  Restriction to $K$ is injective on $\Gal(L/E)$, since $KE = L$. Hence $\Gal(F/\QQ)$ is
  isomorphic to a subgroup of the cyclic group $\Gal(K/\QQ)$, so it is cyclic of order dividing
  $p^m$.

  The prime $q$ is unramified in the inertia field $F$ (Mathlib PR \#36733). Let $q'$ be a prime
  unramified in $K$. Then $q' \neq q$, so $q'$ is also unramified in $E = \QQ(\zeta_q)$, hence in
  the compositum $L = KE$ (Mathlib PR \#36843), and finally in the subfield $F$ of $L$.
\end{proof}

\begin{lemma}
  \label{lem:kw_reduce_to_unramified_outside_p}
  \lean{kw_reduce_to_unramified_outside_p}
  \leanok
  \uses{lem:kw_ramification_reduction, def:unramified_outside}
  Let $K \subseteq A$ be a cyclic extension of $\QQ$ of degree $p^m$. Then there exist a cyclic
  extension $F \subseteq A$ of $\QQ$ of $p$-power degree, unramified outside $p$, and an integer
  $n \geq 1$ such that $K\QQ(\zeta_n) = F\QQ(\zeta_n)$.
\end{lemma}

\begin{proof}
  \leanok
  Let $S$ be the finite set of prime divisors of the discriminant of $K$. Every prime $q \neq p$
  outside $S$ is unramified in $K$. We prove the statement by induction on $S$, for all cyclic
  extensions $K$ of $p$-power degree in which every prime $q \neq p$ outside $S$ is unramified.

  If $S$ is empty, take $F = K$ and $n = 1$. Otherwise write $S = \{r\} \cup S'$. If $r = p$, or
  if $r$ is unramified in $K$, the induction hypothesis applies to $K$ and $S'$. Otherwise
  \ref{lem:kw_ramification_reduction} gives a cyclic extension $E$ of $p$-power degree, unramified
  at $r$ and at every prime unramified in $K$, with $K\QQ(\zeta_r) = E\QQ(\zeta_r)$. The
  induction hypothesis applied to $E$ and $S'$ gives $F$ and $n$ with $E\QQ(\zeta_n) =
  F\QQ(\zeta_n)$. Then
  \[
    K\QQ(\zeta_{\operatorname{lcm}(n,r)}) = K\QQ(\zeta_r)\QQ(\zeta_n)
      = E\QQ(\zeta_r)\QQ(\zeta_n) = F\QQ(\zeta_n)\QQ(\zeta_r)
      = F\QQ(\zeta_{\operatorname{lcm}(n,r)}).
  \]
\end{proof}

\section{Comparing cyclic extensions}

\begin{lemma}
  \label{lem:kw_cyclic_compositum}
  \lean{kw_cyclic_compositum}
  \leanok
  Let $K, K' \subseteq A$ be number fields such that $KK'/\QQ$ is Galois with cyclic Galois
  group. If $[K:\QQ]$ divides $[K':\QQ]$, then $K \subseteq K'$.
\end{lemma}

\begin{proof}
  \leanok
  Let $G = \Gal(KK'/\QQ)$. By the Galois correspondence, it suffices to show that
  $\Gal(KK'/K') \subseteq \Gal(KK'/K)$. In the cyclic group $G$, a subgroup is contained in
  another if and only if its order divides the order of the other. The orders are $[KK':K'] =
  [KK':\QQ]/[K':\QQ]$ and $[KK':K] = [KK':\QQ]/[K:\QQ]$, and the first divides the second since
  $[K:\QQ]$ divides $[K':\QQ]$.
\end{proof}

\begin{lemma}
  \label{lem:kw_fixedField_index}
  \lean{fixedField_spec_of_index_eq_prime_pow}
  \leanok
  \uses{def:unramified_outside}
  Let $C \subseteq A$ be an abelian extension of $\QQ$, unramified outside $p$, and let $H$ be a
  subgroup of $\Gal(C/\QQ)$ of index $p^m$ with cyclic quotient. Then the fixed field $C^{H}$ is a
  cyclic extension of $\QQ$ of degree $p^m$, unramified outside $p$.
\end{lemma}

\begin{proof}
  \leanok
  By the Galois correspondence, $[C^{H} : \QQ] = [\Gal(C/\QQ) : H] = p^m$ and $\Gal(C^{H}/\QQ)
  \cong \Gal(C/\QQ)/H$, which is cyclic. A subfield of $C$ is unramified outside $p$ because $C$
  is.
\end{proof}

\begin{lemma}
  \label{lem:kw_eq_of_unique_prime_subfield}
  \lean{kw_eq_of_unique_prime_subfield}
  \leanok
  \uses{lem:kw_cyclic_compositum, lem:unramified_outside_sup}
  Let $K, K' \subseteq A$ be abelian extensions of $\QQ$ of the same degree $p^m$, both unramified
  outside $p$. Assume that any two cyclic subextensions of $KK'$ of degree $p$ over $\QQ$ and
  unramified outside $p$ are equal. Then $K = K'$.
\end{lemma}

\begin{proof}
  \leanok
  The group $\Gal(KK'/\QQ)$ is an abelian $p$-group, since $[KK':\QQ]$ divides $p^{2m}$. If it
  were not cyclic, it would have two distinct subgroups of index $p$. Their fixed fields would be
  two distinct cyclic subfields of $KK'$ of degree $p$, unramified outside $p$ by
  \ref{lem:unramified_outside_sup}, a contradiction. So $KK'/\QQ$ is cyclic, and
  \ref{lem:kw_cyclic_compositum} gives $K \subseteq K'$. The degrees being equal, $K = K'$.
\end{proof}

\section{The odd case}

\subsection{The Kummer extension}

In this subsection, $p$ is an odd prime and $F$ is a number field with $F \cong \QQ(\zeta_p)$.

\begin{lemma}
  \label{lem:kw_kummer_degree}
  \lean{kw_kummer₀}
  \leanok
  Let $L/F$ be a finite extension such that $L/\QQ$ is Galois and $[L:\QQ] = p(p-1)$. Then $L/F$ is
  Galois, cyclic of degree $p$.
\end{lemma}

\begin{proof}
  \leanok
  Since $[F:\QQ] = p - 1$, the tower formula gives $[L:F] = p$. The extension $L/F$ is Galois as
  $L/\QQ$ is, and a group of prime order is cyclic.
\end{proof}

\begin{lemma}
  \label{lem:kw_kummer}
  \lean{kw_kummer}
  \leanok
  Let $L/F$ be a cyclic extension of degree $p$. Then there is a nonzero $\mu \in \cO_F$ such that
  $L$ is the splitting field of $X^p - \mu$ over $F$.
\end{lemma}

\begin{proof}
  \leanok
  The field $F$ contains the $p$-th roots of unity, so by Kummer theory $L = F(\alpha)$ with
  $\alpha^p = x \in F$. Choose a nonzero integer $d$ with $dx \in \cO_F$. Then $\mu = d^p x$ lies
  in $\cO_F$, the element $d\alpha$ is a root of $X^p - \mu$, and $L = F(d\alpha)$ is the
  splitting field of $X^p - \mu$. Finally $\mu \neq 0$, since otherwise $L = F$.
\end{proof}

For the rest of this subsection, $L/F$ is a cyclic extension of degree $p$ with $L/\QQ$ abelian,
and $\mu \in \cO_F$ is nonzero, with $X^p - \mu$ irreducible over $F$ and $L$ the splitting field
of $X^p - \mu$. For $a \in (\ZZ/p\ZZ)^\times$, we write $\sigma_a$ for the element of
$\Gal(F/\QQ)$ with $\sigma_a(\zeta) = \zeta^a$ for every $p$-th root of unity $\zeta$, and we
identify $a$ with its representative in $\{1, \dots, p-1\}$.

\begin{lemma}
  \label{lem:kw_exists_pth_root}
  \lean{exists_pth_root}
  \leanok
  There is a nonzero $\alpha \in \cO_L$ with $\alpha^p = \mu$.
\end{lemma}

\begin{proof}
  \leanok
  Take a root $\alpha \in L$ of $X^p - \mu$. It is integral since $\alpha^p = \mu$ is, and nonzero
  since $L \neq F$.
\end{proof}

\begin{lemma}
  \label{lem:kw_abelian_kummer}
  \lean{kw_abelian_kummer}
  \leanok
  For every $a \in (\ZZ/p\ZZ)^\times$, there is a nonzero $\xi \in F$ with $\sigma_a(\mu) = \xi^p
  \mu^a$.
\end{lemma}

\begin{proof}
  \leanok
  Let $\alpha \in L$ be a root of $X^p - \mu$, let $\tau$ generate $\Gal(L/F)$ and write
  $\tau(\alpha) = \zeta\alpha$, with $\zeta$ a primitive $p$-th root of unity. Extend $\sigma_a$ to
  $\tilde\sigma \in \Gal(L/\QQ)$. As $\tilde\sigma(\alpha)^p = \sigma_a(\mu) \in F$, Kummer theory
  gives $\tilde\sigma(\alpha) = c\alpha^j$ with $c \in F$ and $0 \leq j < p$. Since $L/\QQ$ is
  abelian, $\tilde\sigma\tau = \tau\tilde\sigma$. Applying both sides to $\alpha$ gives $\zeta^a c
  \alpha^j = c\zeta^j\alpha^j$, so $j = a$. Therefore $\sigma_a(\mu) = \tilde\sigma(\alpha)^p =
  c^p\mu^a$.
\end{proof}

\begin{lemma}
  \label{lem:kw_split_prime}
  \lean{kw_split_prime}
  \leanok
  \uses{lem:kw_abelian_kummer}
  Let $\fq$ be a prime of $\cO_F$ such that $p \nmid v_{\fq}(\mu)$. Then every $\sigma \in
  \Gal(F/\QQ)$ with $\sigma(\fq) = \fq$ is the identity.
\end{lemma}

\begin{proof}
  \leanok
  Write $\sigma = \sigma_a$ and $r = v_{\fq}(\mu)$. By \ref{lem:kw_abelian_kummer},
  $\sigma(\mu) = \xi^p\mu^a$. Choose a nonzero integer $d$ such that $\beta = d\xi$ lies in
  $\cO_F$; then $\sigma(d^p\mu) = \beta^p\mu^a$. Since $\sigma(\fq) = \fq$, the valuations at
  $\fq$ of $\sigma(x)$ and $x$ agree, so
  \[
    p\,v_{\fq}(d) + r = p\,v_{\fq}(\beta) + a r.
  \]
  Reducing modulo $p$ gives $r \equiv ar \pmod p$. As $p \nmid r$, this forces $a \equiv 1
  \pmod p$, that is, $\sigma = 1$.
\end{proof}

\begin{lemma}
  \label{lem:kw_mu_val_p}
  \lean{kw_mu_val_p}
  \leanok
  \uses{lem:kw_split_prime, lem:kw_exists_pth_root, def:unramified_outside}
  Assume that $L$ is unramified outside $p$. Then $p \mid v_{\fq}(\mu)$ for every prime $\fq$ of
  $\cO_F$.
\end{lemma}

\begin{proof}
  \leanok
  Let $q$ be the prime number below $\fq$.

  Assume first that $q = p$. The prime $p$ is totally ramified in $F$, so the stabilizer of $\fq$
  in $\Gal(F/\QQ)$ has order $e(\fq\mid p) f(\fq\mid p) = p - 1 > 1$, as $p$ is odd. It contains
  some $\sigma \neq 1$, and \ref{lem:kw_split_prime} gives $p \mid v_{\fq}(\mu)$.

  Assume now that $q \neq p$, and let $\fQ$ be a prime of $L$ above $\fq$. The prime $q$ is
  unramified in $L$ and in $F$, so $e(\fQ\mid\fq) = 1$. Let $\alpha \in \cO_L$ with $\alpha^p =
  \mu$ (\ref{lem:kw_exists_pth_root}). Then
  \[
    v_{\fq}(\mu) = e(\fQ\mid\fq)\, v_{\fq}(\mu) = v_{\fQ}(\mu) = v_{\fQ}(\alpha^p)
      = p\, v_{\fQ}(\alpha).
  \]
\end{proof}

\begin{lemma}
  \label{lem:kw_mu_pth_power_ideal}
  \lean{kw_mu_pth_power_ideal}
  \leanok
  \uses{lem:kw_mu_val_p}
  Assume that $L$ is unramified outside $p$. Then $(\mu) = \mathfrak{a}^p$ for some nonzero ideal
  $\mathfrak{a}$ of $\cO_F$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:kw_mu_val_p}, write $v_{\fq}(\mu) = p\,n_{\fq}$ for every prime $\fq$, and take
  $\mathfrak{a} = \prod_{\fq} \fq^{n_{\fq}}$.
\end{proof}

\subsection{The class group argument}

We keep the setting of the previous subsection.

\begin{lemma}
  \label{lem:kw_class_trivial}
  \lean{kw_class_trivial}
  \leanok
  \uses{cor:stickelberger_annihilates, lem:kw_abelian_kummer}
  Let $\mathfrak{a}$ be a nonzero ideal of $\cO_F$ with $\mathfrak{a}^p = (\mu)$. Then
  $\mathfrak{a}$ is principal.
\end{lemma}

\begin{proof}
  \leanok
  Let $c$ be the class of $\mathfrak{a}$ in $\mathrm{Cl}(F)$. Since $\mathfrak{a}^p = (\mu)$ is
  principal, $c^p = 1$.

  Let $a \in (\ZZ/p\ZZ)^\times$. By \ref{lem:kw_abelian_kummer}, $\sigma_a(\mu) = \xi^p\mu^a$ with
  $\xi \in F^\times$. Hence, as fractional ideals,
  \[
    \sigma_a(\mathfrak{a})^p = \sigma_a\big((\mu)\big) = (\xi)^p\,(\mu)^a = \big((\xi)\,
    \mathfrak{a}^a\big)^p,
  \]
  and since the group of fractional ideals is torsion-free, $\sigma_a(\mathfrak{a}) =
  (\xi)\mathfrak{a}^a$. Thus $\sigma_a(c) = c^a$.

  By \ref{cor:stickelberger_annihilates}, applied to $F \cong \QQ(\zeta_p)$,
  \[
    1 = \prod_{a \in (\ZZ/p\ZZ)^\times} \sigma_a^{-1}(c)^a
      = \prod_{a \in (\ZZ/p\ZZ)^\times} c^{a^{-1}a} = c^{p-1},
  \]
  where $\sigma_a^{-1} = \sigma_{a^{-1}}$ and $c^{a^{-1}a} = c$ because $c^p = 1$. Together with
  $c^p = 1$, this gives $c = 1$.
\end{proof}

\begin{lemma}
  \label{lem:kw_mu_unit}
  \lean{kw_mu_unit}
  \leanok
  Let $\mathfrak{a}$ be a nonzero principal ideal of $\cO_F$ with $\mathfrak{a}^p = (\mu)$. Then
  $\mu = \alpha^p\eta$ for some nonzero $\alpha \in \cO_F$ and some unit $\eta \in \cO_F^\times$.
\end{lemma}

\begin{proof}
  \leanok
  Write $\mathfrak{a} = (\alpha)$. Then $(\alpha^p) = (\mu)$, so $\mu = \alpha^p\eta$ with $\eta$
  a unit, and $\alpha \neq 0$ since $\mu \neq 0$.
\end{proof}

\begin{lemma}
  \label{lem:kw_conj_mul_eq_pow}
  \lean{kw_conj_mul_eq_pow}
  \leanok
  \uses{lem:kw_abelian_kummer}
  Let $\alpha \in \cO_F$ and $\eta \in \cO_F^\times$ with $\mu = \alpha^p\eta$. Then
  $\eta\,\sigma_{-1}(\eta)$ is a $p$-th power in $F$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:kw_abelian_kummer} with $a = p - 1$, there is $\xi \in F^\times$ with
  $\sigma_{-1}(\mu) = \xi^p\mu^{p-1} = (\xi\mu)^p\mu^{-1}$. Substituting $\mu = \alpha^p\eta$,
  \[
    \sigma_{-1}(\alpha)^p\,\sigma_{-1}(\eta) = (\xi\mu)^p\,\alpha^{-p}\,\eta^{-1},
  \]
  whence $\eta\,\sigma_{-1}(\eta) = \big(\xi\mu/(\alpha\,\sigma_{-1}(\alpha))\big)^p$.
\end{proof}

\begin{lemma}
  \label{lem:kw_exists_realUnit_torsion}
  \lean{kw_exists_realUnit_torsion}
  \leanok
  \uses{lem:kw_conj_mul_eq_pow}
  Let $\alpha \in \cO_F$ and $\eta \in \cO_F^\times$ with $\mu = \alpha^p\eta$. Then
  $\eta^4 = \zeta\,\delta^p$ for some $p$-th root of unity $\zeta$ and some nonzero $\delta \in F$.
\end{lemma}

\begin{proof}
  \leanok
  The field $F$ is a CM field, and the automorphism $\sigma_{-1}$ is its complex conjugation. Every
  unit of a CM field has a square of the form $\zeta_0\varepsilon_0$, with $\zeta_0$ a root of unity
  and $\varepsilon_0$ a real unit
  (\texttt{NumberField.IsCMField.exists\_torsion\_realunits\_pow\_two\_eq\_mul}). Squaring,
  $\eta^4 = \zeta\varepsilon$ with $\zeta = \zeta_0^2$ and $\varepsilon = \varepsilon_0^2$. The
  roots of unity of $F$ form a group of order $2p$, so $\zeta^p = \zeta_0^{2p} = 1$.

  The automorphism $\sigma_{-1}$ fixes $\varepsilon$ and inverts $\zeta$, so $\sigma_{-1}(\eta)^4 =
  \zeta^{-1}\varepsilon$. Multiplying, $\varepsilon^2 = (\eta\,\sigma_{-1}(\eta))^4$. By
  \ref{lem:kw_conj_mul_eq_pow}, $\eta\,\sigma_{-1}(\eta) = w^p$, so $\varepsilon^2 = (w^4)^p$. As
  $\gcd(2,p) = 1$, it follows that $\varepsilon = \delta^p$ for some nonzero $\delta \in F$.
\end{proof}

\begin{lemma}
  \label{lem:kw_unit_root_of_unity}
  \lean{kw_unit_root_of_unity}
  \leanok
  \uses{lem:kw_class_trivial, lem:kw_mu_unit, lem:kw_exists_realUnit_torsion}
  Let $\mathfrak{a}$ be a nonzero ideal of $\cO_F$ with $\mathfrak{a}^p = (\mu)$. Then $L$ is the
  cyclotomic field of level $p^2$, that is, $L = \QQ(\xi)$ for a primitive $p^2$-th root of unity
  $\xi \in L$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:kw_class_trivial} and \ref{lem:kw_mu_unit}, $\mu = \alpha^p\eta$ with $\alpha \neq 0$
  and $\eta$ a unit, and by \ref{lem:kw_exists_realUnit_torsion}, $\eta^4 = \zeta\delta^p$. As
  $\gcd(4,p) = 1$, the polynomial $X^p - \mu^4$ is irreducible over $F$ and $L$ is its splitting
  field. Since $\mu^4 = \zeta(\alpha^4\delta)^p$, the field $L$ is also the splitting field of
  $X^p - \zeta$. In particular $\zeta \neq 1$, since otherwise $L = F$, so $\zeta$ is a primitive
  $p$-th root of unity. A root $\xi \in L$ of $X^p - \zeta$ is then a primitive $p^2$-th root of
  unity. Finally $\QQ(\xi) \subseteq L$ and both fields have degree $p(p-1)$ over $\QQ$, so
  $L = \QQ(\xi)$.
\end{proof}

\subsection{Cyclic extensions of odd prime power degree}

\begin{proposition}
  \label{prop:kw_exponent_p}
  \lean{prop_kw_exponent_p}
  \leanok
  \uses{lem:unramified_outside_sup, lem:kw_kummer_degree, lem:kw_kummer,
    lem:kw_mu_pth_power_ideal, lem:kw_unit_root_of_unity}
  Let $p$ be an odd prime and $K \subseteq A$ a cyclic extension of $\QQ$ of degree $p$,
  unramified outside $p$. Then $K \subseteq \QQ(\zeta_{p^2})$.
\end{proposition}

\begin{proof}
  \leanok
  Put $F = \QQ(\zeta_{p^2}^p)$, so that $F \cong \QQ(\zeta_p)$, and $M = KF$. The field $F$ is
  unramified outside $p$, hence so is $M$ by \ref{lem:unramified_outside_sup}, and $M/\QQ$ is
  abelian. Since $[K:\QQ] = p$ and $[F:\QQ] = p-1$ are coprime, $K \cap F = \QQ$ and $[M:\QQ] =
  p(p-1)$. By \ref{lem:kw_kummer_degree}, the extension $M/F$ is cyclic of degree $p$, and by
  \ref{lem:kw_kummer}, $M$ is the splitting field of $X^p - \mu$ for some nonzero $\mu \in \cO_F$.
  The polynomial $X^p - \mu$ is irreducible: otherwise $\mu$ would be a $p$-th power in $F$, and
  $X^p - \mu$ would split over $F$, giving $M = F$.

  By \ref{lem:kw_mu_pth_power_ideal}, $(\mu) = \mathfrak{a}^p$ with $\mathfrak{a} \neq 0$. By
  \ref{lem:kw_unit_root_of_unity}, the field $M$ is the cyclotomic field of level $p^2$, so $M =
  \QQ(\zeta_{p^2})$. Hence $K \subseteq M = \QQ(\zeta_{p^2})$.
\end{proof}

\begin{corollary}
  \label{cor:kw_exponent_p_unique}
  \lean{prop_kw_exponent_p_eq}
  \leanok
  \uses{prop:kw_exponent_p}
  Let $p$ be an odd prime. Any two cyclic extensions $K_1, K_2 \subseteq A$ of $\QQ$ of degree
  $p$, unramified outside $p$, are equal.
\end{corollary}

\begin{proof}
  \leanok
  By \ref{prop:kw_exponent_p}, both fields lie in $\QQ(\zeta_{p^2})$, which has degree $p(p-1)$.
  So $[K_1K_2 : K_1] \leq p - 1$. As $K_2/\QQ$ is Galois, $[K_1K_2 : K_1]$ divides $[K_2:\QQ] =
  p$, hence $[K_1K_2 : K_1] = 1$ and $K_2 \subseteq K_1$. The degrees being equal, $K_1 = K_2$.
\end{proof}

\begin{proposition}
  \label{prop:kw_odd_prime_power}
  \lean{prop_kw_odd_prime_power}
  \leanok
  \uses{cor:kw_exponent_p_unique, lem:kw_eq_of_unique_prime_subfield, lem:kw_fixedField_index}
  Let $p$ be an odd prime, $m \geq 1$, and $K \subseteq A$ an abelian extension of $\QQ$ of degree
  $p^m$, unramified outside $p$. Then $K \subseteq \QQ(\zeta_{p^{m+1}})$.
\end{proposition}

\begin{proof}
  \leanok
  Put $C = \QQ(\zeta_{p^{m+1}})$. It is unramified outside $p$, and since $p$ is odd,
  $\Gal(C/\QQ) \cong (\ZZ/p^{m+1}\ZZ)^\times$ is cyclic of order $p^{m}(p-1)$. Let $\sigma$ be a
  generator and $H = \langle \sigma^{p^m}\rangle$, a subgroup of index $p^m$ with cyclic quotient.
  By \ref{lem:kw_fixedField_index}, the field $K' = C^{H}$ is a cyclic extension of $\QQ$ of
  degree $p^m$, unramified outside $p$. By \ref{cor:kw_exponent_p_unique}, the hypothesis of
  \ref{lem:kw_eq_of_unique_prime_subfield} holds, so $K = K' \subseteq C$.
\end{proof}

\section{The case $p = 2$}

The argument follows Lemmermeyer (Prop.~1.5) and does not need the Kummer theory of the odd
case.

\begin{lemma}
  \label{lem:kw_2_quadratic_discr}
  \lean{kw_2_quadratic_discr}
  \leanok
  \uses{def:unramified_outside}
  A quadratic field unramified outside $2$ has discriminant $-8$, $-4$ or $8$.
\end{lemma}

\begin{proof}
  \leanok
  Let $d$ be the discriminant. A prime number divides $d$ if and only if it ramifies, so $2$ is
  the only possible prime divisor of $d$. The discriminant of a quadratic field is a fundamental
  discriminant (Mathlib PR \#43490), so $d \equiv 0$ or $1 \pmod 4$.

  If $d \equiv 1 \pmod 4$, then $d$ is odd. By Minkowski's bound $|d| > 2$, so $d$ has a prime
  divisor, which is odd: a contradiction. Hence $d = 4m$. Since $4m$ is a fundamental
  discriminant, the integer $m$ is squarefree and $m \not\equiv 1 \pmod 4$. As the only prime
  dividing $m$ is $2$, this leaves $m \in \{-1, 2, -2\}$, that is, $d \in \{-4, 8, -8\}$.
\end{proof}

\begin{proposition}
  \label{prop:kw_2_quadratic_real_unique}
  \lean{prop_kw_2_quadratic_real_unique}
  \leanok
  \uses{lem:kw_2_quadratic_discr}
  Any two totally real quadratic fields $K_1, K_2 \subseteq A$ unramified outside $2$ are equal.
\end{proposition}

\begin{proof}
  \leanok
  By \ref{lem:kw_2_quadratic_discr}, each discriminant is $-8$, $-4$ or $8$. The discriminant of a
  totally real quadratic field is positive (Mathlib PR \#43493), so both discriminants equal $8$.
  Two quadratic fields with the same discriminant are isomorphic (Mathlib PR \#43490), and a
  quadratic field is normal over $\QQ$, so $K_1 = K_2$ as subfields of $A$.
\end{proof}

\begin{lemma}
  \label{lem:kw_maximalReal_fixedField}
  \lean{maximalReal_eq_lift_fixedField}
  \leanok
  Let $L \subseteq A$ be a CM field with complex conjugation $c \in \Gal(L/\QQ)$. Then its maximal
  real subfield $L^+$ is the fixed field of $\langle c\rangle$.
\end{lemma}

\begin{proof}
  \leanok
  An element of $L$ lies in $L^+$ if and only if it is fixed by $c$. Since $c$ has order $2$,
  being fixed by $c$ is the same as being fixed by $\langle c\rangle$.
\end{proof}

\begin{lemma}
  \label{lem:galEquivZMod_complexConj}
  \lean{IsCyclotomicExtension.Rat.galEquivZMod_ratComplexConj}
  \leanok
  Under the isomorphism $\Gal(\QQ(\zeta_n)/\QQ) \cong (\ZZ/n\ZZ)^\times$, complex conjugation
  corresponds to $-1$.
\end{lemma}

\begin{proof}
  \leanok
  Complex conjugation sends every root of unity to its inverse, so it sends $\zeta_n$ to
  $\zeta_n^{-1} = \zeta_n^{n-1}$.
\end{proof}

\begin{lemma}
  \label{lem:zmod_two_pow_units_quotient}
  \lean{ZMod.isCyclic_units_two_pow_quotient_neg_one}
  \leanok
  For every $n \geq 0$, the quotient $(\ZZ/2^{n+2}\ZZ)^\times / \langle -1 \rangle$ is cyclic.
\end{lemma}

\begin{proof}
  \leanok
  The subgroups generated by $5$ and by $-1$ are complementary in $(\ZZ/2^{n+2}\ZZ)^\times$, so the
  quotient is isomorphic to the cyclic group generated by $5$.
\end{proof}

\begin{proposition}
  \label{prop:kw_2_power_real}
  \lean{prop_kw_2_power_real}
  \leanok
  \uses{prop:kw_2_quadratic_real_unique, lem:kw_eq_of_unique_prime_subfield,
    lem:kw_fixedField_index, lem:kw_maximalReal_fixedField, lem:galEquivZMod_complexConj,
    lem:zmod_two_pow_units_quotient}
  Let $K \subseteq A$ be a totally real cyclic extension of $\QQ$ of degree $2^m$, unramified
  outside $2$. Then $K \subseteq \QQ(\zeta_{2^{m+2}})$.
\end{proposition}

\begin{proof}
  \leanok
  Put $C = \QQ(\zeta_{2^{m+2}})$, a CM field unramified outside $2$, and let $c$ be its complex
  conjugation. The subgroup $\langle c\rangle$ has order $2$, hence index $2^m$ in $\Gal(C/\QQ)$.
  By \ref{lem:galEquivZMod_complexConj}, the quotient $\Gal(C/\QQ)/\langle c\rangle$ is isomorphic
  to $(\ZZ/2^{m+2}\ZZ)^\times/\langle -1\rangle$, which is cyclic by
  \ref{lem:zmod_two_pow_units_quotient}. By \ref{lem:kw_fixedField_index} and
  \ref{lem:kw_maximalReal_fixedField}, the maximal real subfield $K' = C^+$ is a cyclic extension
  of $\QQ$ of degree $2^m$, unramified outside $2$, and it is totally real.

  The compositum $KK'$ is totally real, so every subfield of $KK'$ is totally real. By
  \ref{prop:kw_2_quadratic_real_unique}, any two quadratic subfields of $KK'$ unramified outside
  $2$ are equal. Hence \ref{lem:kw_eq_of_unique_prime_subfield} gives $K = K' \subseteq C$.
\end{proof}

\begin{proposition}
  \label{prop:kw_2_power}
  \lean{prop_kw_2_power}
  \leanok
  \uses{prop:kw_2_power_real, lem:unramified_outside_sup}
  Let $K \subseteq A$ be a cyclic extension of $\QQ$ of degree $2^m$, unramified outside $2$.
  Then $K \subseteq \QQ(\zeta_{2^{m+2}})$.
\end{proposition}

\begin{proof}
  \leanok
  If $K$ is totally real, this is \ref{prop:kw_2_power_real}. Otherwise let $L = K\QQ(i)$, with
  $i = \zeta_4$. It is a totally complex abelian extension of $\QQ$, unramified outside $2$ by
  \ref{lem:unramified_outside_sup}, of degree $2^l$ with $l \leq m+1$; it is a CM field. Let
  $M = L^+$ be its maximal real subfield, of degree $2^k$ with $k \leq l$. As $L$ is totally
  complex, $M \neq L$, so $k \leq m$. The field $M$ is abelian and unramified outside $2$.

  The field $M$ is cyclic over $\QQ$. Indeed, $\Gal(M/\QQ) \cong \Gal(L/\QQ)/\langle c\rangle$,
  where $c$ is the complex conjugation of $L$. The restriction map $\Gal(L/\QQ) \to \Gal(K/\QQ)
  \times \Gal(\QQ(i)/\QQ)$ is injective, so $\Gal(M/\QQ)$ embeds into the quotient of
  $\Gal(K/\QQ) \times \Gal(\QQ(i)/\QQ)$ by the image of $\langle c\rangle$. The restriction of $c$
  to $\QQ(i)$ generates $\Gal(\QQ(i)/\QQ)$, so this quotient is generated by the image of the
  cyclic group $\Gal(K/\QQ) \times 1$. Hence it is cyclic, and so is $\Gal(M/\QQ)$.

  By \ref{prop:kw_2_power_real}, $M \subseteq \QQ(\zeta_{2^{k+2}})$. Since $[L:M] = 2$ and
  $i \notin M$, we get $L = M\QQ(i)$. Therefore
  \[
    K \subseteq L = M\QQ(i) \subseteq \QQ(\zeta_{2^{k+2}})\QQ(\zeta_4)
      \subseteq \QQ(\zeta_{2^{m+2}}).
  \]
\end{proof}
