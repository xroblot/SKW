\chapter{Setup}

\section{The fields and the primes}

Let $p$ be a prime number and $f$ a positive integer. Set $q = p^f$.
Let $L = \QQ(\zeta_{p(q-1)})$ be the $p(q-1)$-th cyclotomic field. Inside $L$, we
consider the subfields $K = \QQ(\zeta_{q-1})$ and $F = \QQ(\zeta_p)$, and we fix a primitive
$p$-th root of unity $\zeta_p \in \cO_F$. We fix a maximal ideal $P$ of $\cO_K$ lying above $p$
and a maximal ideal $\fP$ of $\cO_L$ lying above $P$. Since the order of $p$ modulo $q - 1$ is
$f$, the residue field $\cO_K/P$ has $q$ elements.

\begin{remark}
  In the formalization, most results of Chapters 1 to 3 are proved for an arbitrary integer
  $n \geq 1$ in place of $q - 1$, with $K = \QQ(\zeta_n)$, any number field $L$ containing $K$
  and $F$, and the hypothesis that $P$ has Teichmüller lifts of order $n$
  (\ref{def:hasTeichmullerLift}). Some of them then carry hypotheses such as $1 < n$,
  $e(\fP \mid P) \geq 2$ or $\zeta_p - 1 \notin \fP^2$, which all hold in the setting above.
  The case $n = q - 1$ is obtained with \ref{lem:hasTeichmullerLift}.
\end{remark}

\section{The Teichmüller character}

\begin{definition}
  \label{def:hasTeichmullerLift}
  \lean{Ideal.HasTeichmullerLift}
  \leanok
  Let $R$ be a commutative ring, $I$ an ideal of $R$ and $n$ a positive integer. The ideal $I$
  \emph{has Teichmüller lifts of order $n$} if the reduction map
  $\mu_n(R) \to (R/I)^\times$ is bijective.
\end{definition}

\begin{lemma}
  \label{lem:hasTeichmullerLift}
  \uses{def:hasTeichmullerLift}
  \lean{Stickelberger.hasTeichmullerLift}
  \leanok
  The ideal $P$ has Teichmüller lifts of order $q - 1$.
\end{lemma}

\begin{proof}
  \leanok
  The ring $\cO_K$ contains $q - 1$ distinct $(q-1)$-th roots of unity, and the integer $q - 1$
  is not in $P$. Hence these roots stay distinct modulo $P$, so the reduction map is injective,
  and it is bijective since $(\cO_K/P)^\times$ has $q - 1$ elements.
\end{proof}

\begin{definition}
  \label{def:teichmuller}
  \uses{def:hasTeichmullerLift}
  \lean{MulChar.teichmuller}
  \leanok
  Assume that $P$ has Teichmüller lifts of order $q - 1$ (\ref{lem:hasTeichmullerLift}).
  The \emph{Teichmüller character} $\om : \cO_K/P \to \cO_K$ is the multiplicative character
  whose restriction to $(\cO_K/P)^\times$ is the inverse of the reduction map
  $\mu_{q-1}(\cO_K) \xrightarrow{\sim} (\cO_K/P)^\times$, extended by $\om(0) = 0$.
\end{definition}

\begin{lemma}
  \label{lem:mk_teichmuller}
  \uses{def:teichmuller}
  \lean{MulChar.mk_teichmuller_apply}
  \leanok
  For all $x \in \cO_K/P$, we have $\om(x) \equiv x \pmod{P}$.
\end{lemma}

\begin{proof}
  \leanok
  This is the definition for $x \neq 0$, and both sides vanish for $x = 0$.
\end{proof}

\begin{lemma}
  \label{lem:orderOf_teichmuller}
  \uses{def:teichmuller}
  \lean{MulChar.orderOf_teichmuller}
  \leanok
  The order of $\om$ as a multiplicative character is $q - 1$.
\end{lemma}

\begin{proof}
  \leanok
  The values of $\om$ on units are $(q-1)$-th roots of unity, so $\om^{q-1} = 1$. The value of
  $\om$ at the reduction of a primitive $(q-1)$-th root of unity is that root, so no smaller
  power of $\om$ is trivial.
\end{proof}

\begin{lemma}
  \label{lem:teichmuller_sum}
  \uses{lem:orderOf_teichmuller}
  \lean{Stickelberger.teichmuller_ringHomComp_zpow_ne_one}
  \leanok
  Let $a \in \ZZ$ with $(q-1) \nmid a$. The character $\om^{-a}$, viewed with values in $\cO_L$,
  is nontrivial. In particular,
  \[
    \sum_{x \in \cO_K/P} \om(x)^{-a} = 0.
  \]
\end{lemma}

\begin{proof}
  \leanok
  The character $\om^{-a}$ is nontrivial by \ref{lem:orderOf_teichmuller}, and stays nontrivial
  after composing with the injection $\cO_K \to \cO_L$. The vanishing of the sum is then
  \texttt{MulChar.sum\_eq\_zero\_of\_ne\_one}.
\end{proof}

\begin{lemma}
  \label{lem:map_ringHomComp_teichmuller_zpow_apply}
  \uses{def:teichmuller}
  \lean{MulChar.map_ringHomComp_teichmuller_zpow_apply}
  \leanok
  Let $R$ be a ring containing $\cO_K$ and let $\sigma$ be a ring endomorphism of $R$ such that
  $\sigma(\zeta_{q-1}) = \zeta_{q-1}^m$ for some $m \in \mathbb{N}$. Then
  $\sigma(\om(x)^a) = \om(x)^{am}$ for all $x \in \cO_K/P$ and $a \in \ZZ$.
\end{lemma}

\begin{proof}
  \leanok
  For $x \neq 0$, the value $\om(x)^a$ is a power $\zeta_{q-1}^k$, so
  $\sigma(\om(x)^a) = \zeta_{q-1}^{mk} = \om(x)^{am}$. Both sides vanish for $x = 0$.
\end{proof}

\section{The additive character}

\begin{definition}
  \label{def:addchar}
  \lean{AddChar.traceChar}
  \leanok
  The \emph{additive character} $\psi : \cO_K/P \to \cO_F$ is defined by
  \[
    \psi(x) = \zeta_p^{\tr_{(\cO_K/P)/(\ZZ/p\ZZ)}(x)}.
  \]
  We also view it as a character with values in $\cO_L$.
\end{definition}

\begin{lemma}
  \label{lem:addCharTrace_ne_one}
  \uses{def:addchar}
  \lean{AddChar.traceChar_ne_one}
  \leanok
  The character $\psi$ is nontrivial, i.e., $\psi \neq 1$.
\end{lemma}

\begin{proof}
  \leanok
  Since the extension $\cO_K/P$ over $\ZZ/p\ZZ$ is separable, the trace
  $\tr : \cO_K/P \to \ZZ/p\ZZ$ is nonzero, so there exists $x$ with
  $\tr(x) \neq 0$, hence $\psi(x) = \zeta_p^{\tr(x)} \neq 1$.
\end{proof}

\begin{lemma}
  \label{lem:addCharTrace_isPrimitive}
  \uses{lem:addCharTrace_ne_one}
  \lean{AddChar.isPrimitive_traceChar}
  \leanok
  The character $\psi$ is primitive.
\end{lemma}

\begin{proof}
  \leanok
  Since $\cO_K/P$ is a field, any nontrivial additive character is primitive.
  The character $\psi$ is nontrivial by \ref{lem:addCharTrace_ne_one}.
\end{proof}

\begin{lemma}
  \label{lem:addCharTrace_frob_apply}
  \uses{def:addchar}
  \lean{AddChar.traceChar_apply_pow}
  \leanok
  For all $x \in \cO_K/P$, $\psi(x^p) = \psi(x)$.
\end{lemma}

\begin{proof}
  \leanok
  The Frobenius map $x \mapsto x^p$ is a $(\ZZ/p\ZZ)$-algebra automorphism of $\cO_K/P$, so
  it preserves the trace.
\end{proof}

\begin{lemma}
  \label{lem:addCharTrace_mk_sq_eq}
  \uses{def:addchar}
  \lean{AddChar.mk_traceChar_apply_eq_one_add_smul}
  \leanok
  Let $\fQ$ be an ideal of $\cO_L$ with $\zeta_p - 1 \in \fQ$. Then
  \[
    \psi(x) \equiv 1 + \tr(x) \cdot (\zeta_p - 1) \pmod{\fQ^2}
  \]
  for all $x \in \cO_K/P$.
\end{lemma}

\begin{proof}
  \leanok
  Write $\tr(x) = a$ with $a \in \mathbb{N}$. By the binomial theorem,
  $\psi(x) = (1 + (\zeta_p - 1))^a \equiv 1 + a(\zeta_p - 1) \pmod{\fQ^2}$.
\end{proof}

\begin{lemma}
  \label{lem:addCharTrace_mk_eq_one}
  \uses{lem:addCharTrace_mk_sq_eq}
  \lean{AddChar.mk_traceChar_apply_eq_one}
  \leanok
  Let $\fQ$ be an ideal of $\cO_L$ with $\zeta_p - 1 \in \fQ$.
  Then $\psi(x) \equiv 1 \pmod{\fQ}$ for all $x \in \cO_K/P$.
\end{lemma}

\begin{proof}
  \leanok
  This follows from \ref{lem:addCharTrace_mk_sq_eq}, since $\tr(x) \cdot (\zeta_p - 1) \in \fQ$.
\end{proof}

\section{Primes above $p$}

\begin{lemma}
  \label{lem:zeta_sub_one_mem}
  \lean{Stickelberger.zeta_sub_one_mem}
  \leanok
  For every prime ideal $\fQ$ of $\cO_L$ above $p$, we have $\zeta_p - 1 \in \fQ$.
\end{lemma}

\begin{proof}
  \leanok
  The ideal $(\zeta_p - 1)$ is the only prime of $\cO_F$ above $p$, so it is
  $\fQ \cap \cO_F$.
\end{proof}

\begin{lemma}
  \label{lem:ramificationIdx_under_eq_one}
  \uses{lem:ramificationIdx_eq_p_sub_one}
  \lean{Stickelberger.ramificationIdx_under_eq_one}
  \leanok
  Every prime ideal of $\cO_L$ above $p$ is unramified over $\cO_F$.
\end{lemma}

\begin{proof}
  \leanok
  Let $\fQ$ be such a prime. By \ref{lem:ramificationIdx_eq_p_sub_one}, $e(\fQ \mid p) = p - 1$,
  and this is also the ramification index of $p$ in $F$. By multiplicativity of ramification
  indices in the tower $L/F/\QQ$, the ramification index of $\fQ$ over $\cO_F$ is $1$.
\end{proof}

\chapter{The Gauss sum}
\label{chap:gauss_sum}

Throughout this chapter and the next two, we assume that $p$ is an odd prime.

\section{Gauss and Jacobi sums}

\begin{definition}
  \label{def:gauss_sum_s}
  \uses{def:teichmuller, def:addchar}
  \lean{Stickelberger.gaussSum}
  \leanok
  For $a \in \ZZ$, the \emph{Gauss sum} is
  \[
    g_a = g(\om^{-a}, \psi) = \sum_{x \in \cO_K/P} \om(x)^{-a} \cdot \psi(x) \in \cO_L.
  \]
\end{definition}

\begin{definition}
  \label{def:jacobi_sum}
  \uses{def:teichmuller}
  \lean{Stickelberger.jacobiSum}
  \leanok
  For $a, b \in \ZZ$, the \emph{Jacobi sum} is
  \[
    J(a, b) = J(\om^{-a}, \om^{-b}) = \sum_{x + y = 1} \om(x)^{-a} \om(y)^{-b} \in \cO_K.
  \]
\end{definition}

\begin{lemma}
  \label{lem:GaussSum_periodic}
  \uses{def:gauss_sum_s, lem:orderOf_teichmuller}
  \lean{Stickelberger.gaussSum_periodic}
  \leanok
  If $(q-1) \mid k$, then for all $a \in \ZZ$, $g_{a+k} = g_a$.
\end{lemma}

\begin{proof}
  \leanok
  Since $\om^{q-1} = 1$ by \ref{lem:orderOf_teichmuller}, we have $\om^{-k} = 1$, so
  $\om^{-(a+k)} = \om^{-a}$.
\end{proof}

\begin{lemma}
  \label{lem:gaussSum_of_dvd}
  \uses{lem:GaussSum_periodic, lem:gaussSum_one_left, lem:addCharTrace_ne_one}
  \lean{Stickelberger.gaussSum_of_dvd}
  \leanok
  If $(q-1) \mid a$, then $g_a = -1$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:GaussSum_periodic}, $g_a = g_0 = g(1, \psi)$, which is $-1$ by
  \ref{lem:gaussSum_one_left} since $\psi$ is nontrivial by \ref{lem:addCharTrace_ne_one}.
\end{proof}

\begin{lemma}
  \label{lem:GaussSum_frob}
  \uses{def:gauss_sum_s, lem:addCharTrace_frob_apply}
  \lean{Stickelberger.gaussSum_frob}
  \leanok
  For all $a \in \ZZ$, $g_{pa} = g_a$.
\end{lemma}

\begin{proof}
  \leanok
  Substitute $x \mapsto x^p$, a bijection of $\cO_K/P$, in the sum defining $g_a$. We have
  $\om(x^p)^{-a} = \om(x)^{-pa}$ and $\psi(x^p) = \psi(x)$ by
  \ref{lem:addCharTrace_frob_apply}.
\end{proof}

\begin{lemma}
  \label{lem:GaussSum_mul_GaussSum}
  \uses{def:gauss_sum_s, def:jacobi_sum, lem:teichmuller_sum}
  \lean{Stickelberger.gaussSum_mul_gaussSum}
  \leanok
  If $(q-1) \nmid (a+b)$, then
  \[
    g_a \cdot g_b = g_{a+b} \cdot J(a, b).
  \]
\end{lemma}

\begin{proof}
  \leanok
  The character $\om^{-(a+b)}$ is nontrivial by \ref{lem:teichmuller_sum}, so this is
  \texttt{jacobiSum\_mul\_nontrivial} applied to $\om^{-a}$ and $\om^{-b}$.
\end{proof}

\begin{lemma}
  \label{lem:GaussSum_mul_GaussSum_neg}
  \uses{def:gauss_sum_s, lem:addCharTrace_isPrimitive, lem:teichmuller_sum}
  \lean{Stickelberger.gaussSum_mul_gaussSum_neg}
  \leanok
  If $(q-1) \nmid a$, then
  \[
    g_a \cdot g_{-a} = \om^{-a}(-1) \cdot p^f.
  \]
\end{lemma}

\begin{proof}
  \leanok
  The character $\om^{-a}$ is nontrivial by \ref{lem:teichmuller_sum} and $\psi$ is primitive
  by \ref{lem:addCharTrace_isPrimitive}. By \texttt{gaussSum\_mul\_gaussSum\_eq\_card},
  $g(\om^{-a}, \psi) \cdot g(\om^{a}, \psi^{-1}) = |\cO_K/P| = p^f$, and the change of
  variable $x \mapsto -x$ gives $g(\om^{a}, \psi^{-1}) = \om^{a}(-1) \cdot g_{-a}$.
\end{proof}

\section{Congruences}

\begin{lemma}
  \label{lem:gaussSum_mem_span_zeta_sub_one}
  \uses{def:gauss_sum_s, lem:addCharTrace_mk_eq_one, lem:teichmuller_sum}
  \lean{Stickelberger.gaussSum_mem_span_zeta_sub_one}
  \leanok
  If $(q-1) \nmid a$, then $g_a \in (\zeta_p - 1)\cO_L$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:addCharTrace_mk_eq_one}, $\psi(x) \equiv 1$ modulo $(\zeta_p - 1)$, so
  $g_a \equiv \sum_x \om(x)^{-a}$, which is $0$ by \ref{lem:teichmuller_sum}.
\end{proof}

\begin{lemma}
  \label{lem:GaussSum_mem}
  \uses{lem:gaussSum_mem_span_zeta_sub_one, lem:zeta_sub_one_mem}
  \lean{Stickelberger.gaussSum_mem}
  \leanok
  If $(q-1) \nmid a$, then $g_a \in \fP$.
\end{lemma}

\begin{proof}
  \leanok
  Combine \ref{lem:gaussSum_mem_span_zeta_sub_one} and \ref{lem:zeta_sub_one_mem}.
\end{proof}

\begin{lemma}
  \label{lem:not_isUnit_gaussSum}
  \uses{lem:gaussSum_mem_span_zeta_sub_one}
  \lean{Stickelberger.not_isUnit_gaussSum}
  \leanok
  If $(q-1) \nmid a$, then $g_a$ is not a unit of $\cO_L$.
\end{lemma}

\begin{proof}
  \leanok
  The element $\zeta_p - 1$ is not a unit of $\cO_L$, so this follows from
  \ref{lem:gaussSum_mem_span_zeta_sub_one}.
\end{proof}

\begin{lemma}
  \label{lem:norm_GaussSum}
  \uses{lem:GaussSum_mul_GaussSum_neg, lem:not_isUnit_gaussSum}
  \lean{Stickelberger.norm_gaussSum}
  \leanok
  If $(q-1) \nmid a$, then there exists $k \geq 1$ such that
  $|N_{L/\QQ}(g_a)| = p^k$.
\end{lemma}

\begin{proof}
  \leanok
  Taking norms in \ref{lem:GaussSum_mul_GaussSum_neg}, the integer $|N_{L/\QQ}(g_a)|$ divides
  a power of $p$, so it is a power $p^k$. Since $g_a$ is not a unit by
  \ref{lem:not_isUnit_gaussSum}, we have $k \geq 1$.
\end{proof}

\begin{lemma}
  \label{lem:gaussSum_ne_zero}
  \uses{lem:gaussSum_of_dvd, lem:norm_GaussSum, lem:GaussSum_mul_GaussSum}
  \lean{Stickelberger.gaussSum_ne_zero, Stickelberger.jacobiSum_ne_zero}
  \leanok
  For all $a \in \ZZ$, we have $g_a \neq 0$. If $(q-1) \nmid (a+b)$, then $J(a, b) \neq 0$.
\end{lemma}

\begin{proof}
  \leanok
  If $(q-1) \mid a$, then $g_a = -1$ by \ref{lem:gaussSum_of_dvd}. Otherwise its norm is
  nonzero by \ref{lem:norm_GaussSum}. The second claim follows from the first and
  \ref{lem:GaussSum_mul_GaussSum}.
\end{proof}

\begin{lemma}
  \label{lem:mk_sq_gausssum_eq}
  \uses{lem:addCharTrace_mk_sq_eq, lem:teichmuller_sum, lem:mk_teichmuller,
        lem:zeta_sub_one_mem, lem:ramificationIdx_eq_p_sub_one}
  \lean{Stickelberger.mk_sq_gaussSum_eq}
  \leanok
  We have
  \[
    g_1 \equiv -(\zeta_p - 1) \pmod{\fP^2}.
  \]
\end{lemma}

\begin{proof}
  \leanok
  The extension $K/\QQ$ is unramified at $p$, so $e(\fP \mid P) = p - 1 \geq 2$ by
  \ref{lem:ramificationIdx_eq_p_sub_one}, hence $P \subseteq \fP^2$. Since
  $\zeta_p - 1 \in \fP$ by \ref{lem:zeta_sub_one_mem}, \ref{lem:addCharTrace_mk_sq_eq} gives
  \[
    g_1 \equiv \sum_{x} \om(x)^{-1} + (\zeta_p-1) \sum_{x} \om(x)^{-1} \tr(x) \pmod{\fP^2}.
  \]
  The first sum vanishes by \ref{lem:teichmuller_sum} since $q - 1 \geq 2$. For the second, we
  have $\om(x) \equiv x \pmod{P}$ by \ref{lem:mk_teichmuller} and
  $\tr(x) = \sum_{i<f} x^{p^i}$, and $\sum_{x \in \cO_K/P} x^{-1} x^{p^i}$ is $-1$ for $i = 0$
  and $0$ for $0 < i < f$. Hence the second sum is $-1$ modulo $\fP^2$.
\end{proof}

\section{Galois action}

\begin{lemma}
  \label{lem:galFEquiv}
  \lean{Stickelberger.galFEquiv}
  \leanok
  Restriction to $K$ induces an isomorphism $\Gal(L/F) \cong \Gal(K/\QQ) \cong
  (\ZZ/(q-1)\ZZ)^\times$. For $\sigma \in \Gal(L/F)$, we denote by
  $\fn(\sigma) \in \{0, \ldots, q-2\}$ the representative of its image, so that
  $\sigma(\zeta_{q-1}) = \zeta_{q-1}^{\fn(\sigma)}$.
\end{lemma}

\begin{proof}
  \leanok
  Since $\gcd(q-1, p) = 1$, the fields $K$ and $F$ are linearly disjoint over $\QQ$ and
  $L = KF$.
\end{proof}

\begin{lemma}
  \label{lem:gal_gaussSum_eq_gaussSum}
  \uses{def:gauss_sum_s, lem:galFEquiv, lem:map_ringHomComp_teichmuller_zpow_apply}
  \lean{Stickelberger.gal_gaussSum_eq_gaussSum}
  \leanok
  For all $\sigma \in \Gal(L/F)$ and $a \in \ZZ$,
  \[
    \sigma(g_a) = g_{a \cdot \fn(\sigma)}.
  \]
\end{lemma}

\begin{proof}
  \leanok
  The automorphism $\sigma$ fixes $\zeta_p$, hence the values of $\psi$, and
  $\sigma(\om(x)^{-a}) = \om(x)^{-a \fn(\sigma)}$ by
  \ref{lem:map_ringHomComp_teichmuller_zpow_apply}.
\end{proof}

\begin{lemma}
  \label{lem:gauss_sum_galois_action_K}
  \uses{def:gauss_sum_s, def:addchar, lem:map_ringHomComp_teichmuller_zpow_apply}
  \lean{Stickelberger.gal_gaussSum_eq_mul_gaussSum}
  \leanok
  Let $\tau \in \Gal(L/K)$ and $e \in (\ZZ/p\ZZ)^\times$ with $\tau(\zeta_p) = \zeta_p^e$.
  Then for all $a \in \ZZ$,
  \[
    \tau(g_a) = \om(e)^{a} \cdot g_a.
  \]
\end{lemma}

\begin{proof}
  \leanok
  The automorphism $\tau$ fixes the values of $\om$ and $\tau(\psi(x)) = \psi(ex)$. The
  substitution $y = ex$ in the sum gives
  $\tau(g_a) = \sum_y \om(y/e)^{-a}\psi(y) = \om(e)^{a} g_a$.
\end{proof}

\chapter{The valuation function $s$}

\section{General properties}

\begin{definition}
  \label{def:s}
  \uses{def:gauss_sum_s}
  \lean{Stickelberger.valGauss}
  \leanok
  For $a \in \ZZ$ and a maximal ideal $\fQ$ of $\cO_L$, let $s_{\fQ}(a) \in \mathbb{N}$ be the
  multiplicity of $\fQ$ in the ideal $(g_a)$. We write $s(a) = s_{\fP}(a) = v_{\fP}(g_a)$.
\end{definition}

\begin{lemma}
  \label{lem:valGauss_periodic}
  \uses{def:s, lem:GaussSum_periodic}
  \lean{Stickelberger.valGauss_periodic}
  \leanok
  If $a \equiv b \pmod{q-1}$, then $s(a) = s(b)$.
\end{lemma}

\begin{proof}
  \leanok
  Indeed $g_a = g_b$ by \ref{lem:GaussSum_periodic}.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_eq_zero}
  \uses{def:s, lem:gaussSum_of_dvd}
  \lean{Stickelberger.valGauss_eq_zero}
  \leanok
  If $(q - 1) \mid a$, then $s(a) = 0$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:gaussSum_of_dvd}, $g_a = -1$ is a unit, so $(g_a) = \cO_L$.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_add_valGauss}
  \uses{def:s, lem:GaussSum_mul_GaussSum, lem:gaussSum_ne_zero}
  \lean{Stickelberger.valGauss_add_valGauss}
  \leanok
  If $(q-1) \nmid (a+b)$, then $s(a) + s(b) = s(a+b) + v_{\fP}(J(a, b))$.
\end{lemma}

\begin{proof}
  \leanok
  Take valuations in \ref{lem:GaussSum_mul_GaussSum}. All the elements involved are nonzero by
  \ref{lem:gaussSum_ne_zero}, so the valuations are finite.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_subadditive}
  \uses{lem:valGauss_add_valGauss, lem:valGauss_eq_zero}
  \lean{Stickelberger.valGauss_subadditive}
  \leanok
  For all $a, b \in \ZZ$, $s(a + b) \leq s(a) + s(b)$.
\end{lemma}

\begin{proof}
  \leanok
  If $(q-1) \mid (a+b)$, then $s(a+b) = 0$ by \ref{lem:valGauss_eq_zero}. Otherwise use
  \ref{lem:valGauss_add_valGauss}.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_add_valGauss_neg_self}
  \uses{def:s, lem:GaussSum_mul_GaussSum_neg, lem:gaussSum_ne_zero}
  \lean{Stickelberger.valGauss_add_valGauss_neg_self}
  \leanok
  If $(q-1) \nmid a$, then $s(a) + s(-a) = f \cdot e(\fP \mid p)$.
\end{lemma}

\begin{proof}
  \leanok
  Take valuations in \ref{lem:GaussSum_mul_GaussSum_neg}: the factor $\om^{-a}(-1)$ is a root of
  unity, hence a unit, and $v_{\fP}(p^f) = f \cdot e(\fP \mid p)$.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_add_valGauss_sub_self}
  \uses{lem:valGauss_add_valGauss_neg_self, lem:valGauss_periodic, lem:valGauss_eq_zero}
  \lean{Stickelberger.valGauss_add_valGauss_sub_self}
  \leanok
  Assume $(q-1) \mid k$. Then
  \[
    s(a) + s(k - a) = \begin{cases} 0 & \text{if } (q-1) \mid a, \\
    f \cdot e(\fP \mid p) & \text{otherwise.} \end{cases}
  \]
\end{lemma}

\begin{proof}
  \leanok
  We have $s(k - a) = s(-a)$ by \ref{lem:valGauss_periodic}. Conclude with
  \ref{lem:valGauss_eq_zero} and \ref{lem:valGauss_add_valGauss_neg_self}.
\end{proof}

\begin{lemma}
  \label{lem:s_sum}
  \uses{lem:valGauss_add_valGauss_sub_self, lem:valGauss_eq_zero}
  \lean{Stickelberger.two_mul_sum_valGauss}
  \leanok
  We have
  \[
    2 \sum_{a=0}^{q-2} s(a) = (q-2) \cdot f \cdot e(\fP \mid p).
  \]
\end{lemma}

\begin{proof}
  \leanok
  Pair the term $a$ with $q - 1 - a$. By \ref{lem:valGauss_add_valGauss_sub_self} with
  $k = q - 1$, each of the $q - 2$ values $1 \leq a \leq q-2$ gives
  $s(a) + s(q-1-a) = f \cdot e(\fP \mid p)$, and $s(0) = s(q-1) = 0$ by
  \ref{lem:valGauss_eq_zero}.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_eq_zero_of_not_liesOver}
  \uses{def:s, lem:norm_GaussSum, lem:valGauss_eq_zero}
  \lean{Stickelberger.valGauss_eq_zero_of_not_liesOver}
  \leanok
  Let $\fQ$ be a maximal ideal of $\cO_L$ not lying above $p$. Then $s_{\fQ}(a) = 0$ for all
  $a \in \ZZ$.
\end{lemma}

\begin{proof}
  \leanok
  If $(q-1) \mid a$, use \ref{lem:valGauss_eq_zero}. Otherwise, if $g_a \in \fQ$, then the norm
  of $\fQ$ divides $N_{L/\QQ}(g_a)$, which is a power of $p$ by \ref{lem:norm_GaussSum}, so
  $\fQ$ lies above $p$.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_smul}
  \uses{def:s, lem:gal_gaussSum_eq_gaussSum}
  \lean{Stickelberger.valGauss_smul}
  \leanok
  For all $\sigma \in \Gal(L/F)$ and $a \in \ZZ$,
  $s_{\sigma(\fP)}(a) = s_{\fP}(a \cdot \fn(\sigma^{-1}))$.
\end{lemma}

\begin{proof}
  \leanok
  The multiplicity of $\sigma(\fP)$ in $(g_a)$ is the multiplicity of $\fP$ in
  $(\sigma^{-1}(g_a))$, and $\sigma^{-1}(g_a) = g_{a \fn(\sigma^{-1})}$ by
  \ref{lem:gal_gaussSum_eq_gaussSum}.
\end{proof}

\section{Upper bound by the sum of digits}

\begin{lemma}
  \label{lem:valGauss_one}
  \uses{lem:GaussSum_mem, lem:mk_sq_gausssum_eq, lem:zeta_sub_one_not_mem_sq}
  \lean{Stickelberger.valGauss_one}
  \leanok
  We have $s(1) = 1$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:GaussSum_mem}, $g_1 \in \fP$, so $s(1) \geq 1$.
  By \ref{lem:mk_sq_gausssum_eq}, $g_1 \equiv -(\zeta_p - 1) \pmod{\fP^2}$, and
  $\zeta_p - 1 \notin \fP^2$ by \ref{lem:zeta_sub_one_not_mem_sq}, so $s(1) \leq 1$.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_le_self}
  \uses{lem:valGauss_subadditive, lem:valGauss_one, lem:valGauss_eq_zero}
  \lean{Stickelberger.valGauss_le_self}
  \leanok
  For all $a \in \mathbb{N}$, $s(a) \leq a$.
\end{lemma}

\begin{proof}
  \leanok
  By induction on $a$, using $s(0) = 0$ (\ref{lem:valGauss_eq_zero}),
  $s(a+1) \leq s(a) + s(1)$ (\ref{lem:valGauss_subadditive}) and $s(1) = 1$
  (\ref{lem:valGauss_one}).
\end{proof}

\begin{lemma}
  \label{lem:valGauss_le_sum_digits}
  \uses{lem:valGauss_subadditive, lem:valGauss_le_self, lem:GaussSum_frob}
  \lean{Stickelberger.valGauss_le_sum_digits}
  \leanok
  Let $a \in \mathbb{N}$ and let
  $a = a_0 + a_1 p + \cdots + a_{t-1} p^{t-1}$ be its $p$-adic expansion.
  Then
  \[
    s(a) \leq a_0 + a_1 + \cdots + a_{t-1}.
  \]
\end{lemma}

\begin{proof}
  \leanok
  By induction on the list of digits. Write $a = a_0 + p a'$. Then
  $s(a) \leq s(a_0) + s(pa') \leq a_0 + s(a')$ by \ref{lem:valGauss_subadditive},
  \ref{lem:valGauss_le_self}, and $g_{pa'} = g_{a'}$ (\ref{lem:GaussSum_frob}).
\end{proof}

\section{The digit formula}

\begin{proposition}
  \label{prop:valGauss_eq_sum_digits}
  \uses{lem:valGauss_le_sum_digits, lem:s_sum, lem:ramificationIdx_eq_p_sub_one,
        lem:zeta_sub_one_not_mem_sq}
  \lean{Stickelberger.valGauss_eq_sum_digits}
  \leanok
  Let $0 \leq a \leq q - 2$ and let
  $a = a_0 + a_1 p + \cdots + a_{f-1} p^{f-1}$, $0 \leq a_i \leq p - 1$,
  be the $p$-adic expansion of $a$. Then
  \[
    s(a) = a_0 + a_1 + \cdots + a_{f-1}.
  \]
\end{proposition}

\begin{proof}
  \leanok
  By \ref{lem:ramificationIdx_eq_p_sub_one}, $e(\fP \mid p) = p - 1$, so
  \ref{lem:s_sum} gives $\sum_{a=0}^{q-2} s(a) = (q-2)f(p-1)/2$. A direct count shows that
  the sum of the digit sums of $0, \ldots, q-2$ has the same value. Since each term satisfies
  $s(a) \leq a_0 + \cdots + a_{f-1}$ by \ref{lem:valGauss_le_sum_digits}, all these
  inequalities are equalities.
\end{proof}

\begin{lemma}
  \label{lem:sub_one_mul_sum_mod_eq}
  \lean{Nat.sub_one_mul_sum_mod_eq}
  \leanok
  Let $b > 1$, $l \geq 0$ and $0 \leq x < b^l - 1$. Then
  \[
    (b - 1) \sum_{i=0}^{l-1} (x b^i \bmod (b^l - 1)) = (b^l - 1) \cdot S_b(x),
  \]
  where $S_b(x)$ is the sum of the base $b$ digits of $x$.
\end{lemma}

\begin{proof}
  \leanok
  Modulo $b^l - 1$, multiplication by $b$ cyclically rotates the $l$ base $b$ digits of $x$.
  Hence each digit of $x$ occurs exactly once in each position in the sum, which is therefore
  $S_b(x)(1 + b + \cdots + b^{l-1})$.
\end{proof}

\begin{lemma}
  \label{lem:s_mod}
  \uses{prop:valGauss_eq_sum_digits, lem:sub_one_mul_sum_mod_eq}
  \lean{Stickelberger.mul_valGauss_eq_mul_sum}
  \leanok
  Let $m, d$ be positive integers with $q - 1 = md$, and let $a \in \mathbb{N}$ with
  $da \leq q - 2$. Then
  \[
    m \cdot s(da) = (p-1) \sum_{i=0}^{f-1} (ap^i \bmod m).
  \]
\end{lemma}

\begin{proof}
  \leanok
  Multiply both sides by $d$. Since $d (ap^i \bmod m) = dap^i \bmod (q-1)$, this follows from
  \ref{prop:valGauss_eq_sum_digits} and \ref{lem:sub_one_mul_sum_mod_eq} applied to $b = p$,
  $l = f$ and $x = da$.
\end{proof}

\chapter{Ideal factorization of Gauss sums}

\section{Setup for Stickelberger's theorem}

Let $m$ be a positive integer with $p \nmid m$, and assume that $f$ is the multiplicative order
of $p$ modulo $m$. Then $m \mid q - 1$, and we set $d = (q-1)/m$. Let $k = \QQ(\zeta_m)$, a
subfield of $L$, and let $\fp$ be a maximal ideal of $\cO_k$ lying above $p$. We assume that
$\fP$ lies above $\fp$.

Let $G = \Gal(k/\QQ) \cong (\ZZ/m\ZZ)^\times$.
For $a \in (\ZZ/m\ZZ)^\times$, let $\sigma_a \in G$ denote the element with
$\sigma_a(\zeta_m) = \zeta_m^a$. The \emph{Stickelberger element} is
\[
  \Theta = \Theta(k) = \sum_{a \in (\ZZ/m\ZZ)^\times} a \cdot \sigma_a^{-1} \in \ZZ[G],
\]
where $a$ is identified with its representative in $\{0, \ldots, m-1\}$. For an ideal
$\mathfrak{a}$ of $\cO_k$, we write
$\mathfrak{a}^{\Theta} = \prod_{a \in (\ZZ/m\ZZ)^\times} \sigma_a^{-1}(\mathfrak{a})^{a}$.
\begin{remark}
  With the notation of Washington, $\Theta = m\theta$ where
  $\theta = \sum_{(a,m)=1} \{a/m\} \sigma_a^{-1} \in \QQ[G]$.
\end{remark}

\section{Descent of $g_d^m$ to $k$}

\begin{lemma}
  \label{lem:gauss_sum_power_fixed_2}
  \uses{lem:gauss_sum_galois_action_K, lem:orderOf_teichmuller}
  \lean{Stickelberger.smul_gaussSum_eq_gaussSum}
  \leanok
  For any $\tau \in \Gal(L/K)$, we have $\tau(g_d^m) = g_d^m$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:gauss_sum_galois_action_K}, $\tau(g_d) = \om(e)^d g_d$ for some $e$, and
  $\om(e)^{dm} = \om(e)^{q-1} = 1$ by \ref{lem:orderOf_teichmuller}.
\end{proof}

\begin{lemma}
  \label{lem:gauss_sum_power_fixed_1}
  \uses{lem:gal_gaussSum_eq_gaussSum, lem:GaussSum_periodic}
  \lean{Stickelberger.smul_gaussSum_eq_gaussSum'}
  \leanok
  Let $E = \QQ(\zeta_{mp}) \subseteq L$. For any $\tau \in \Gal(L/E)$, we have
  $\tau(g_d) = g_d$.
\end{lemma}

\begin{proof}
  \leanok
  The automorphism $\tau$ fixes $F$ and $\zeta_m$, so $\fn(\tau) \equiv 1 \pmod{m}$, hence
  $d \fn(\tau) \equiv d \pmod{q-1}$. Conclude with \ref{lem:gal_gaussSum_eq_gaussSum} and
  \ref{lem:GaussSum_periodic}.
\end{proof}

\begin{lemma}
  \label{lem:gauss_sum_power_in_cyclotomic}
  \uses{lem:gauss_sum_power_fixed_1, lem:gauss_sum_power_fixed_2}
  \lean{Stickelberger.exists_mem_gaussSum_pow_eq}
  \leanok
  There exists $\Gamma \in \cO_k$ with $g_d^m = \Gamma$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:gauss_sum_power_fixed_2} and \ref{lem:gauss_sum_power_fixed_1}, the element
  $g_d^m$ lies in $K \cap E = \QQ(\zeta_{\gcd(q-1, mp)}) = k$, and it is integral.
\end{proof}

\section{Factorization in $L$}

\begin{lemma}
  \label{lem:gaussSum_factorization_L}
  \uses{def:s, lem:gaussSum_ne_zero, lem:valGauss_eq_zero_of_not_liesOver}
  \lean{Stickelberger.span_singleton_gaussSum_eq_prod}
  \leanok
  For all $a \in \ZZ$, we have the equality of ideals of $\cO_L$
  \[
    (g_a) = \prod_{\fQ \mid p} \fQ^{s_{\fQ}(a)},
  \]
  where the product runs over the primes of $\cO_L$ above $p$.
\end{lemma}

\begin{proof}
  \leanok
  The ideal $(g_a)$ is nonzero by \ref{lem:gaussSum_ne_zero}, and no prime outside $p$ divides

\end{proof}

\section{Primes of $k$ above $p$}

\begin{lemma}
  \label{lem:stabilizer_tP0}
  \lean{IsCyclotomicExtension.Rat.galEquivZMod_stabilizer}
  \leanok
  The stabilizer of $\fp$ in $G$ is the subgroup $\langle \sigma_p \rangle$, which has
  order $f$.
\end{lemma}

\begin{proof}
  \leanok
  The stabilizer of $\fp$ is its decomposition group, which acts on the residue field through
  powers of the Frobenius, and $\sigma_p$ induces the Frobenius. The two groups have order $f$.
\end{proof}

\begin{definition}
  \label{def:equivUnits}
  \uses{lem:stabilizer_tP0}
  \lean{Stickelberger.equivUnits}
  \leanok
  For each prime $\fq$ of $\cO_k$ above $p$, choose $a_{\fq} \in (\ZZ/m\ZZ)^\times$ with
  $\sigma_{a_{\fq}}^{-1}(\fp) = \fq$. The map
  \[
    \{\fq \mid p\} \times \{0, \ldots, f-1\} \to (\ZZ/m\ZZ)^\times, \qquad
    (\fq, j) \mapsto a_{\fq} p^j
  \]
  is a bijection, and the image $a$ of $(\fq, j)$ satisfies $\sigma_a^{-1}(\fp) = \fq$.
\end{definition}

\begin{proof}
  \leanok
  The group $G$ acts transitively on the primes above $p$, so $a_{\fq}$ exists. The bijectivity
  and the last claim follow from \ref{lem:stabilizer_tP0}.
\end{proof}

\begin{lemma}
  \label{lem:valuation_stickelberger}
  \uses{def:equivUnits}
  \lean{Stickelberger.sum_equivUnits}
  \leanok
  Let $\fq$ be a prime of $\cO_k$ above $p$ and $a \in (\ZZ/m\ZZ)^\times$ with
  $\sigma_a^{-1}(\fp) = \fq$. Then the sum of the (representatives of the) images of
  $(\fq, j)$, $0 \leq j < f$, under the bijection of \ref{def:equivUnits} is
  $\sum_{j=0}^{f-1} (ap^j \bmod m)$. In particular,
  $v_{\fq}(\fp^{\Theta}) = \sum_{j=0}^{f-1} (ap^j \bmod m)$.
\end{lemma}

\begin{proof}
  \leanok
  The images of the $(\fq, j)$ are the $a p^j$ up to a cyclic shift of $j$ modulo $f$. The
  second claim follows since the exponents of $\fq$ in $\fp^{\Theta}$ are the $b$ with
  $\sigma_b^{-1}(\fp) = \fq$.
\end{proof}

\section{Valuations and the factorization}

\begin{lemma}
  \label{lem:ramificationIdx_eq_sub_one}
  \uses{lem:ramificationIdx_eq_p_sub_one}
  \lean{Stickelberger.ramificationIdx_eq_sub_one}
  \leanok
  Let $\fq$ be a prime of $\cO_k$ above $p$ and $\fQ$ a prime of $\cO_L$ above $\fq$. Then
  $e(\fQ \mid \fq) = p - 1$.
\end{lemma}

\begin{proof}
  \leanok
  We have $e(\fQ \mid p) = p - 1$ by \ref{lem:ramificationIdx_eq_p_sub_one}, and $k/\QQ$ is
  unramified at $p$ since $p \nmid m$.
\end{proof}

\begin{lemma}
  \label{lem:valuation_gauss_sum_power}
  \uses{lem:s_mod, lem:valGauss_smul, lem:valGauss_periodic, lem:valuation_stickelberger,
        lem:galFEquiv}
  \lean{Stickelberger.mul_valGauss_eq}
  \leanok
  Let $\fq$ be a prime of $\cO_k$ above $p$ and $\fQ$ a prime of $\cO_L$ above $\fq$. Then
  \[
    m \cdot s_{\fQ}(d) = (p - 1) \sum_{j=0}^{f-1} b_j,
  \]
  where $b_j \in \{0, \ldots, m-1\}$ represents the image of $(\fq, j)$ under the bijection of
  \ref{def:equivUnits}.
\end{lemma}

\begin{proof}
  \leanok
  The group $\Gal(L/F)$ acts transitively on the primes of $\cO_L$ above $p$, since $F$ has a
  single prime above $p$. Choose $\tau \in \Gal(L/F)$ with $\tau(\fP) = \fQ$ and let
  $a \in (\ZZ/m\ZZ)^\times$ be the image of $\fn(\tau^{-1})$ (\ref{lem:galFEquiv}). The
  restriction of $\tau$ to $k$ is $\sigma_a^{-1}$, so $\sigma_a^{-1}(\fp) = \fq$. By
  \ref{lem:valGauss_smul} and \ref{lem:valGauss_periodic}, $s_{\fQ}(d) = s_{\fP}(da)$, and we
  conclude with \ref{lem:s_mod} and \ref{lem:valuation_stickelberger}.
\end{proof}

\begin{lemma}
  \label{lem:gauss_sum_factorization_cyclotomic}
  \uses{lem:gauss_sum_power_in_cyclotomic, lem:gaussSum_factorization_L, def:equivUnits,
        lem:ramificationIdx_eq_sub_one, lem:valuation_gauss_sum_power}
  \lean{Stickelberger.gaussSum_factorization}
  \leanok
  Let $\Gamma \in \cO_k$ with $g_d^m = \Gamma$ (\ref{lem:gauss_sum_power_in_cyclotomic}). Then
  \[
    (\Gamma) = \fp^{\Theta}
  \]
  as ideals of $\cO_k$.
\end{lemma}

\begin{proof}
  \leanok
  Since $\cO_L$ is faithfully flat over $\cO_k$, it suffices to compare the extensions of both
  sides to $\cO_L$. By \ref{lem:gaussSum_factorization_L}, the ideal $(\Gamma)\cO_L = (g_d)^m$
  is $\prod_{\fQ \mid p} \fQ^{m s_{\fQ}(d)}$. Regrouping the factors of $\fp^{\Theta}$ with
  \ref{def:equivUnits} and using $e(\fQ \mid \fq) = p - 1$
  (\ref{lem:ramificationIdx_eq_sub_one}), the extension of $\fp^{\Theta}$ is
  $\prod_{\fq \mid p} \prod_{\fQ \mid \fq} \fQ^{(p-1) \sum_j b_j}$, with the notation of
  \ref{lem:valuation_gauss_sum_power}. The exponents agree by
  \ref{lem:valuation_gauss_sum_power}.
\end{proof}

\section{Stickelberger's theorem}

\begin{lemma}
  \label{lem:stickelberger_in_L}
  \uses{lem:gauss_sum_power_in_cyclotomic, lem:gauss_sum_factorization_cyclotomic,
        lem:hasTeichmullerLift}
  \lean{Stickelberger.Stickelberger₀}
  \leanok
  In the setting of this chapter, the ideal $\fp^{\Theta}$ of $\cO_k$ is principal.
\end{lemma}

\begin{proof}
  \leanok
  Choose a maximal ideal $\fP$ of $\cO_L$ above $\fp$ and let $P = \fP \cap \cO_K$. The ideal
  $P$ has Teichmüller lifts of order $q - 1$ by \ref{lem:hasTeichmullerLift}. Take $\Gamma$ as in
  \ref{lem:gauss_sum_power_in_cyclotomic}; then $\fp^{\Theta} = (\Gamma)$ by
  \ref{lem:gauss_sum_factorization_cyclotomic}.
\end{proof}

\begin{theorem}
  \label{thm:stickelberger}
  \uses{lem:stickelberger_in_L}
  \lean{Stickelberger.Stickelberger₁}
  \leanok
  Let $k$ be a number field isomorphic to $\QQ(\zeta_m)$, let $p$ be an odd prime with
  $p \nmid m$ and let $\fp$ be a maximal ideal of $\cO_k$ above $p$. Then the ideal
  $\fp^{\Theta}$ is principal.
\end{theorem}

\begin{proof}
  \leanok
  Let $f$ be the order of $p$ modulo $m$ and $q = p^f$. The field
  $L = \QQ(\zeta_{p(q-1)})$ contains a subfield $k_0 = \QQ(\zeta_m)$, and an isomorphism
  $k_0 \cong k$ commutes with the action of the $\sigma_a$. Apply
  \ref{lem:stickelberger_in_L} to the preimage of $\fp$ in $\cO_{k_0}$ and transport the
  result to $k$.
\end{proof}

\begin{lemma}
  \label{lem:class_coprime_rep}
  \lean{ClassGroup.exists_mk0_eq_and_isCoprime}
  \leanok
  Let $R$ be a Dedekind domain and $J$ a nonzero ideal of $R$. Every ideal class of $R$
  contains an integral ideal coprime to $J$.
\end{lemma}

\begin{proof}
  \leanok
  Let $I_0$ be an integral ideal in the inverse class. By the Chinese remainder theorem, there is
  $x \in I_0$ such that $(x) I_0^{-1}$ is coprime to $J$, and this ideal lies in the given class.
\end{proof}

\begin{corollary}
  \label{cor:stickelberger_annihilates}
  \uses{thm:stickelberger, lem:class_coprime_rep}
  \lean{Stickelberger.Stickelberger}
  \leanok
  Let $k$ be a number field isomorphic to $\QQ(\zeta_m)$. The Stickelberger element $\Theta$
  annihilates the ideal class group $\mathrm{Cl}(k)$: for every ideal class $c$,
  \[
    c^{\Theta} = \prod_{a \in (\ZZ/m\ZZ)^\times} \sigma_a^{-1}(c)^{a} = 1.
  \]
\end{corollary}

\begin{proof}
  \leanok
  By \ref{lem:class_coprime_rep}, choose an integral ideal $\mathfrak{b} \in c$ coprime to
  $2m$. Every prime factor of $\mathfrak{b}$ lies above an odd prime not dividing $m$, so
  \ref{thm:stickelberger} applied to each prime factor shows that $\mathfrak{b}^{\Theta}$ is
  principal. Hence $c^{\Theta} = [\mathfrak{b}^{\Theta}] = 1$.
\end{proof}

\begin{remark}
  Mathematically, coprimality of the representative to $m$ alone suffices for
  Stickelberger's theorem. We additionally require the norm to be odd (i.e.\
  coprimality to $2m$) only because the formalized form of \ref{thm:stickelberger}
  is proved for primes of odd residue characteristic.
\end{remark}

\chapter{The Kronecker-Weber theorem}

\section{Reduction to the prime degree case}

Throughout this chapter, $\zeta_m$ denotes a primitive $m$-th root of unity.
We say that an extension is \emph{unramified} if it is unramified at all
finite primes. A normal extension $K/F$ has \emph{exponent} $m$ if
$\Gal(K/F)$ has exponent $m$.

The goal of this chapter is to prove the following theorem.

\begin{theorem}
  \label{thm:kronecker_weber}
  \lean{kronecker_weber}
  \leanok
  \uses{lem:kw_reduce_to_prime_power, prop:kw_odd_prime_power, prop:kw_2_power}
  Every abelian extension of $\QQ$ is contained in a cyclotomic field.
\end{theorem}

\begin{proof}
  \leanok
  By \ref{lem:kw_reduce_to_prime_power}, it suffices to prove that every cyclic
  extension of $\QQ$ of prime power degree $p^m$ unramified outside $p$ is
  contained in a cyclotomic field. For odd $p$, this is
  \ref{prop:kw_odd_prime_power}; for $p = 2$, this is \ref{prop:kw_2_power}.
\end{proof}

\begin{lemma}
  \label{lem:kw_abelian_cyclic_decomp}
  \lean{IsAbelianGalois.exists_isCyclic_primePow_iSup_eq_top}
  \leanok
  Every finite abelian extension $K/\QQ$ is the compositum of cyclic
  extensions of prime power degree.
\end{lemma}

\begin{proof}
  \leanok
  The Galois group $\Gal(K/\QQ)$ is a finite abelian group, hence a direct
  product of cyclic groups of prime power order. Each factor corresponds to
  a cyclic subextension of prime power degree, and $K$ is their compositum.
\end{proof}

\begin{lemma}
  \label{lem:kw_reduce_to_unramified_outside_p}
  \lean{kw_reduce_to_unramified_outside_p}
  \leanok
  \uses{lem:kw_ramification_reduction}
  Let $K/\QQ$ be a cyclic extension of $p$-power degree. Then there
  exists a cyclic extension $F/\QQ$ of $p$-power degree unramified
  outside $p$ such that $K$ is cyclotomic if and only if $F$ is cyclotomic.
\end{lemma}

\begin{proof}
  \leanok
  The discriminant of $K$ is a finite integer, so only finitely many primes
  $q \neq p$ are ramified in $K/\QQ$. We apply \ref{lem:kw_ramification_reduction}
  iteratively. At each step, given a cyclic $p$-power-degree extension $K'/\QQ$
  with some $q \neq p$ ramified, \ref{lem:kw_ramification_reduction} provides
  a cyclotomic extension $\Lambda/\QQ$ and a cyclic $p$-power-degree extension
  $K''/\QQ$ in which $q$ is unramified, such that $K'\Lambda = K''\Lambda$.
  Since $\Lambda$ is cyclotomic, $K'$ is cyclotomic if and only if $K''$ is
  (as $K'\Lambda = K''\Lambda$ and $\Lambda$ is cyclotomic).
  Since $K'' \subseteq K'\Lambda$ and $\Lambda$ is only ramified at $q$, the
  primes ramified in $K''$ are a subset of those of $K'$ union $\{q\}$, which
  equals the primes of $K'$. Since $q$ is unramified in $K''$ by construction,
  the set of primes $\neq p$ ramified in $K''$ is strictly smaller than that of
  $K'$. The process terminates with the desired $F$.
\end{proof}

\begin{lemma}
  \label{lem:kw_cyclic_primePow_le_cyclotomic}
  \lean{kw_cyclic_primePow_le_cyclotomic}
  \leanok
  \uses{lem:kw_reduce_to_unramified_outside_p, prop:kw_odd_prime_power, prop:kw_2_power}
  Every cyclic extension of $\QQ$ of prime power degree $p^m$ is contained in a cyclotomic field.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:kw_reduce_to_unramified_outside_p} there is a cyclic $p$-power extension $F$
  unramified outside $p$ with $K\,\QQ(\zeta_n) = F\,\QQ(\zeta_n)$ for some $n > 0$. By
  \ref{prop:kw_odd_prime_power} (odd $p$) or \ref{prop:kw_2_power} ($p = 2$), $F$ is cyclotomic,
  say $F \subseteq \QQ(\zeta_c)$. Then $K \subseteq K\,\QQ(\zeta_n) = F\,\QQ(\zeta_n) \subseteq
  \QQ(\zeta_{\operatorname{lcm}(c,n)})$.
\end{proof}

\begin{lemma}
  \label{lem:kw_reduce_to_prime_power}
  \lean{kw_reduce_to_prime_power}
  \leanok
  \uses{lem:kw_abelian_cyclic_decomp, lem:kw_cyclic_primePow_le_cyclotomic}
  It suffices to prove that every cyclic extension of $\QQ$ of prime power
  degree unramified outside $p$ is cyclotomic.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:kw_abelian_cyclic_decomp}, any finite abelian extension of
  $\QQ$ is a compositum of cyclic extensions of prime power degree. By
  \ref{lem:kw_cyclic_primePow_le_cyclotomic}, each such extension is cyclotomic, and a
  compositum of cyclotomic fields is cyclotomic.
\end{proof}

\begin{lemma}
  \label{lem:kw_cyclic_compositum}
  \lean{kw_cyclic_compositum}
  \leanok
  Let $L/\QQ$ be a Galois extension with cyclic Galois group, and let
  $K, K'$ be intermediate fields with $[K:\QQ] \mid [K':\QQ]$. Then
  $K \subseteq K'$.
\end{lemma}

\begin{proof}
  \leanok
  Via the Galois correspondence, $K \subseteq K'$ is equivalent to
  $\mathrm{Fix}(K') \leq \mathrm{Fix}(K)$ in $\Gal(L/\QQ)$. In a cyclic
  group, the subgroup lattice is totally ordered by divisibility of indices,
  and $[\Gal(L/\QQ) : \mathrm{Fix}(K)] = [K:\QQ]$ divides
  $[\Gal(L/\QQ) : \mathrm{Fix}(K')] = [K':\QQ]$, so $\mathrm{Fix}(K') \leq \mathrm{Fix}(K)$.
\end{proof}

Throughout this subsection we fix the following local data, and develop the tame character
from scratch (no higher ramification groups). Let $L/K$ be a finite Galois extension of number
fields, $\mathfrak{Q}$ a prime of $L$ above a prime $\mathfrak{p}$ of $K$. Localize at
$\mathfrak{Q}$: let $\mathcal{O} = \mathcal{O}_{L,\mathfrak{Q}}$ be the (discrete) valuation ring,
$\mathfrak{m} = \mathfrak{Q}\mathcal{O}$ its maximal ideal, $v = v_{\mathfrak{Q}}$ the valuation,
$k = k(\mathfrak{Q}) = \mathcal{O}/\mathfrak{m}$ the residue field of characteristic $\ell$, and
$e = e(\mathfrak{Q}\mid\mathfrak{p})$. Let $I = I(\mathfrak{Q}\mid\mathfrak{p})$ be the inertia
group; by definition each $\tau \in I$ fixes $\mathfrak{Q}$ (so $v\circ\tau = v$, i.e.\ $\tau$
preserves $\mathcal{O}$, $\mathfrak{m}$ and each $\mathfrak{m}^n$) and acts trivially on $k$. For
$x \in \mathcal{O}$ we write $\bar x$ for its residue in $k$.

\begin{definition}
  \label{def:tame_character}
  \lean{Ideal.tameCharacter}
  \leanok
  The cotangent space $\mathfrak{m}/\mathfrak{m}^2$ is a one-dimensional $k$-vector space. Each
  $\tau \in I$ preserves $\mathfrak{m}$ and $\mathfrak{m}^2$ and acts trivially on $k$, so it induces
  a $k$-linear automorphism of $\mathfrak{m}/\mathfrak{m}^2$. The \emph{tame character} is its
  determinant,
  \[
    \theta : I \to k^\times, \qquad \theta(\tau) = \det\!\big(\tau \mid \mathfrak{m}/\mathfrak{m}^2\big).
  \]
  Being a determinant, $\theta$ is a group homomorphism by construction. Since
  $\mathfrak{m}/\mathfrak{m}^2$ is one-dimensional, $\theta(\tau)$ is the scalar by which $\tau$ acts;
  for a uniformizer $\pi$ this recovers $\theta(\tau) = \overline{\tau(\pi)/\pi}$.
\end{definition}

\begin{lemma}
  \label{lem:tame_character_smul}
  \lean{Ideal.cotangentInertiaAction_apply_eq_smul}
  \leanok
  \uses{def:tame_character}
  For $\tau \in I$ and $\bar x \in \mathfrak{m}/\mathfrak{m}^2$ one has $\tau\,\bar x = \theta(\tau)\,\bar x$.
\end{lemma}

\begin{proof}
  \leanok
  A $k$-linear endomorphism of the one-dimensional $k$-vector space $\mathfrak{m}/\mathfrak{m}^2$ is
  multiplication by its determinant, and that determinant is $\theta(\tau)$ by definition.
\end{proof}

\begin{lemma}[Rigidity]
  \label{lem:tame_character_rigidity}
  \lean{Ideal.eq_one_of_tameCharacter_eq_one}
  \leanok
  \uses{def:tame_character}
  If $\tau \in I$ has order coprime to $\ell$ and $\theta(\tau) = 1$, then $\tau = \mathrm{id}$.
\end{lemma}

\begin{proof}
  \leanok
  Let $m = \operatorname{ord}(\tau)$, coprime to $\ell$. We show by induction on $n \geq 1$ that
  $\tau$ acts trivially on $\mathcal{O}/\mathfrak{m}^n$; since $\bigcap_n \mathfrak{m}^n = 0$
  (Krull intersection), this forces $\tau = \mathrm{id}$ on $\mathcal{O}$, hence on $L$.

  For $n = 1$, $\tau$ is trivial on $\mathcal{O}/\mathfrak{m} = k$ ($\tau \in I$). For $n = 2$,
  $\theta(\tau) = 1$ means $\tau(\pi) \equiv \pi \pmod{\mathfrak{m}^2}$; as $\mathcal{O}/\mathfrak{m}^2$
  is generated as a ring by $k$ and $\bar\pi$, $\tau$ is trivial on $\mathcal{O}/\mathfrak{m}^2$.

  Assume $n \geq 3$ and $\tau \equiv \mathrm{id}$ on $\mathcal{O}/\mathfrak{m}^{n-1}$. Set $R =
  \mathcal{O}/\mathfrak{m}^n$ and $J = \mathfrak{m}^{n-1}/\mathfrak{m}^n \subseteq R$, a
  one-dimensional $k$-vector space (spanned by $\bar\pi^{\,n-1}$) with $J^2 = 0$. The hypothesis
  gives $\delta := \tau - \mathrm{id} : R \to J$. As $\tau$ is a ring homomorphism and $\delta$
  takes values in the square-zero ideal $J$,
  \[
    \delta(xy) = \tau(x)\tau(y) - xy = x\,\delta(y) + y\,\delta(x) + \delta(x)\delta(y)
      = x\,\delta(y) + y\,\delta(x),
  \]
  so $\delta$ is a derivation. Moreover $\tau$ acts on $J$ by multiplication by $\theta(\tau)^{n-1}
  = 1$ (since $\tau(\pi^{n-1}) = (\tau\pi)^{n-1} \equiv \theta(\tau)^{n-1}\pi^{n-1}
  \pmod{\mathfrak{m}^n}$), so $\delta|_J = 0$, whence $\delta^2 = 0$. Therefore $\tau^m =
  (\mathrm{id} + \delta)^m = \mathrm{id} + m\,\delta$. From $\tau^m = \mathrm{id}$ we get
  $m\,\delta = 0$; as $J$ is a $k$-vector space and $m$ is invertible in $k$ ($m$ coprime to
  $\ell = \operatorname{char} k$), $\delta = 0$, i.e.\ $\tau \equiv \mathrm{id}$ on
  $\mathcal{O}/\mathfrak{m}^n$.
\end{proof}

\begin{lemma}
  \label{lem:tame_inertia_cyclic}
  \lean{Ideal.isCyclic_inertia}
  \leanok
  \uses{def:tame_character, lem:tame_character_rigidity}
  If $\mathfrak{Q}\mid\mathfrak{p}$ is tamely ramified (i.e.\ $\ell \nmid e$), then $\theta$ is
  injective; consequently $I(\mathfrak{Q}\mid\mathfrak{p})$ is cyclic of order $e$.
\end{lemma}

\begin{proof}
  \leanok
  By construction $\theta$ is a homomorphism (\ref{def:tame_character}). Since $\ell \nmid e = |I|$, every
  $\tau \in I$ has order dividing $e$, hence coprime to $\ell$; so if $\theta(\tau) = 1$ then
  $\tau = \mathrm{id}$ by \ref{lem:tame_character_rigidity}. Thus $\theta$ is injective and $I$
  embeds in $k^\times$. A finite subgroup of the multiplicative group of a field is cyclic, so
  $I \cong \theta(I)$ is cyclic of order $|I| = e$ (with $\theta(I) = \mu_e(k)$).
\end{proof}

\begin{lemma}
  \label{lem:tame_character_frob}
  \lean{Ideal.tameCharacter_conj}
  \leanok
  \uses{def:tame_character, lem:tame_character_smul}
  Let $\sigma$ lie in the decomposition group $D(\mathfrak{Q}\mid\mathfrak{p})$ and reduce to the
  Frobenius $x \mapsto x^{N\mathfrak{p}}$ of $k/k(\mathfrak{p})$. Then for all $\tau \in I$,
  $\theta(\sigma\tau\sigma^{-1}) = \theta(\tau)^{N\mathfrak{p}}$.
\end{lemma}

\begin{proof}
  \leanok
  Write $\rho = \sigma\tau\sigma^{-1} \in I$ (inertia is normal in $D$). The element $\sigma$
  preserves $\mathfrak{m}$ and $\mathfrak{m}^2$, hence acts on $\mathfrak{m}/\mathfrak{m}^2$, but
  $\bar\sigma$-semilinearly: $\sigma(c\,\bar v) = \bar\sigma(c)\,\sigma(\bar v)$ for $c \in k$, where
  $\bar\sigma$ is the action of $\sigma$ on $k$, namely the $N\mathfrak{p}$-power Frobenius. Using
  \ref{lem:tame_character_smul} (that $\tau$ acts by the scalar $\theta(\tau)$) and writing
  $\rho = \sigma \circ \tau \circ \sigma^{-1}$ on $\mathfrak{m}/\mathfrak{m}^2$, for any
  $\bar v$ we get
  \[
    \rho\,\bar v = \sigma\big(\tau(\sigma^{-1}\bar v)\big)
      = \sigma\big(\theta(\tau)\,\sigma^{-1}\bar v\big)
      = \bar\sigma(\theta(\tau))\,\bar v .
  \]
  Thus $\rho$ acts on the one-dimensional space $\mathfrak{m}/\mathfrak{m}^2$ by the scalar
  $\bar\sigma(\theta(\tau))$, so $\theta(\sigma\tau\sigma^{-1}) = \bar\sigma(\theta(\tau)) =
  \theta(\tau)^{N\mathfrak{p}}$.
\end{proof}

\begin{lemma}
  \label{lem:abelian_tame_ramificationIdx_dvd}
  \lean{Ideal.card_inertia_dvd_card_sub_one}
  \leanok
  \uses{lem:tame_inertia_cyclic, lem:tame_character_frob}
  Let $L/K$ be a finite \emph{abelian} extension of number fields and
  $\mathfrak{Q}\mid\mathfrak{p}$ a tamely ramified prime. Then $e(\mathfrak{Q}\mid\mathfrak{p})$
  divides $N\mathfrak{p} - 1 = \#k(\mathfrak{p}) - 1$.
\end{lemma}

\begin{proof}
  \leanok
  Let $\tau$ generate $I$ (cyclic by \ref{lem:tame_inertia_cyclic}) and let $\sigma \in
  D(\mathfrak{Q}\mid\mathfrak{p})$ reduce to Frobenius. As $\sigma$ normalizes $I$,
  $\sigma\tau\sigma^{-1} \in I$, and by \ref{lem:tame_character_frob} together with injectivity
  of $\theta$ (\ref{lem:tame_inertia_cyclic}), $\theta(\sigma\tau\sigma^{-1}) =
  \theta(\tau)^{N\mathfrak{p}} = \theta(\tau^{N\mathfrak{p}})$ gives $\sigma\tau\sigma^{-1} =
  \tau^{N\mathfrak{p}}$. Since $L/K$ is abelian, $\sigma\tau\sigma^{-1} = \tau$, so
  $\tau^{N\mathfrak{p}-1} = 1$ and $e = \operatorname{ord}(\tau)$ divides $N\mathfrak{p} - 1$.
\end{proof}

\begin{lemma}[Tame Abhyankar]
  \label{lem:ramificationIdx_sup_dvd_lcm}
  \lean{ramificationIdx_sup_dvd_lcm}
  \leanok
  \uses{lem:tame_inertia_cyclic}
  Let $K/\QQ$ be a finite Galois extension, $E, F$ two Galois intermediate fields with $EF = K$,
  and $\mathfrak{P}$ a prime of $K$ above a rational prime $p$ that is tame in $K$
  (i.e.\ $p \nmid [K:\QQ]$). Then
  \[
    e(\mathfrak{P}\mid p) \ \big|\ \operatorname{lcm}\big(e(\mathfrak{P}\cap E\mid p),\ e(\mathfrak{P}\cap F\mid p)\big).
  \]
\end{lemma}

\begin{proof}
  \leanok
  Let $G = \Gal(K/\QQ)$ and $I = I(\mathfrak{P}\mid p) \subseteq G$ the inertia group. As
  $p \nmid [K:\QQ]$, the prime is tame, so $I$ is cyclic by \ref{lem:tame_inertia_cyclic} and
  $|I| = e(\mathfrak{P}\mid p)$. Write $H_E = \Gal(K/E)$ and $H_F = \Gal(K/F)$. Under the inclusion,
  $I \cap H_E$ is the inertia of $\mathfrak{P}$ in $\Gal(K/E)$, so the ramification tower for
  $\QQ \subseteq E \subseteq K$ gives $\#(I\cap H_E)\cdot e(\mathfrak{P}\cap E\mid p) = |I|$, i.e.\
  $[I : I\cap H_E] = e(\mathfrak{P}\cap E\mid p)$; likewise for $F$. Since $EF = K$ we have
  $H_E \cap H_F = 1$, so the subgroups $I\cap H_E$ and $I\cap H_F$ of the cyclic group $I$ meet
  trivially, whence $|I|$ divides $\operatorname{lcm}$ of their indices.
\end{proof}

\begin{lemma}[Tame inertia complement]
  \label{lem:inertia_isComplement_fixingSubgroup}
  \lean{inertia_isComplement_fixingSubgroup}
  \leanok
  \uses{lem:tame_inertia_cyclic, lem:abelian_tame_ramificationIdx_dvd, lem:ramificationIdx_sup_dvd_lcm}
  Let $L/\QQ$ be a finite abelian extension containing $E = \QQ(\zeta_q)$ (totally ramified at $q$,
  so $e_q(E) = q-1$) and a subfield $K$ of $p$-power degree with $KE = L$. For a prime
  $\mathfrak{Q}\mid q$ of $L$ (tame, since $q \nmid [L:\QQ]$), the inertia group
  $I = I_q(L) \subseteq \Gal(L/\QQ)$ is a complement of $\Gal(L/E)$: one has
  $I \cap \Gal(L/E) = 1$ and $|I|\cdot|\Gal(L/E)| = |\Gal(L/\QQ)|$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:tame_inertia_cyclic}, $I$ is cyclic with $|I| = e_q(L)$. The Abhyankar bound
  \ref{lem:ramificationIdx_sup_dvd_lcm} gives $e_q(L) \mid \operatorname{lcm}(e_q(K), e_q(E))$;
  here $e_q(K) = p^s$ is a $p$-power and, as $K/\QQ$ is abelian,
  \ref{lem:abelian_tame_ramificationIdx_dvd} gives $p^s \mid q-1 = e_q(E)$, so the lcm is $q-1$.
  With $e_q(E) \mid e_q(L)$ from the ramification tower $E \subseteq L$ we conclude
  $e_q(L) = q-1 = e_q(E)$, i.e.\ $q$ is unramified in $L/E$, which is exactly $I \cap \Gal(L/E) = 1$.
  The order identity is the ramification tower $e_q(L) = e_q(E)\cdot e(L/E)$ together with
  $|\Gal(L/E)| = [L:E]$.
\end{proof}

\begin{lemma}[Inertia field from a complement]
  \label{lem:inertia_field_from_complement}
  \lean{IsGaloisGroup.fixedPoints_sup_eq_top_of_isComplement}
  \leanok
  Let $G = \Gal(L/\QQ)$, let $I \trianglelefteq G$ be a normal subgroup complemented by $\Gal(L/E)$
  for an intermediate field $E$. Then the fixed field $F = L^{I}$ satisfies $FE = L$ and
  $G/I \cong \Gal(L/E)$.
\end{lemma}

\begin{proof}
  \leanok
  The identity $FE = L$ is the Galois correspondence sending intersection of subgroups to
  compositum of fixed fields: $F = L^{I}$ and $E = L^{\Gal(L/E)}$, so
  $FE = L^{I \cap \Gal(L/E)} = L^{1} = L$. The isomorphism $G/I \cong \Gal(L/E)$ is the quotient
  map attached to the complement.
\end{proof}

\begin{lemma}
  \label{lem:kw_ramification_reduction}
  \lean{kw_ramification_reduction}
  \leanok
  \uses{lem:inertia_isComplement_fixingSubgroup, lem:inertia_field_from_complement}
  Let $K/\QQ$ be a cyclic extension of $p$-power degree. If $q \neq p$ is ramified in $K/\QQ$,
  then there exist a cyclic cyclotomic extension $\Lambda/\QQ$ and a cyclic extension $F/\QQ$
  of $p$-power degree in which $q$ is unramified, such that $K\Lambda = F\Lambda$ and every
  prime ramified in $F$ is ramified in $K$.
\end{lemma}

\begin{proof}
  \leanok
  Write $[K:\QQ] = p^m$, put $E = \QQ(\zeta_q)$ and $L = KE$, and fix a prime $\mathfrak{Q}\mid q$
  of $L$ with inertia $I = I_q(L) \subseteq \Gal(L/\QQ)$. By
  \ref{lem:inertia_isComplement_fixingSubgroup}, $I$ is a complement of $\Gal(L/E)$. Let
  $F = L^{I}$ be the inertia field of $q$ in $L/\QQ$. By \ref{lem:inertia_field_from_complement},
  $FE = L = KE$ and
  \[
    \Gal(F/\QQ) \cong \Gal(L/\QQ)/I \cong \Gal(L/E) \cong \Gal\!\big(K/(K\cap E)\big),
  \]
  a subgroup of the cyclic $\Gal(K/\QQ)$, hence cyclic of $p$-power degree. Finally $F = L^{I}$ is
  the inertia field, so $q$ is unramified in $F/\QQ$; and $F \subseteq L = KE$ with
  $E = \QQ(\zeta_q)$ ramified only at $q$, so every prime ramified in $F$ is ramified in $K$.
  Taking $\Lambda = E$ gives the claim.
\end{proof}

\begin{lemma}
  \label{lem:kw_2_quadratic_discr}
  \lean{kw_2_quadratic_discr}
  \leanok
  A quadratic field unramified outside $2$ has discriminant $-8$, $-4$ or $8$; the three fields
  are $\QQ(\sqrt{-2})$, $\QQ(i)$ and $\QQ(\sqrt 2)$, all contained in $\QQ(\zeta_8)$.
\end{lemma}

\begin{proof}
  \leanok
  A prime divides the discriminant of a number field iff it ramifies, so every prime dividing
  $\operatorname{disc} K$ is $2$. Moreover $\operatorname{disc} K$ is a fundamental discriminant,
  hence congruent to $0$ or $1$ modulo $4$. If $\operatorname{disc} K \equiv 1 \pmod 4$ then it is
  odd, and since $|\operatorname{disc} K| \neq 1$ (Minkowski) it has a prime divisor, necessarily
  odd, a contradiction. So $\operatorname{disc} K = 4m$, and being fundamental forces $m$ to be
  squarefree with $m \not\equiv 1 \pmod 4$; as the only prime dividing $m$ is $2$, this leaves
  $m \in \{-1, 2, -2\}$, that is $\operatorname{disc} K \in \{-4, 8, -8\}$.
\end{proof}

\begin{proposition}
  \label{prop:kw_2_quadratic_real_unique}
  \lean{prop_kw_2_quadratic_real_unique}
  \leanok
  \uses{lem:kw_2_quadratic_discr}
  Any two totally real quadratic extensions of $\QQ$ unramified outside $2$ coincide (both equal
  $\QQ(\sqrt 2) = \QQ(\zeta_8)^+$).
\end{proposition}

\begin{proof}
  \leanok
  By \ref{lem:kw_2_quadratic_discr} each discriminant is $-8$, $-4$ or $8$, and it is positive

  of quadratic fields, the two fields are isomorphic over $\QQ$; being normal, they are then equal
  as subfields of the ambient field.
\end{proof}

\begin{proposition}
  \label{prop:kw_2_power_real}
  \lean{prop_kw_2_power_real}
  \leanok
  \uses{prop:kw_2_quadratic_real_unique, lem:kw_eq_of_unique_prime_subfield,
    lem:kw_fixedField_index, lem:kw_maximalReal_fixedField}
  Let $K/\QQ$ be a totally real cyclic extension of degree $2^m$ unramified outside $2$. Then
  $K \subseteq \QQ(\zeta_{2^{m+2}})$.
\end{proposition}

\begin{proof}
  \leanok
  Let $K' = \QQ(\zeta_{2^{m+2}})^+$ be the maximal real subfield of $\QQ(\zeta_{2^{m+2}})$; since
  $\Gal(\QQ(\zeta_{2^{m+2}})/\QQ) \cong \ZZ/2\ZZ \times \ZZ/2^m\ZZ$ with complex conjugation
  generating the $\ZZ/2\ZZ$ factor, the quotient $\Gal(K'/\QQ) \cong \ZZ/2^m\ZZ$ is cyclic, and
  $K'$ is unramified outside $2$. If $K'K/\QQ$ were not cyclic, then $\Gal(K'K/\QQ)$ — a non-cyclic
  finite abelian $2$-group — would have two distinct subgroups of index $2$, hence two distinct
  quadratic subfields of $K'K$; both are unramified outside $2$ and real (as $K'K$ is totally
  real), contradicting \ref{prop:kw_2_quadratic_real_unique}. Hence $K'K/\QQ$ is cyclic, and
  \ref{lem:kw_cyclic_compositum} gives $K \subseteq K' \subseteq \QQ(\zeta_{2^{m+2}})$.
\end{proof}

\begin{proposition}
  \label{prop:kw_2_power}
  \lean{prop_kw_2_power}
  \leanok
  \uses{prop:kw_2_power_real}
  Let $K/\QQ$ be a cyclic extension of degree $2^m$ unramified outside $2$.
  Then $K$ is cyclotomic.
\end{proposition}

\begin{proof}
  \leanok
  If $K$ is totally real this is \ref{prop:kw_2_power_real}. Otherwise let $M$ be the maximal real
  subfield of $K(i)$. Then $K(i) = K \cdot \QQ(i) = M(i)$, so $K$ is cyclotomic $\iff K(i)$ is
  cyclotomic $\iff M$ is cyclotomic. Now $K(i)/\QQ$ is abelian (a compositum of the abelian
  extensions $K$ and $\QQ(i)$) and unramified outside $2$, hence so is $M$; moreover $M$ is cyclic:
  if $i \notin K$ then $\Gal(K(i)/\QQ) \cong \ZZ/2^m\ZZ \times \ZZ/2\ZZ$ with complex conjugation
  equal to $(2^{m-1}, 1)$, and the quotient by $\langle(2^{m-1},1)\rangle$ is cyclic of order $2^m$;
  if $i \in K$ then $M = K^+$ is a subfield of the cyclic field $K$. Thus $M$ is real, cyclic of
  $2$-power degree, and unramified outside $2$, so \ref{prop:kw_2_power_real} makes $M$ — and hence
  $K$ — cyclotomic. (This is Lemmermeyer's argument, Prop.~1.5.)
\end{proof}

\begin{lemma}
  \label{lem:kw_fixedField_index}
  \lean{fixedField_spec_of_index_eq_prime_pow}
  \leanok
  Let $C/\QQ$ be abelian and unramified outside $p$, and let $H \leq \Gal(C/\QQ)$ be of index
  $p^m$ with cyclic quotient. Then $C^{H}$ is a cyclic extension of $\QQ$ of degree $p^m$,
  unramified outside $p$.
\end{lemma}

\begin{proof}
  \leanok
  By the Galois correspondence $[C^{H} : \QQ] = [\Gal(C/\QQ) : H] = p^m$ and
  $\Gal(C^{H}/\QQ) \cong \Gal(C/\QQ)/H$, which is cyclic by hypothesis. A subfield of $C$ is
  unramified outside $p$ because $C$ is.
\end{proof}

\begin{lemma}
  \label{lem:kw_eq_of_unique_prime_subfield}
  \lean{kw_eq_of_unique_prime_subfield}
  \leanok
  \uses{lem:kw_cyclic_compositum}
  Let $K, K'$ be cyclic extensions of $\QQ$ of the same degree $p^m$, both unramified outside
  $p$, inside a common ambient field. If any two subfields of $KK'$ of degree $p$ over $\QQ$ and
  unramified outside $p$ coincide, then $K = K'$.
\end{lemma}

\begin{proof}
  \leanok
  $\Gal(KK'/\QQ)$ is an abelian $p$-group. Were it not cyclic it would have two distinct
  subgroups of index $p$, hence two distinct degree-$p$ subfields of $KK'$, both unramified
  outside $p$, contrary to the hypothesis. So $KK'/\QQ$ is cyclic and
  \ref{lem:kw_cyclic_compositum} gives $K \subseteq K'$; the degrees being equal and finite,
  $K = K'$.
\end{proof}

\begin{proposition}
  \label{prop:kw_odd_prime_power}
  \lean{prop_kw_odd_prime_power}
  \leanok
  \uses{prop:kw_exponent_p, lem:kw_eq_of_unique_prime_subfield, lem:kw_fixedField_index,
    lem:kw_minkowski}
  Let $K/\QQ$ be a cyclic extension of odd prime power degree $p^m$
  unramified outside $p$. Then $K$ is cyclotomic.
\end{proposition}

\begin{proof}
  \leanok
  Put $C = \QQ(\zeta_{p^{m+1}})$. As $p$ is odd, $\Gal(C/\QQ) \cong (\ZZ/p^{m+1}\ZZ)^\times$ is
  cyclic of order $p^{m}(p-1)$; let $\sigma$ be a generator and $H = \langle \sigma^{p^m}\rangle$,
  of index $p^m$ and with cyclic quotient. By \ref{lem:kw_fixedField_index}, $K' = C^{H}$ is
  cyclic of degree $p^m$ over $\QQ$ and unramified outside $p$. Two degree-$p$ subfields of
  $KK'$ unramified outside $p$ coincide by \ref{prop:kw_exponent_p}, so
  \ref{lem:kw_eq_of_unique_prime_subfield} gives $K = K' \subseteq \QQ(\zeta_{p^{m+1}})$.
\end{proof}

\section{The prime degree case}

\begin{proposition}
  \label{prop:kw_exponent_p}
  \lean{prop_kw_exponent_p}
  \leanok
  \uses{thm:stickelberger, lem:kw_kummer, lem:kw_abelian_kummer,
        lem:kw_split_prime, lem:kw_mu_val_p, lem:kw_mu_pth_power_ideal, lem:kw_class_trivial,
        lem:kw_mu_unit, lem:kw_unit_root_of_unity, lem:kw_ramification_reduction}
  Let $p$ be an odd prime, $A$ a field of characteristic $0$, and $\xi \in A$ a primitive
  $p^2$-th root of unity. Let $K \subseteq A$ be a subextension of $A/\QQ$ that is Galois and
  cyclic of degree $p$ over $\QQ$ and unramified outside $p$. Then $K \subseteq \QQ(\xi)$
  (the cyclotomic subfield $\QQ(\zeta_{p^2})$ of $A$). In particular $K$ is the unique subfield
  of degree $p$ of $\QQ(\xi)$, and the maximal abelian extension of $\QQ$ of exponent $p$
  unramified outside $p$ is cyclic of degree $p$.
\end{proposition}

\begin{proof}
  \leanok
  Set $\zeta = \xi^p$, a primitive $p$-th root of unity, $F = \QQ(\zeta) \subseteq A$, and
  $L = K \cdot F \subseteq A$ the compositum.

  \begin{enumerate}
    \item By \ref{lem:kw_kummer}, $L = F(\sqrt[p]{\mu})$ for some $\mu \in
      \cO_F$, $\mu \neq 0$, with $X^p - \mu$ irreducible over $F$.
    \item By \ref{lem:kw_mu_val_p}, $p \mid v_{\mathfrak{q}}(\mu)$ for every prime
      $\mathfrak{q}$ of $\cO_F$.
    \item By \ref{lem:kw_mu_pth_power_ideal}, $(\mu) = \mathfrak{a}^p$ for
      some ideal $\mathfrak{a}$ of $\cO_F$.
    \item By \ref{lem:kw_class_trivial}, $\mathfrak{a}$ is principal.
    \item By \ref{lem:kw_mu_unit}, $\mu = \alpha^p \eta$ for some $\alpha \in
      \cO_F$ and unit $\eta \in \cO_F^\times$.
    \item By \ref{lem:kw_unit_root_of_unity}, $L = \QQ(\xi)$ (the $p^2$-cyclotomic subfield).
  \end{enumerate}

  Therefore $K \subseteq L = \QQ(\xi)$. The Galois group
  $\Gal(\QQ(\xi)/\QQ) \cong (\ZZ/p^2\ZZ)^\times \cong \ZZ/p\ZZ \times
  \ZZ/(p-1)\ZZ$ has a unique subgroup of index $p$, giving a unique subfield
  of degree $p$ over $\QQ$ in $\QQ(\xi)$. Since $K \cap F = \QQ$
  (as $\gcd([K:\QQ], [F:\QQ]) = \gcd(p, p-1) = 1$) and $[\QQ(\xi):F] = p$,
  $K$ is exactly this unique subfield. The same applies to any other cyclic
  extension of degree $p$ unramified outside $p$, so the maximal such abelian
  extension of exponent $p$ is cyclic of degree $p$.
\end{proof}

The proof of \ref{prop:kw_exponent_p} proceeds via the following lemmas.
Let $K/\QQ$ be a cyclic extension of prime degree $p$ unramified outside $p$.
Set $F = \QQ(\zeta_p)$ and $L = KF$.

\begin{lemma}
  \label{lem:kw_kummer}
  \uses{lem:kw_abelian_kummer}
  \lean{kw_kummer, unramifiedOutside_sup}
  \leanok
  With the above notation, $L = F(\sqrt[p]{\mu})$ for some nonzero
  $\mu \in \cO_F$, and $L/F$ is unramified outside $p$.
\end{lemma}

\begin{proof}
  \leanok
  Since $[K:\QQ] = p$ and $[F:\QQ] = p-1$ are coprime, $K \cap F = \QQ$,
  so $[L:F] = [KF:F] = [K:\QQ] = p$. The extension $L/F$ is Galois with
  cyclic Galois group of order $p$, and $F$ contains the $p$-th roots of
  unity $\mu_p$. By Kummer theory (\texttt{IsCyclic} + \texttt{Kummer}),
  $L = F(\sqrt[p]{\mu})$ for some $\mu \in F^\times$. Multiplying by a
  $p$-th power if necessary, we may assume $\mu \in \cO_F$.

  For the ramification: let $q \neq p$ be a prime. Since $K/\QQ$ is
  unramified at $q$ and $F/\QQ$ is unramified at $q$ (as $F = \QQ(\zeta_p)$
  is only ramified at $p$), the compositum $L = KF$ is also unramified
  at $q$.
\end{proof}

\begin{lemma}
  \label{lem:kw_abelian_kummer}
  \lean{kw_abelian_kummer}
  \leanok
  Since $L/\QQ$ is Galois, for every $\sigma_a \in \Gal(F/\QQ)$ there exists
  $\xi \in F^\times$ and $a \in \ZZ$ such that $\sigma_a(\mu) = \xi^p \mu^a$.
\end{lemma}

\begin{proof}
  \leanok
  This is the backward direction of \texttt{isGalois\_iff\_forall\_apply\_eq\_pow\_mul\_zpow}:
  since $L/\QQ$ is Galois (which holds because $L = KF$ is the compositum of
  the abelian extensions $K/\QQ$ and $F/\QQ$), the criterion gives the
  required $\xi$ and $a$ for every $\sigma_a \in \Gal(F/\QQ)$.
\end{proof}


\begin{lemma}
  \label{lem:kw_split_prime}
  \lean{kw_split_prime}
  \leanok
  \uses{lem:kw_abelian_kummer}
  Let $\mathfrak{q}$ be a prime ideal of $F$ with $(\mu) = \mathfrak{q}^r \mathfrak{a}$
  with $\mathfrak{q} \nmid \mathfrak{a}$. If $p \nmid r$ and $L/\QQ$ is
  abelian, then $\mathfrak{q}$ splits completely in $F/\QQ$.
\end{lemma}

\begin{proof}
  \leanok
  Let $\sigma_a$ be an element of the decomposition group of $\mathfrak{q}$
  in $\Gal(F/\QQ)$, so $\sigma_a(\mathfrak{q}) = \mathfrak{q}$.
  Since $L/\QQ$ is abelian, by \ref{lem:kw_abelian_kummer} there exists
  $\xi \in F^\times$ such that $\sigma_a(\mu) = \xi^p \mu^a$. Since
  $\sigma_a(\mathfrak{q}) = \mathfrak{q}$, applying $\sigma_a$ to the
  factorization $(\mu) = \mathfrak{q}^r \mathfrak{a}$ gives
  $(\xi^p \mu^a) = \mathfrak{q}^r \sigma_a(\mathfrak{a})$, hence
  $\mathfrak{q}^r \| \xi^p \mu^a$. But also $(\mu^a) = \mathfrak{q}^{ar}\mathfrak{a}^a$,
  so $\mathfrak{q}^r \| \xi^p \mu^a$ implies $r \equiv ar \pmod{p}$.
  Since $p \nmid r$, this forces $a \equiv 1 \pmod{p}$,
  so $\sigma_a = 1$ and $\mathfrak{q}$ splits completely in $F/\QQ$.
\end{proof}

\begin{lemma}
  \label{lem:kw_mu_val_p}
  \uses{lem:kw_kummer, lem:kw_split_prime}
  \lean{kw_mu_val_p}
  \leanok
  For any prime $\mathfrak{q}$ of $\cO_F$, $p \mid v_{\mathfrak{q}}(\mu)$.
\end{lemma}

\begin{proof}
  \leanok
  Write $r = v_{\mathfrak{q}}(\mu)$.

  \textbf{Case 1: $\mathfrak{q} \nmid p$.}
  By \ref{lem:kw_kummer}, $L/F$ is unramified at $\mathfrak{q}$, so
  $e(\mathfrak{Q}|\mathfrak{q}) = 1$ for any prime $\mathfrak{Q}$ of $L$
  above $\mathfrak{q}$. Let $\alpha \in L$ satisfy $\alpha^p = \mu$. Then
  \[
    p \cdot v_{\mathfrak{Q}}(\alpha) = v_{\mathfrak{Q}}(\alpha^p)
    = v_{\mathfrak{Q}}(\mu) = e(\mathfrak{Q}|\mathfrak{q})\cdot v_{\mathfrak{q}}(\mu)
    = v_{\mathfrak{q}}(\mu) = r,
  \]
  so $p \mid r$.

  \textbf{Case 2: $\mathfrak{q} = \mathfrak{p} = (1-\zeta_p)$.}
  Since $F/\QQ$ is totally ramified at $p$, $\mathfrak{p}$ is the unique prime
  of $F$ above $p$, so its decomposition group in $\Gal(F/\QQ)$ is all of
  $\Gal(F/\QQ)$. Since $|\Gal(F/\QQ)| = p-1 > 1$, $\mathfrak{p}$ does not
  split completely in $F/\QQ$. By the contrapositive of
  \ref{lem:kw_split_prime}, $p \mid r$.
\end{proof}

\begin{lemma}
  \label{lem:kw_not_dvd_nu}
  \uses{lem:kw_kummer, lem:kw_mu_val_p}
  \lean{kw_not_dvd_nu}
  \leanok
  There exists $\nu \in \cO_F$, $\nu \neq 0$, such that $L = F(\sqrt[p]{\nu})$,
  $X^p - \nu$ is irreducible over $F$, and $(1-\zeta_p) \nmid \nu$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:kw_mu_val_p}, $p \mid v_{\mathfrak{p}}(\mu)$; write
  $v_{\mathfrak{p}}(\mu) = pk$. Set $\nu = \mu / (1-\zeta_p)^{pk} \in \cO_F$.
  Since $(1-\zeta_p)^k \in F^\times$, we have $F(\sqrt[p]{\nu}) =
  F(\sqrt[p]{\mu}/(1-\zeta_p)^k) = F(\sqrt[p]{\mu}) = L$
  (by \ref{lem:kw_kummer}). By construction $v_{\mathfrak{p}}(\nu) = 0$,
  i.e. $(1-\zeta_p) \nmid \nu$.
\end{proof}

\begin{lemma}
  \label{lem:kw_mu_pth_power_ideal}
  \uses{lem:kw_mu_val_p}
  \lean{kw_mu_pth_power_ideal}
  \leanok
  $(\mu) = \mathfrak{a}^p$ for some ideal $\mathfrak{a}$ of $\cO_F$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:kw_mu_val_p}, $p \mid v_{\mathfrak{q}}(\mu)$ for every prime
  $\mathfrak{q}$. Therefore $(\mu) = \mathfrak{a}^p$ where
  $\mathfrak{a} = \prod_{\mathfrak{q}} \mathfrak{q}^{v_{\mathfrak{q}}(\mu)/p}$.
\end{proof}

\begin{lemma}
  \label{lem:kw_class_trivial}
  \lean{kw_class_trivial}
  \leanok
  \uses{lem:kw_mu_pth_power_ideal, cor:stickelberger_annihilates, lem:kw_abelian_kummer}
  The ideal class $[\mathfrak{a}]$ is trivial, i.e., $\mathfrak{a}$ is principal.
\end{lemma}

\begin{proof}
  \leanok
  From $(\mu) = \mathfrak{a}^p$ and $\sigma_a(\mu) = \xi^p \mu^a$
  (by \ref{lem:kw_abelian_kummer}), we get
  $\sigma_a(\mathfrak{a})^p = \sigma_a(\mathfrak{a}^p) = \sigma_a((\mu)) =
  (\xi^p \mu^a) = (\xi)^p \cdot \mathfrak{a}^{pa}$. Since the ideal monoid is
  torsion-free, $\sigma_a(\mathfrak{a}) = (\xi) \cdot \mathfrak{a}^a$, hence
  $\sigma_a(c) = c^a$ for the ideal class $c = [\mathfrak{a}]$ and every
  $1 \leq a < p$.

  Note that we work throughout with the \emph{class} $c$: unlike in the per-ideal
  form of Stickelberger, we never need $\mathfrak{a}$ itself to be coprime to $p$
  or to have odd norm. By \ref{cor:stickelberger_annihilates} applied with
  $m = p$, the class $c$ is annihilated by the Stickelberger element $\Theta =
  \sum_{a=1}^{p-1} a\sigma_a^{-1} \in \ZZ[G]$. We compute $c^{\Theta}$:
  \[
    1 = c^{\Theta} = \prod_{a=1}^{p-1} \sigma_a^{-1}(c)^a
    = \prod_{a=1}^{p-1} c^{a^{-1} \cdot a} = c^{p-1},
  \]
  where we used $\sigma_a^{-1}(c) = \sigma_{a^{-1}}(c) = c^{a^{-1}}$, with
  $a^{-1}$ the inverse of $a$ modulo $p$. Since $(\mu) = \mathfrak{a}^p$, the
  ideal $\mathfrak{a}^p$ is principal, so $c^p = 1$ and hence
  $c^{p-1} = c^{-1}$. Combining with $c^{\Theta} = 1$ gives $c^{-1} = 1$, i.e.\
  $c = [\mathfrak{a}] = 1$ and $\mathfrak{a}$ is principal.
\end{proof}

\begin{lemma}
  \label{lem:kw_mu_unit}
  \lean{kw_mu_unit}
  \leanok
  \uses{lem:kw_class_trivial}
  There exists a unit $\eta \in \cO_F^\times$ such that $\mu = \alpha^p \eta$
  for some $\alpha \in \cO_F$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:kw_class_trivial}, $\mathfrak{a} = (\alpha)$ for some
  $\alpha \in \cO_F$. Then $(\mu) = \mathfrak{a}^p = (\alpha^p)$, so
  $\mu/\alpha^p \in \cO_F^\times$. Setting $\eta = \mu/\alpha^p$ gives
  $\mu = \alpha^p \eta$.
\end{proof}

\begin{lemma}
  \label{lem:kw_conj_mul_eq_pow}
  \lean{kw_conj_mul_eq_pow}
  \leanok
  \uses{lem:kw_abelian_kummer}
  Let $\sigma_{-1}$ denote complex conjugation on $F$. With $\mu = \alpha^p \eta$, the product
  $\eta\,\sigma_{-1}(\eta)$ is a $p$-th power in $F^\times$.
\end{lemma}

\begin{proof}
  \leanok
  Since $L/\QQ$ is abelian, \ref{lem:kw_abelian_kummer} applied to $\sigma_{-1}$ gives
  $\xi \in F^\times$ with $\sigma_{-1}(\mu) = \xi^p\, \mu^{-1}$. Substituting $\mu = \alpha^p \eta$
  and using that $\sigma_{-1}$ is a ring homomorphism,
  \[
    \sigma_{-1}(\alpha)^p\, \sigma_{-1}(\eta) = \xi^p\, \alpha^{-p}\, \eta^{-1},
  \]
  whence
  \[
    \eta\, \sigma_{-1}(\eta)
      = \xi^p\, \alpha^{-p}\, \sigma_{-1}(\alpha)^{-p}
      = \bigl(\xi/(\alpha\, \sigma_{-1}(\alpha))\bigr)^p .
  \]
\end{proof}

\begin{lemma}
  \label{lem:kw_exists_realUnit_torsion}
  \lean{kw_exists_realUnit_torsion}
  \leanok
  \uses{lem:kw_conj_mul_eq_pow}
  The fourth power $\eta^4$ factors as $\eta^4 = \zeta\, \delta^p$ with $\zeta$ a $p$-th root of
  unity and $\delta \in F$.
\end{lemma}

\begin{proof}
  \leanok
  Let $\sigma_{-1}$ be complex conjugation on $F = \QQ(\zeta_p)$. For any unit
  $\eta \in \cO_F^\times$, the ratio $\eta/\sigma_{-1}(\eta)$ has absolute value $1$ at every
  archimedean place, hence is a root of unity, while $\eta\,\sigma_{-1}(\eta)$ is a real unit;
  their product being $\eta^2$, we get $\eta^2 = \zeta_0\,\varepsilon_0$ with $\zeta_0$ a root of
  unity and $\sigma_{-1}(\varepsilon_0) = \varepsilon_0$. Squaring,
  \[
    \eta^4 = \zeta\, \varepsilon, \qquad \zeta = \zeta_0^2,\ \varepsilon = \varepsilon_0^2 .
  \]
  The roots of unity in $F$ form $\mu_{2p}$, so $\zeta_0^{2p} = 1$ and hence $\zeta^p =
  \zeta_0^{2p} = 1$: $\zeta$ is a $p$-th root of unity. (Squaring twice lands the torsion factor
  in $\mu_p$, sidestepping any sign normalization.)

  Now $\sigma_{-1}$ inverts the root of unity $\zeta$ and fixes the real unit $\varepsilon$, so
  applying it to $\eta^4 = \zeta\,\varepsilon$ gives $\sigma_{-1}(\eta)^4 = \zeta^{-1}\varepsilon$;
  multiplying the two relations the $\zeta$ cancels:
  \[
    \varepsilon^2 = \bigl(\eta\,\sigma_{-1}(\eta)\bigr)^4 .
  \]
  By \ref{lem:kw_conj_mul_eq_pow}, $\eta\,\sigma_{-1}(\eta) = w^p$, so $\varepsilon^2 = (w^4)^p$.
  As $p$ is odd, $\gcd(2,p) = 1$, hence $\varepsilon = \delta^p$ for some $\delta \in F$.
  Therefore $\eta^4 = \zeta\, \delta^p$.
\end{proof}

\begin{lemma}
  \label{lem:kw_unit_root_of_unity}
  \lean{kw_unit_root_of_unity}
  \leanok
  \uses{lem:kw_mu_unit, lem:kw_exists_realUnit_torsion}
  The unit $\eta$ is a $p$-th power times a root of unity. Therefore $L = \QQ(\zeta_{p^2})$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:kw_exists_realUnit_torsion}, $\eta^4 = \zeta\, \delta^p$ with $\zeta$ a $p$-th root
  of unity and $\delta \in F$. With $\mu = \alpha^p \eta$ (\ref{lem:kw_mu_unit}),
  \[
    \mu^4 = \alpha^{4p}\, \zeta\, \delta^p = \zeta\,(\alpha^4 \delta)^p \in
    \zeta_p^{\ZZ} (F^\times)^p .
  \]
  As $\gcd(4,p) = 1$ we have $F(\sqrt[p]{\mu}) = F(\sqrt[p]{\mu^4})$, and $\mu^4 \in
  \zeta_p^{\ZZ}(F^\times)^p$ gives $F(\sqrt[p]{\mu^4}) = F(\sqrt[p]{\zeta}) \subseteq
  \QQ(\zeta_{p^2})$. Since $[L:F] = p = [\QQ(\zeta_{p^2}):F]$, we conclude $L = \QQ(\zeta_{p^2})$.
\end{proof}

\begin{lemma}

  where $\psi = 1$ denotes the trivial character (identically equal to $1$).

\begin{proof}
\leanok
\end{proof}
