\chapter{Setup}

\section{The cyclotomic field $K$}

Let $p$ be a prime number and $f$ a positive integer. Set $q = p^f$.
Let $K = \QQ(\zeta_{q-1})$ be the $(q-1)$-th cyclotomic field.
We fix a maximal ideal $P$ of $\cO_K$ lying above $p$.

\section{The Teichmüller character}

\begin{definition}
  \label{def:teichmuller}
  \lean{teichmuller}
  \leanok
  The \emph{Teichmüller character} is the unique group isomorphism
  \[
    \om : (\cO_K/P)^\times \xrightarrow{\sim} \mu_{q-1}(\cO_K)
  \]
  satisfying $\om(x) \equiv x \pmod{P}$ for all $x \in (\cO_K/P)^\times$,
  extended to all of $\cO_K/P$ by $\om(0) = 0$.
\end{definition}

\begin{lemma}
  \label{lem:orderOf_teichmuller}
  \uses{def:teichmuller}
  \lean{orderOf_teichmuller}
  \leanok
  The order of $\om$ as a multiplicative character is $q - 1$.
\end{lemma}

\begin{proof}
  \leanok
  Since $\om$ is an isomorphism between groups of order $q - 1$, it has
  order $q - 1$ as a group homomorphism.
\end{proof}

\begin{lemma}
  \label{lem:map_teichmuller_zpow_eq}
  \lean{map_teichmuller_zpow_eq}
  \leanok
  \uses{def:teichmuller}
  Let $R$ be a ring containing $\cO_K$ and let $\sigma$ be a ring endomorphism of $R$ such that
  $\sigma(\zeta_{q-1}) = \zeta_{q-1}^m$ for some $m \in \ZZ$. Then
  $\sigma(\om(x)) = \om(x)^m$ for all $x \in \cO_K/P$.
\end{lemma}

\begin{proof}
  \leanok
  Since $\om(x) \in \mu_{q-1}(\cO_K)$, we have $\om(x) = \zeta_{q-1}^k$
  for some $k$. Then $\sigma(\om(x)) = \sigma(\zeta_{q-1})^k = \zeta_{q-1}^{mk}
  = \om(x)^m$.
\end{proof}

\begin{lemma}
  \label{lem:teichmuller_sum}
  \uses{def:teichmuller, lem:orderOf_teichmuller}
  \leanok
  For all $a \in \ZZ$ with $(q-1) \nmid a$,
  \[
    \sum_{x \in \cO_K/P} \om(x)^{-a} = 0.
  \]
\end{lemma}

\begin{proof}
  \leanok
  Since $(q-1) \nmid a$, the character $\om^{-a}$ is nontrivial by
  \ref{lem:orderOf_teichmuller}. By \texttt{MulChar.sum\_eq\_zero\_of\_ne\_one},
  $\sum_{x \in \cO_K/P} \om(x)^{-a} = 0$.
\end{proof}

\section{The additive character}

Let $L$ be an extension of $K$ containing a primitive $p$-th root of unity
$\zeta_p$. We fix such a $\zeta_p \in \cO_L$.

\begin{definition}
  \label{def:addchar}
  \lean{addCharTrace}
  \leanok
  The \emph{additive character} $\psi : \cO_K/P \to \mu_p(\cO_L)$ is defined by
  \[
    \psi(x) = \zeta_p^{\tr_{(\cO_K/P)/(\ZZ/p\ZZ)}(x)}
  \]
  where the exponent is lifted canonically to $\ZZ$.
\end{definition}

\begin{lemma}
  \label{lem:addCharTrace_ne_one}
  \uses{def:addchar}
  \lean{addCharTrace_ne_one}
  \leanok
  The character $\psi$ is nontrivial, i.e., $\psi \neq 1$.
\end{lemma}

\begin{proof}
  \leanok
  Since the extension $\cO_K/P$ over $\ZZ/p\ZZ$ is separable, the trace
  $\tr : \cO_K/P \to \ZZ/p\ZZ$ is nonzero, so there exists $x$ with
  $\tr(x) \neq 0$, hence $\psi(x) = \zeta_p^{\tr(x)} \neq 1$.
\end{proof}

\begin{lemma}
  \label{lem:addCharTrace_isPrimitive}
  \uses{lem:addCharTrace_ne_one}
  \lean{addCharTrace_isPrimitive}
  \leanok
  The character $\psi$ is primitive.
\end{lemma}

\begin{proof}
  \leanok
  Since $\cO_K/P$ is a field, any nontrivial additive character is primitive.
  The character $\psi$ is nontrivial by \ref{lem:addCharTrace_ne_one}.
\end{proof}

\begin{lemma}
  \label{lem:addCharTrace_frob_apply}
  \uses{def:addchar}
  \lean{addCharTrace_frob_apply}
  \leanok
  For all $x \in \cO_K/P$, $\psi(x^p) = \psi(x)$.
\end{lemma}

\begin{proof}
  \leanok
  The Frobenius map $\varphi : x \mapsto x^p$ is a $(\ZZ/p\ZZ)$-algebra
  automorphism of $\cO_K/P$. By \texttt{Algebra.trace\_eq\_of\_algEquiv},
  $\tr(\varphi(x)) = \tr(x)$, i.e., $\tr(x^p) = \tr(x)$. Therefore
  $\psi(x^p) = \zeta_p^{\tr(x^p)} = \zeta_p^{\tr(x)} = \psi(x)$.
\end{proof}

\begin{lemma}
  \label{lem:addCharTrace_mk_sq_eq}
  \uses{def:addchar}
  \lean{addCharTrace_mk_sq_eq}
  \leanok
  Let $\fP$ be a prime ideal of $\cO_L$ with $\zeta_p - 1 \in \fP$.
  Then
  \[
    \psi(x) \equiv 1 + \tr(x) \cdot (\zeta_p - 1) \pmod{\fP^2}
  \]
  for all $x \in \cO_K/P$.
\end{lemma}

\begin{proof}
  \leanok
  There exists $a \in \mathbb{N}$ such that $\tr(x) = a$ and $\psi(x) = \zeta_p^a$.
  By the binomial theorem,
  \[
    \psi(x) = \zeta_p^a = (1 + (\zeta_p - 1))^a = \sum_{n=0}^{a} \binom{a}{n} (\zeta_p - 1)^n.
  \]
  Since $\zeta_p - 1 \in \fP$, we have $(\zeta_p - 1)^n \in \fP^n \subseteq \fP^2$ for all
  $n \geq 2$. Therefore
  \[
    \psi(x) \equiv 1 + a(\zeta_p - 1) = 1 + \tr(x) \cdot (\zeta_p - 1) \pmod{\fP^2}.
  \]
\end{proof}

\begin{lemma}
  \label{lem:addCharTrace_mk_eq_one}
  \uses{lem:addCharTrace_mk_sq_eq}
  \lean{addCharTrace_mk_eq_one}
  \leanok
  Let $\fP$ be a prime ideal of $\cO_L$ with $\zeta_p - 1 \in \fP$.
  Then $\psi(x) \equiv 1 \pmod{\fP}$ for all $x \in \cO_K/P$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:addCharTrace_mk_sq_eq}, $\psi(x) \equiv 1 + \tr(x) \cdot (\zeta_p - 1) \pmod{\fP^2}$.
  Since $\zeta_p - 1 \in \fP$, the term $\tr(x) \cdot (\zeta_p - 1) \in \fP$,
  hence $\psi(x) \equiv 1 \pmod{\fP}$.
\end{proof}

\chapter{The Gauss sum}
\label{chap:gauss_sum}

Throughout this chapter and the next two, we assume that $p$ is an odd prime.

Let $\fP$ be a prime ideal of $\cO_L$ with $\zeta_p - 1 \in \fP$. Let $F$ be
the subfield of $L$ with $F = \QQ(\zeta_p)$. The Galois group $\Gal(L/F)$
is isomorphic to $(\ZZ/(q-1)\ZZ)^\times$ via the map $\sigma \mapsto a$
where $\sigma(\zeta_{q-1}) = \zeta_{q-1}^a$. We denote by $\fn(\sigma) \in
\{1, \ldots, q-2\}$ the natural number corresponding to $\sigma$ under this
isomorphism.

\section{Gauss sums}

\begin{definition}
  \label{def:gauss_sum_s}
  \uses{def:teichmuller, def:addchar}
  \lean{GaussSum}
  \leanok
  For $a \in \ZZ$, the \emph{Gauss sum} is
  \[
    g_a = g(\om^{-a}, \psi) = \sum_{x \in \cO_K/P} \om(x)^{-a} \cdot \psi(x).
  \]
\end{definition}

\begin{lemma}
  \label{lem:GaussSum_periodic}
  \uses{def:gauss_sum_s, lem:orderOf_teichmuller}
  \lean{GaussSum_periodic}
  \leanok
  If $(q-1) \mid k$, then for all $a \in \ZZ$, $g_{a+k} = g_a$.
\end{lemma}

\begin{proof}
  \leanok
  Since $\omega^{q-1} = 1$ by \ref{lem:orderOf_teichmuller}, $(q-1) \mid k$ implies
  $\omega^{-k} = 1$, so $\omega^{-(a+k)} = \omega^{-a}$, hence $g_{a+k} = g_a$.
\end{proof}

\begin{lemma}
  \label{lem:GaussSum_mem}
  \uses{def:gauss_sum_s, lem:addCharTrace_mk_eq_one, lem:orderOf_teichmuller, lem:teichmuller_sum}
  \lean{GaussSum_mem}
  \leanok
  If $(q-1) \nmid a$, then $g_a \in \fP$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:addCharTrace_mk_eq_one}, $\psi(x) \equiv 1 \pmod{\fP}$ for all $x \in \cO_K/P$.
  Therefore
  \[
    g_a = \sum_{x \in \cO_K/P} \omega(x)^{-a} \psi(x)
    \equiv \sum_{x \in \cO_K/P} \omega(x)^{-a} \pmod{\fP}.
  \]
  Since $(q-1) \nmid a$, we have $\omega^{-a} \neq 1$ by \ref{lem:orderOf_teichmuller}.
  Hence $\sum_{x \in \cO_K/P} \omega(x)^{-a} = 0$ by \ref{lem:teichmuller_sum}.
  Therefore $g_a \equiv 0 \pmod{\fP}$, i.e.\ $g_a \in \fP$.
\end{proof}

\begin{lemma}
  \label{lem:GaussSum_mul_GaussSum_neg}
  \uses{def:gauss_sum_s, lem:addCharTrace_isPrimitive, def:teichmuller}
  \lean{GaussSum_mul_GaussSum_neg}
  \leanok
  If $(q-1) \nmid a$, then
  \[
    g_a \cdot g_{-a} = \omega^a(-1) \cdot p^f.
  \]
\end{lemma}

\begin{proof}
  \leanok
  Since $(q-1) \nmid a$, $\omega^{-a}$ is nontrivial and $\psi$ is primitive
  by \ref{lem:addCharTrace_isPrimitive}. By \texttt{gaussSum\_mul\_gaussSum\_eq\_card},
  $g(\omega^{-a}, \psi) \cdot g(\omega^{a}, \psi^{-1}) = p^f$.
  By the change of variable $x \mapsto -x$,
  $g(\omega^{a}, \psi^{-1}) = \omega^{a}(-1) \cdot g_{-a}$.
  Hence $g_a \cdot g_{-a} = \omega^a(-1) \cdot p^f$.
\end{proof}

\begin{lemma}
  \label{lem:norm_GaussSum}
  \uses{def:gauss_sum_s, lem:GaussSum_mul_GaussSum_neg, lem:GaussSum_mem}
  \lean{norm_GaussSum}
  \leanok
  If $(q-1) \nmid a$, then there exists $k \geq 1$ such that
  $N_{L/\QQ}(g_a) = \pm p^k$.
  In particular, the only rational prime dividing $N_{L/\QQ}(g_a)$ is $p$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:GaussSum_mul_GaussSum_neg}, $g_a \cdot g_{-a} = \omega^a(-1) \cdot p^f = \pm p^f$.
  Taking norms, $N_{L/\QQ}(g_a) \cdot N_{L/\QQ}(g_{-a}) = \pm p^{f \cdot [L:\QQ]}$.
  Since both norms are integers and their product is $\pm p^{f \cdot [L:\QQ]}$,
  each is of the form $\pm p^k$ for some $k \geq 0$. By \ref{lem:GaussSum_mem},
  $g_a \in \fP$, so $g_a$ is not a unit in $\cO_L$, hence $N_{L/\QQ}(g_a) \neq \pm 1$,
  thus $k \geq 1$.
\end{proof}

\begin{lemma}
  \label{lem:mk_sq_gausssum_eq}
  \uses{lem:addCharTrace_mk_sq_eq, lem:orderOf_teichmuller}
  \lean{mk_sq_gausssum_eq}
  \leanok
  Assume $\zeta_p - 1 \in \fP$. Then
  \[
    g_1 \equiv -(\zeta_p - 1) \pmod{\fP^2}.
  \]
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:addCharTrace_mk_sq_eq}, $\psi(x) \equiv 1 + \tr(x) \cdot (\zeta_p - 1) \pmod{\fP^2}$
  for all $x \in \cO_K/P$. Therefore
  \[
    g_1 \equiv \sum_{x} \omega(x)^{-1} + (\zeta_p-1) \sum_{x} \omega(x)^{-1} \tr(x) \pmod{\fP^2}.
  \]
  Since $p$ is odd, $q - 1 \geq 2$, so $\omega^{-1} \neq 1$ by
  \ref{lem:orderOf_teichmuller}, and $\sum_x \omega(x)^{-1} = 0$ by orthogonality
  of characters. Moreover,
  Computing the trace via powers of Frobenius, $\sum_{x \in \cO_K/P} \omega(x)^{-1} \cdot \tr(x) = -1$.
  Hence $g_1 \equiv -(\zeta_p - 1) \pmod{\fP^2}$.
\end{proof}

\begin{lemma}
  \label{lem:galFEquiv}
  \uses{def:gauss_sum_s}
  \lean{galFEquiv}
  \leanok
  The extension $L/F$ is Galois with $\Gal(L/F) \cong (\ZZ/(q-1)\ZZ)^\times$,
  via the action on $\zeta_{q-1}$.
\end{lemma}

\begin{proof}
  \leanok
  Since $\gcd(q-1, p) = 1$, the fields $K = \QQ(\zeta_{q-1})$ and
  $F = \QQ(\zeta_p)$ are linearly disjoint over $\QQ$. Therefore
  $L = KF$ is Galois over $F$ with $\Gal(L/F) \cong \Gal(K/\QQ) \cong
  (\ZZ/(q-1)\ZZ)^\times$.
\end{proof}

\begin{lemma}
  \label{lem:gal_gaussSum_eq_gaussSum}
  \uses{def:gauss_sum_s, lem:map_teichmuller_zpow_eq}
  \lean{gal_gaussSum_eq_gaussSum}
  \leanok
  For all $\sigma \in \Gal(L/F)$ and $a \in \ZZ$,
  \[
    \sigma(g_a) = g_{a \cdot \fn(\sigma)}.
  \]
\end{lemma}

\begin{proof}
  \leanok
  Since $\sigma \in \Gal(L/F)$, it fixes $\zeta_p$, hence $\psi$.
  By \ref{lem:map_teichmuller_zpow_eq}, $\sigma(\omega(x)) = \omega(x)^{\fn(\sigma)}$
  for all $x$. Therefore
  \[
    \sigma(g_a) = \sum_{x \in \cO_K/P} \sigma(\omega(x)^{-a}) \cdot \psi(x)
    = \sum_{x \in \cO_K/P} \omega(x)^{-a \cdot \fn(\sigma)} \cdot \psi(x)
    = g_{a \cdot \fn(\sigma)}.
  \]
\end{proof}

\begin{lemma}
  \label{lem:GaussSum_frob}
  \uses{def:gauss_sum_s, lem:addCharTrace_frob_apply}
  \lean{GaussSum_frob}
  \leanok
  For all $a \in \ZZ$, $g_{pa} = g_a$.
\end{lemma}

\begin{proof}
  \leanok
  The Frobenius map $\varphi : x \mapsto x^p$ is a bijection on $\cO_K/P$.
  Using the substitution $y = \varphi(x)$,
  \[
    g_{pa} = \sum_{x \in \cO_K/P} \omega(x)^{-pa} \psi(x)
    = \sum_{y \in \cO_K/P} \omega(\varphi^{-1}(y))^{-pa} \psi(\varphi^{-1}(y)).
  \]
  By multiplicativity of $\omega$, $\omega(\varphi^{-1}(y))^p = \omega(\varphi^{-1}(y)^p)
  = \omega(y)$, so $\omega(\varphi^{-1}(y))^{-pa} = \omega(y)^{-a}$.
  By \ref{lem:addCharTrace_frob_apply}, $\psi(\varphi^{-1}(y)) = \psi(y)$.
  Therefore $g_{pa} = g_a$.
\end{proof}

\chapter{The valuation function $s$}

\begin{definition}
  \lean{valGauss}
  \leanok
  \label{def:s}
  For $a \in \ZZ$, we define
  \[
    s(a) = v_{\fP}(g_a)
  \]
\end{definition}

\begin{lemma}
  \label{lem:valGauss_eq_zero}
  \uses{def:s, lem:gaussSum_one_left, lem:addCharTrace_ne_one, lem:orderOf_teichmuller}
  \lean{valGauss_eq_zero}
  \leanok
  If $(q - 1) \mid a$, then $s(a) = 0$.
\end{lemma}

\begin{proof}
  \leanok
  Since $(q-1) \mid a$, $\omega^{-a} = 1$ by \ref{lem:orderOf_teichmuller},
  so $g_a = \sum_{x} \psi(x)$, which equals $0$ by \ref{lem:gaussSum_one_left}
  since $\psi$ is nontrivial by \ref{lem:addCharTrace_ne_one}.
  Hence $s(a) = v_{\fP}(0) = 0$.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_subadditive}
  \uses{def:s, lem:GaussSum_mul_GaussSum, lem:valGauss_eq_zero}
  \lean{valGauss_subadditive}
  \leanok
  For all $a, b \in \ZZ$, $s(a + b) \leq s(a) + s(b)$.
\end{lemma}

\begin{proof}
  \leanok
  If $(q-1) \mid (a+b)$ then $s(a+b) = 0$ by \ref{lem:valGauss_eq_zero} and the result is trivial.
  Otherwise, by \ref{lem:GaussSum_mul_GaussSum},
  \[
    g_a \cdot g_b = g_{a+b} \cdot J(\omega^{-a}, \omega^{-b}).
  \]
  Taking valuations and using the fact that $J(\omega^{-a}, \omega^{-b}) \in \cO_L$,
  hence $v_{\fP}(J) \geq 0$, we obtain $s(a) + s(b) \geq s(a+b)$.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_add_valGauss_sub_self}
  \uses{def:s, lem:ramificationIdx_eq_p_sub_one, lem:valGauss_eq_zero, lem:GaussSum_periodic, lem:GaussSum_mul_GaussSum_neg}
  \lean{valGauss_add_valGauss_sub_self}
  \leanok
  Assume $(q-1) \mid k$. Then
  \[
    s(a) + s(k - a) = \begin{cases} 0 & \text{if } (q-1) \mid a \\
    (p-1) \cdot f & \text{otherwise.} \end{cases}
  \]
\end{lemma}

\begin{proof}
  \leanok
  Since $(q-1) \mid k$, we have $s(k-a) = s(-a)$ by periodicity of $g$:
  by \ref{lem:GaussSum_periodic}, $g_{a+(q-1)} = g_a$, so $s$ has period $q-1$.

  If $(q-1) \mid a$, then $(q-1) \mid (-a)$, so $s(a) = s(-a) = 0$ by \ref{lem:valGauss_eq_zero} and the result is clear.

  If $(q-1) \nmid a$, by \ref{lem:GaussSum_mul_GaussSum_neg},
  $g_a \cdot g_{-a} = \omega^{a}(-1) \cdot p^f$.
  Since $\omega^{a}(-1)$ is a root of unity, it is a unit in $\cO_L$, hence
  $v_{\fP}(\omega^{a}(-1)) = 0$. By \ref{lem:ramificationIdx_eq_p_sub_one},
  $v_{\fP}(p^f) = f \cdot e(\fP \mid p) = f(p-1)$.
  Therefore
  \[
    s(a) + s(-a) = v_{\fP}(g_a) + v_{\fP}(g_{-a}) = v_{\fP}(g_a \cdot g_{-a})
    = v_{\fP}(p^f) = (p-1) \cdot f.
  \]
\end{proof}

\begin{lemma}
  \label{lem:valGauss_frob}
  \uses{def:s, lem:GaussSum_frob}
  \lean{valGauss_frob}
  \leanok
  For all $a \in \ZZ$, $s(pa) = s(a)$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:GaussSum_frob}, $g_{pa} = g_a$, hence
  $s(pa) = v_{\fP}(g_{pa}) = v_{\fP}(g_a) = s(a)$.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_one}
  \uses{lem:GaussSum_mem, lem:mk_sq_gausssum_eq, lem:zeta_sub_one_not_mem_sq}
  \lean{valGauss_one}
  \leanok
  $s(1) = 1$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:GaussSum_mem}, $g_1 \in \fP$, so $s(1) \geq 1$.
  By \ref{lem:mk_sq_gausssum_eq}, $g_1 \equiv -(\zeta_p - 1) \pmod{\fP^2}$.
  Since $\zeta_p - 1 \notin \fP^2$ by \ref{lem:zeta_sub_one_not_mem_sq},
  we have $g_1 \notin \fP^2$, so $s(1) \leq 1$.
\end{proof}

\begin{lemma}
  \label{lem:valGauss_le_self}
  \uses{def:s, lem:valGauss_subadditive, lem:valGauss_one, lem:valGauss_eq_zero}
  \lean{valGauss_le_self}
  \leanok
  For all $a \in \ZZ$ with $a \geq 0$, $s(a) \leq a$.
\end{lemma}

\begin{proof}
  \leanok
  By induction on $a$. The case $a = 0$ follows from \ref{lem:valGauss_eq_zero}
  (since $(q-1) \mid 0$). For $a \geq 1$, by \ref{lem:valGauss_subadditive} and
  the induction hypothesis,
  \[
    s(a) \leq s(1) + s(a-1) \leq 1 + (a-1) = a. \qedhere
  \]
\end{proof}

\begin{lemma}
  \label{lem:valGauss_toNat_le_sum_digits}
  \uses{def:s, lem:valGauss_subadditive, lem:valGauss_frob, lem:valGauss_le_self}
  \lean{valGauss_toNat_le_sum_digits}
  \leanok
  Let $0 \leq a$ and let
  $a = a_0 + a_1 p + \cdots + a_{t-1} p^{t-1}$ be the $p$-adic expansion of $a$.
  Then
  \[
    s(a) \leq a_0 + a_1 + \cdots + a_{t-1}.
  \]
\end{lemma}

\begin{proof}
  \leanok
  By induction on the list of digits $(a_0, \ldots, a_{t-1})$.
  The base case $a = 0$ is immediate. For the inductive step,
  write $a = a_0 + p \cdot a'$ where $a' = a_1 + a_2 p + \cdots + a_{t-1} p^{t-2}$.
  By \ref{lem:valGauss_subadditive},
  \[
    s(a_0 + p \cdot a') \leq s(a_0) + s(p \cdot a').
  \]
  By \ref{lem:valGauss_le_self}, $s(a_0) \leq a_0$.
  By \ref{lem:valGauss_frob}, $s(p \cdot a') = s(a')$.
  By the induction hypothesis, $s(a') \leq a_1 + \cdots + a_{t-1}$.
  Hence $s(a) \leq a_0 + a_1 + \cdots + a_{t-1}$.
\end{proof}

\begin{lemma}
  \label{lem:s_sum}
  \uses{def:s, lem:valGauss_add_valGauss_sub_self, lem:valGauss_eq_zero}
  \leanok
  We have
  \[
    \sum_{a=0}^{q-2} s(a) = \frac{(q-2) \cdot f \cdot (p-1)}{2}.
  \]
\end{lemma}

\begin{proof}
  \leanok
  We pair each $a \in \{0, \ldots, q-2\}$ with $q-1-a$. By \ref{lem:valGauss_add_valGauss_sub_self}
  with $k = q-1$, for any $a$ with $(q-1) \nmid a$,
  \[
    s(a) + s(q-1-a) = (p-1) \cdot f.
  \]
  Note that $s(0) = 0$ and $s(q-1) = 0$ by \ref{lem:valGauss_eq_zero}.

  Since $p$ is odd, $q - 1 = p^f - 1$ is even. The set $\{1, \ldots, q-2\}$
  consists of $(q-3)/2$ pairs $\{a, q-1-a\}$ with $a \neq q-1-a$, plus the
  central term $a = (q-1)/2$ which pairs with itself. Since $q \geq 3$, we have
  $1 \leq (q-1)/2 \leq q-2$, so $(q-1) \nmid (q-1)/2$, and
  \ref{lem:valGauss_add_valGauss_sub_self} gives $2s((q-1)/2) = (p-1)f$,
  hence $s((q-1)/2) = (p-1)f/2$. Therefore
  \[
    \sum_{a=0}^{q-2} s(a) = \frac{q-3}{2} \cdot (p-1)f + \frac{(p-1)f}{2}
    = \frac{(q-2)(p-1)f}{2}.
  \]
\end{proof}

\begin{proposition}
  \label{prop:valGauss_toNat_eq_sum_digits}
  \uses{lem:valGauss_toNat_le_sum_digits, lem:s_sum}
  \lean{valGauss_toNat_eq_sum_digits}
  \leanok
  Let $0 \leq a < q - 1$ and let
  $a = a_0 + a_1 p + \cdots + a_{f-1} p^{f-1}$, $0 \leq a_i \leq p - 1$,
  be the $p$-adic expansion of $a$. Then
  \[
    s(a) = a_0 + a_1 + \cdots + a_{f-1}.
  \]
\end{proposition}

\begin{proof}
  \leanok
  By \ref{lem:valGauss_toNat_le_sum_digits}, $s(a) \leq a_0 + \cdots + a_{f-1}$ for all $a$.
  Summing over $0 \leq a \leq q - 2$, we get
  \[
    \sum_{a=0}^{q-2} s(a) \leq \sum_{a=0}^{q-2} (a_0 + \cdots + a_{f-1}).
  \]
  By \texttt{Nat.sum\_sum\_digits\_eq}, the right-hand side equals
  $\frac{(q-2) \cdot f \cdot (p-1)}{2}$, which equals the left-hand side
  by \ref{lem:s_sum}. Since each term satisfies
  $s(a) \leq a_0 + \cdots + a_{f-1}$ and the sums are equal, we must have
  $s(a) = a_0 + \cdots + a_{f-1}$ for all $a$.
\end{proof}

\section{Setup for Stickelberger's theorem}

Let $m$ be a positive integer dividing $q-1$ such that $f$ is the
multiplicative order of $p$ modulo $m$. Let $d = (q-1)/m$.

Let $E = \QQ(\zeta_m, \zeta_p)$. We denote by $\tilde{\fp}$ the prime
ideal of $\cO_E$ lying below $\fP$, i.e.\ $\tilde{\fp} = \fP \cap \cO_E$,
and by $\tilde{\fp}_0$ the prime ideal of $\cO_{\QQ(\zeta_m)}$ lying below
$\fP$, i.e.\ $\tilde{\fp}_0 = \fP \cap \cO_{\QQ(\zeta_m)}$.

Let $G = \Gal(\QQ(\zeta_m)/\QQ) \cong (\ZZ/m\ZZ)^\times$.
For $(a, m) = 1$, let $\sigma_a \in G$ denote the element with
$\sigma_a(\zeta_m) = \zeta_m^a$. The \emph{Stickelberger element} is
\[
  \Theta = \Theta(\QQ(\zeta_m)) = \sum_{\substack{a = 1 \\ (a,m) = 1}}^{m}
  a \cdot \sigma_a^{-1} \in \ZZ[G].
\]
\begin{remark}
  With the notation of Washington, $\Theta = m\theta$ where
  $\theta = \sum_{(a,m)=1} \{a/m\} \sigma_a^{-1} \in \QQ[G]$.
\end{remark}

\begin{lemma}
  \label{lem:s_fractional_part}
  \uses{prop:valGauss_toNat_eq_sum_digits}
  \lean{Nat.sub_one_mul_sum_fract_div_eq_digits_sum}
  \leanok
  For $0 \leq h < q - 1$,
  \[
    s(h) = (p-1) \sum_{i=0}^{f-1} \left\{ \frac{p^i h}{q-1} \right\}.
  \]
\end{lemma}

\begin{proof}
  \leanok
  Let $h = a_0 + a_1 p + \cdots + a_{f-1} p^{f-1}$ be the $p$-adic
  expansion of $h$. By \ref{prop:valGauss_toNat_eq_sum_digits}, $s(h) = a_0 + \cdots + a_{f-1}$.
  On the other hand, $p^i h \equiv a_0 p^i + \cdots + a_{f-1} p^{i+f-1}
  \pmod{q-1}$, so
  \[
    \left\{ \frac{p^i h}{q-1} \right\} = \frac{1}{q-1}(a_0 p^i + a_1 p^{i+1}
    + \cdots + a_{f-1} p^{i-1}),
  \]
  where the indices are taken mod $f$. Summing over $i = 0, \ldots, f-1$,
  each digit $a_j$ appears multiplied by $(p^0 + p^1 + \cdots + p^{f-1})/(q-1) = 1/(p-1)$.
  Therefore
  \[
    (p-1) \sum_{i=0}^{f-1} \left\{ \frac{p^i h}{q-1} \right\} = a_0 + \cdots + a_{f-1} = s(h).
  \]
\end{proof}

\begin{lemma}
  \label{lem:s_mod}
  \uses{lem:s_fractional_part}
  \lean{mul_valGauss_eq_mul_sum}
  \leanok
  For $a \in (\ZZ/m\ZZ)^\times$,
  \[
    m \cdot s(ad) = (p-1) \sum_{i=0}^{f-1} (ap^i \bmod m).
  \]
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:s_fractional_part} applied to $h = ad$,
  \[
    s(ad) = (p-1) \sum_{i=0}^{f-1} \left\{ \frac{p^i \cdot ad}{q-1} \right\}
           = (p-1) \sum_{i=0}^{f-1} \left\{ \frac{ap^i}{m} \right\}.
  \]
  Since $m \cdot \{ap^i/m\} = ap^i \bmod m$ for each $i$, multiplying by $m$
  gives the result.
\end{proof}

\begin{lemma}
  \label{lem:gauss_sum_power_fixed_1}
  \uses{lem:gal_gaussSum_eq_gaussSum}
  \lean{smul_gaussSum_eq_gaussSum'}
  \leanok
  For any $\tau \in \Gal(L/\QQ(\zeta_m, \zeta_p))$,
  $\tau(g_d^m) = g_d^m$.
\end{lemma}

\begin{proof}
  \leanok
  Since $\tau$ fixes $\zeta_m$, we have $\fn(\tau) \equiv 1 \pmod{m}$, so
  $d \cdot \fn(\tau) \equiv d \pmod{q-1}$.
  By \ref{lem:gal_gaussSum_eq_gaussSum},
  $\tau(g_d) = g_{d \cdot \fn(\tau)} = g_d$.
  Therefore $\tau(g_d^m) = g_d^m$.
\end{proof}

\begin{lemma}
  \label{lem:gauss_sum_galois_action_K}
  \uses{def:gauss_sum_s, def:addchar}
  \lean{smul_gaussSum_eq_mul_gaussSum}
  \leanok
  For any $\tau \in \Gal(L/K)$ with $\tau(\zeta_p) = \zeta_p^e$ and $e \in (\mathbb{Z}/p\mathbb{Z})^\times$,
  \[
    \tau(g_d) = \omega^d(e)^{-1} \cdot g_d.
  \]
\end{lemma}

\begin{proof}
   \leanok
  Since $\tau \in \Gal(L/K)$, it fixes $K = \QQ(\zeta_{q-1})$, hence fixes
  all $(q-1)$-th roots of unity, hence fixes $\omega(x)$ for all $x$.
  Since $\tau(\zeta_p) = \zeta_p^e$, we have $\tau(\psi(x)) = \psi(ex)$. Therefore
  \[
    \tau(g_d) = \sum_{x \in \cO_K/P} \omega(x)^{-d} \psi(ex).
  \]
  By the substitution $y = ex$ (bijection on $\cO_K/P$ since
  $e \in (\ZZ/p\ZZ)^\times \subset \FF_q^\times$),
  \[
    \sum_x \omega(x)^{-d}\psi(ex) = \sum_y \omega(y/e)^{-d}\psi(y)
    = \omega^d(e)^{-1} \sum_y \omega(y)^{-d}\psi(y) = \omega^d(e)^{-1} g_d.
  \]
\end{proof}

\begin{lemma}
  \label{lem:gauss_sum_power_fixed_2}
  \uses{lem:gauss_sum_galois_action_K}
  \lean{smul_gaussSum_eq_gaussSum}
  \leanok
  For any $\tau \in \Gal(L/K)$,
  $\tau(g_d^m) = g_d^m$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:gauss_sum_galois_action_K}, $\tau(g_d) = \omega^d(e)^{-1} g_d$
  where $e$ is such that $\tau(\zeta_p) = \zeta_p^e$.
  Since $md = q-1$, we have $\omega^d(e)^{-m} = \omega^{q-1}(e)^{-1} = 1$, hence
  $\tau(g_d^m) = \omega^d(e)^{-m} g_d^m = g_d^m$.
\end{proof}

\begin{lemma}
  \label{lem:gauss_sum_power_in_cyclotomic}
  \uses{def:gauss_sum_s, lem:gauss_sum_power_fixed_1, lem:gauss_sum_power_fixed_2}
  \lean{exists_mem_gaussSum_pow_eq}
  \leanok
  We have $g_d^m \in \cO_{\QQ(\zeta_m)}$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:gauss_sum_power_fixed_1}, $g_d^m$ is fixed by
  $\Gal(L/\QQ(\zeta_m, \zeta_p))$, so $g_d^m \in \QQ(\zeta_m, \zeta_p)$.
  By \ref{lem:gauss_sum_power_fixed_2}, $g_d^m$ is fixed by $\Gal(L/K)$.
  Since $\QQ(\zeta_m, \zeta_p) = KF$ and $\Gal(KF/\QQ(\zeta_m))$ is generated
  by the restrictions of $\Gal(L/\QQ(\zeta_m, \zeta_p))$ and $\Gal(L/K)$,
  we conclude $g_d^m \in \QQ(\zeta_m)$.
  Since $g_d \in \cO_L$, we have $g_d^m \in \cO_{\QQ(\zeta_m)}$.
\end{proof}

\begin{lemma}
  \label{lem:L_E_unramified}
  \lean{ramificationIdx_eq_one}
  \leanok
  The extension $L/E$ is unramified above $p$.
\end{lemma}

\begin{proof}
  \leanok
  We have $L = E(\zeta_{q-1})$ and $E = \QQ(\zeta_m, \zeta_p)$. Since
  $\gcd(q-1, p) = 1$, the extension $\QQ(\zeta_{q-1})/\QQ$ is unramified
  at $p$. Since the ramification above $p$ in $L/\QQ$ comes entirely from
  the adjunction of $\zeta_p$, which is already in $E$, the extension $L/E$
  is unramified above $p$.
\end{proof}

\begin{lemma}
  \label{lem:p_totally_ramified_in_E}
  \lean{ramificationIdx_eq_sub_one}
  \leanok
  The extension $E/\QQ(\zeta_m)$ is totally ramified above $\tilde{\fp}_0$
  with ramification index $p - 1$. In particular,
  $\tilde{\fp}^{p-1} = \tilde{\fp}_0 \cdot \cO_E$.
\end{lemma}

\begin{proof}
  \leanok
  Since $\gcd(m, p) = 1$ (as $m \mid q - 1$ and $\gcd(q-1, p) = 1$),
  the fields $\QQ(\zeta_m)$ and $\QQ(\zeta_p)$ are linearly disjoint over
  $\QQ$, and $E = \QQ(\zeta_m) \cdot \QQ(\zeta_p)$. The extension
  $\QQ(\zeta_p)/\QQ$ is totally ramified at $p$ with ramification index
  $p - 1$. Since $\QQ(\zeta_m)/\QQ$ is unramified at $p$ (as $p \nmid m$),
  the extension $E/\QQ(\zeta_m)$ is totally ramified at $\tilde{\fp}_0$
  with ramification index $p - 1$.
\end{proof}

\begin{lemma}
  \label{lem:stabilizer_tP0}
  \lean{galEquivZMod_stabilizer'}
  \leanok
  The stabilizer of $\tilde{\fp}_0$ in $G = \Gal(\QQ(\zeta_m)/\QQ)$ is
  the subgroup $\langle \sigma_p \rangle$, which has order $f$.
  In particular, for $a \in (\ZZ/m\ZZ)^\times$ and $0 \leq i \leq f-1$,
  \[
    \sigma_{ap^i}^{-1}(\tilde{\fp}_0) = \sigma_a^{-1}(\tilde{\fp}_0).
  \]
\end{lemma}

\begin{proof}
  \leanok
  Since $f$ is the multiplicative order of $p$ modulo $m$ by hypothesis,
  $\langle \sigma_p \rangle$ has order $f$ in $G$. The stabilizer of
  $\tilde{\fp}_0$ in $G$ is the decomposition group of $\tilde{\fp}_0$
  over $p$, which is generated by $\sigma_p$ since $p$ generates the
  decomposition group.
\end{proof}

\begin{lemma}
  \label{lem:valuation_gauss_sum_power}
  \uses{lem:L_E_unramified,
        lem:p_totally_ramified_in_E,
        lem:gal_gaussSum_eq_gaussSum, lem:gauss_sum_galois_action_K, def:s,
        lem:s_mod}
  \lean{emultiplicity_galEquivZMod_symm_smul_gaussSum}
  \leanok
  For $a \in (\ZZ/m\ZZ)^\times$,
  \[
    v_{\sigma_a^{-1}(\tilde{\fp}_0)}(g_d^m) = \sum_{i=0}^{f-1}
    (ap^i \bmod m).
  \]
\end{lemma}

\begin{proof}
  \leanok
  Since $g_d \in \cO_E$, we have
  \[
    v_{\sigma_a^{-1}(\tilde{\fp})}(g_d) = v_{\tilde{\fp}}(\sigma_a(g_d)).
  \]
  Lift $\sigma_a \in G = \Gal(\QQ(\zeta_m)/\QQ)$ to an element
  $\tilde{\sigma}_a \in \Gal(L/F)$ with $\mathfrak{n}(\tilde{\sigma}_a) \equiv a
  \pmod{m}$. By \ref{lem:gal_gaussSum_eq_gaussSum},
  $\tilde{\sigma}_a(g_d) = g_{d \cdot \mathfrak{n}(\tilde{\sigma}_a)}$.
  The ambiguity in the lift is an element of $\Gal(L/\QQ(\zeta_m, \zeta_p))$,
  which acts on $g_d$ by \ref{lem:gauss_sum_galois_action_K} as multiplication
  by a unit $\omega^d(e)^{-1} \in \cO_E^\times$. Therefore
  $v_{\tilde{\fp}}(\sigma_a(g_d)) = v_{\tilde{\fp}}(g_{ad}) = s(ad)$,
  where the last equality uses that $L/E$ is unramified above $p$ by
  \ref{lem:L_E_unramified}, so $v_{\tilde{\fp}}(g_{ad}) = v_{\fP}(g_{ad}) = s(ad)$.
  Since $E/\QQ(\zeta_m)$ is totally ramified of degree $p-1$ above $\tilde{\fp}_0$
  by \ref{lem:p_totally_ramified_in_E}, we have
  $v_{\sigma_a^{-1}(\tilde{\fp})} = (p-1) \cdot v_{\sigma_a^{-1}(\tilde{\fp}_0)}$.
  Therefore
  \[
    v_{\sigma_a^{-1}(\tilde{\fp}_0)}(g_d^m) = \frac{m \cdot s(ad)}{p-1}
    = \sum_{i=0}^{f-1} (ap^i \bmod m),
  \]
  where the last equality is \ref{lem:s_mod}.
\end{proof}

\begin{lemma}
  \label{lem:valuation_stickelberger}
  \uses{lem:stabilizer_tP0}
  \lean{sum_with_mul_inv_mem_eq_sum_range}
  \leanok
  For $a \in (\ZZ/m\ZZ)^\times$,
  \[
    v_{\sigma_a^{-1}(\tilde{\fp}_0)}(\tilde{\fp}_0^{\Theta}) =
    \sum_{i=0}^{f-1} (ap^i \bmod m).
  \]
\end{lemma}

\begin{proof}
  \leanok
  By definition of $\Theta$ and \ref{lem:stabilizer_tP0}, regrouping the
  sum over $(\ZZ/m\ZZ)^\times$ by orbits under $\langle p \rangle$:
  \[
    v_{\sigma_a^{-1}(\tilde{\fp}_0)}(\tilde{\fp}_0^{\Theta})
    = \sum_{b \in (\ZZ/m\ZZ)^\times} b \cdot
      v_{\sigma_a^{-1}(\tilde{\fp}_0)}(\sigma_b^{-1}(\tilde{\fp}_0))
    = \sum_{i=0}^{f-1} (ap^i \bmod m),
  \]
  since $v_{\sigma_a^{-1}(\tilde{\fp}_0)}(\sigma_b^{-1}(\tilde{\fp}_0)) = 1$
  if $b \equiv ap^i \pmod{m}$ for some $i$, and $0$ otherwise.
\end{proof}

\begin{lemma}
  \label{lem:gauss_sum_factorization_cyclotomic}
  \lean{GaussSum_factorization}
  \uses{lem:gauss_sum_power_in_cyclotomic, lem:norm_GaussSum,
        lem:valuation_gauss_sum_power, lem:valuation_stickelberger,
        lem:stabilizer_tP0}
  \leanok
  \[
    (g_d^m) = \tilde{\fp}_0^{\Theta}
  \]
  as ideals of $\cO_{\QQ(\zeta_m)}$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:gauss_sum_power_in_cyclotomic}, $g_d^m \in \cO_{\QQ(\zeta_m)}$.
  By \ref{lem:norm_GaussSum}, $g_d^m$ is only divisible by primes above $p$.
  Similarly, $\tilde{\fp}_0^{\Theta}$ only involves primes above $p$.
  It therefore suffices to check that for every prime $\mathfrak{q} =
  \sigma_a^{-1}(\tilde{\fp}_0)$ of $\cO_{\QQ(\zeta_m)}$ above $p$,
  \[
    v_{\mathfrak{q}}(g_d^m) = v_{\mathfrak{q}}(\tilde{\fp}_0^{\Theta}).
  \]
  By \ref{lem:valuation_gauss_sum_power},
  \[
    v_{\mathfrak{q}}(g_d^m) = \sum_{i=0}^{f-1} (ap^i \bmod m).
  \]
  By \ref{lem:valuation_stickelberger},
  \[
    v_{\mathfrak{q}}(\tilde{\fp}_0^{\Theta}) = \sum_{i=0}^{f-1} (ap^i \bmod m).
  \]
  The two sides are equal, which completes the proof.
\end{proof}

\begin{theorem}
  \label{thm:stickelberger}
  \uses{lem:gauss_sum_factorization_cyclotomic}
  \label{Stickelberger}
  \leanok
  The ideal $\tilde{\fp}_0^{\Theta}$ is principal in $\QQ(\zeta_m)$.
\end{theorem}

\begin{proof}
  \leanok
  By \ref{lem:gauss_sum_factorization_cyclotomic},
  $(g_d^m) = \tilde{\fp}_0^{\Theta}$.
\end{proof}

\begin{corollary}
  \label{cor:stickelberger_annihilates}
  \lean{Stickelberger}
  \leanok
  \uses{thm:stickelberger}
  The Stickelberger element $\Theta$ annihilates the ideal class group
  $\mathrm{Cl}(\QQ(\zeta_m))$: for every ideal class $c$,
  \[
    c^{\Theta} = \prod_{\substack{a = 1 \\ (a,m) = 1}}^{m} \sigma_a^{-1}(c)^{a} = 1.
  \]
\end{corollary}

\begin{proof}
  \leanok
  Every ideal class of a number field contains an integral ideal whose norm is
  coprime to any prescribed integer, so choose a representative $\mathfrak{b} \in c$ with
  $\gcd(N\mathfrak{b}, 2m) = 1$. Then $N\mathfrak{b}$ is odd and coprime to $m$, so applying
  \ref{thm:stickelberger} to each prime factor of $\mathfrak{b}$ and multiplying shows
  that $\mathfrak{b}^{\Theta}$ is principal. Hence
  $c^{\Theta} = [\mathfrak{b}]^{\Theta} = [\mathfrak{b}^{\Theta}] = 1$.
\end{proof}

\begin{remark}
  Mathematically, coprimality of the representative to $m$ alone suffices for
  Stickelberger's theorem. We additionally require the norm to be odd (i.e.\
  coprimality to $2m$) only because the formalized form of \ref{thm:stickelberger}
  is proved for primes of odd residue characteristic.
\end{remark}

\chapter{The Kronecker-Weber theorem}

\section{Reduction to the prime degree case}

Throughout this chapter, $\zeta_m$ denotes a primitive $m$-th root of unity.
We say that an extension is \emph{unramified} if it is unramified at all
finite primes. A normal extension $K/F$ has \emph{exponent} $m$ if
$\Gal(K/F)$ has exponent $m$.

The goal of this chapter is to prove the following theorem.

\begin{theorem}
  \label{thm:kronecker_weber}
  \uses{lem:kw_reduce_to_prime_power, prop:kw_odd_prime_power, prop:kw_2_power}
  Every abelian extension of $\QQ$ is contained in a cyclotomic field.
\end{theorem}

\begin{proof}
  By \ref{lem:kw_reduce_to_prime_power}, it suffices to prove that every cyclic
  extension of $\QQ$ of prime power degree $p^m$ unramified outside $p$ is
  contained in a cyclotomic field. For odd $p$, this is
  \ref{prop:kw_odd_prime_power}; for $p = 2$, this is \ref{prop:kw_2_power}.
\end{proof}

\begin{lemma}
  \label{lem:kw_abelian_cyclic_decomp}
  Every finite abelian extension $K/\QQ$ is the compositum of cyclic
  extensions of prime power degree.
\end{lemma}

\begin{proof}
  The Galois group $\Gal(K/\QQ)$ is a finite abelian group, hence a direct
  product of cyclic groups of prime power order. Each factor corresponds to
  a cyclic subextension of prime power degree, and $K$ is their compositum.
\end{proof}

\begin{lemma}
  \label{lem:kw_reduce_to_unramified_outside_p}
  Let $K/\QQ$ be a cyclic extension of $p$-power degree. Then there
  exists a cyclic extension $F/\QQ$ of $p$-power degree unramified
  outside $p$ such that $K$ is cyclotomic if and only if $F$ is cyclotomic.
\end{lemma}

\begin{proof}
  The discriminant of $K$ is a finite integer, so only finitely many primes
  $q \neq p$ are ramified in $K/\QQ$. We apply \ref{lem:kw_ramification_reduction}
  iteratively. At each step, given a cyclic $p$-power-degree extension $K'/\QQ$
  with some $q \neq p$ ramified, \ref{lem:kw_ramification_reduction} provides
  a cyclotomic extension $\Lambda/\QQ$ and a cyclic $p$-power-degree extension
  $K''/\QQ$ in which $q$ is unramified, such that $K'\Lambda = K''\Lambda$.
  Since $\Lambda$ is cyclotomic, $K'$ is cyclotomic if and only if $K''$ is
  (as $K'\Lambda = K''\Lambda$ and $\Lambda$ is cyclotomic).
  Since $K'' \subseteq K'\Lambda$ and $\Lambda$ is only ramified at $q$, the
  primes ramified in $K''$ are a subset of those of $K'$ union $\{q\}$, which
  equals the primes of $K'$. Since $q$ is unramified in $K''$ by construction,
  the set of primes $\neq p$ ramified in $K''$ is strictly smaller than that of
  $K'$. The process terminates with the desired $F$.
\end{proof}

% Strategy: if q \neq p is ramified in K/Q, the ramification group
% I_q \subseteq \Gal(K/Q) \cong \ZZ/p^m\ZZ is a nontrivial subgroup. One shows
% that there exists a cyclotomic extension L = Q(\zeta_{q^k}) such that KL = FL
% where F/Q is cyclic of degree p^m with q unramified. The key point is that the
% ramification of q in KL/L comes entirely from L, so can be absorbed.
% Iterating over all primes q \neq p ramified in K gives the result.
% See Marcus, Number Fields, Theorem 21.

\begin{lemma}
  \label{lem:kw_reduce_to_prime_power}
  \uses{lem:kw_abelian_cyclic_decomp, lem:kw_reduce_to_unramified_outside_p,
        prop:kw_odd_prime_power, prop:kw_2_power}
  It suffices to prove that every cyclic extension of $\QQ$ of prime power
  degree unramified outside $p$ is cyclotomic.
\end{lemma}

\begin{proof}
  By \ref{lem:kw_abelian_cyclic_decomp}, any finite abelian extension of
  $\QQ$ is a compositum of cyclic extensions of prime power degree. By
  \ref{lem:kw_reduce_to_unramified_outside_p}, each such extension can be
  replaced by a cyclotomic extension times a cyclic extension of prime power
  degree unramified outside $p$. By \ref{prop:kw_odd_prime_power} and
  \ref{prop:kw_2_power}, the latter is cyclotomic.
\end{proof}

\begin{lemma}
  \label{lem:kw_cyclic_compositum}
  \lean{kw_cyclic_compositum}
  \leanok
  Let $L/\QQ$ be a Galois extension with cyclic Galois group, and let
  $K, K'$ be intermediate fields with $[K:\QQ] \mid [K':\QQ]$. Then
  $K \subseteq K'$.
\end{lemma}

\begin{proof}
  \leanok
  Via the Galois correspondence, $K \subseteq K'$ is equivalent to
  $\mathrm{Fix}(K') \leq \mathrm{Fix}(K)$ in $\Gal(L/\QQ)$. In a cyclic
  group, the subgroup lattice is totally ordered by divisibility of indices,
  and $[\Gal(L/\QQ) : \mathrm{Fix}(K)] = [K:\QQ]$ divides
  $[\Gal(L/\QQ) : \mathrm{Fix}(K')] = [K':\QQ]$, so $\mathrm{Fix}(K') \leq \mathrm{Fix}(K)$.
\end{proof}

Throughout this subsection we fix the following local data, and develop the tame character
from scratch (no higher ramification groups). Let $L/K$ be a finite Galois extension of number
fields, $\mathfrak{Q}$ a prime of $L$ above a prime $\mathfrak{p}$ of $K$. Localize at
$\mathfrak{Q}$: let $\mathcal{O} = \mathcal{O}_{L,\mathfrak{Q}}$ be the (discrete) valuation ring,
$\mathfrak{m} = \mathfrak{Q}\mathcal{O}$ its maximal ideal, $v = v_{\mathfrak{Q}}$ the valuation,
$k = k(\mathfrak{Q}) = \mathcal{O}/\mathfrak{m}$ the residue field of characteristic $\ell$, and
$e = e(\mathfrak{Q}\mid\mathfrak{p})$. Let $I = I(\mathfrak{Q}\mid\mathfrak{p})$ be the inertia
group; by definition each $\tau \in I$ fixes $\mathfrak{Q}$ (so $v\circ\tau = v$, i.e.\ $\tau$
preserves $\mathcal{O}$, $\mathfrak{m}$ and each $\mathfrak{m}^n$) and acts trivially on $k$. For
$x \in \mathcal{O}$ we write $\bar x$ for its residue in $k$.

\begin{definition}
  \label{def:tame_character}
  \lean{Ideal.tameCharacter}
  Fix a uniformizer $\pi$ of $\mathcal{O}$ ($v(\pi) = 1$). The \emph{tame character} is
  \[
    \theta : I \to k^\times, \qquad \theta(\tau) = \overline{\tau(\pi)/\pi}.
  \]
  This is well-defined: $v(\tau\pi) = v(\pi) = 1$, so $\tau\pi/\pi \in \mathcal{O}^\times$ and
  $\theta(\tau) = \overline{\tau\pi/\pi} \in k^\times$.
\end{definition}

\begin{lemma}
  \label{lem:tame_character_hom}
  \lean{Ideal.tameCharacter_eq}
  \uses{def:tame_character}
  $\theta$ does not depend on the choice of uniformizer, and it is a group homomorphism.
\end{lemma}

\begin{proof}
  \emph{Independence of $\pi$.} Another uniformizer is $\pi' = u\pi$ with $u \in
  \mathcal{O}^\times$; then $\tau\pi'/\pi' = (\tau u/u)(\tau\pi/\pi)$, and since $\tau$ is trivial
  on $k$ we have $\overline{\tau u} = \bar u$, so $\overline{\tau u/u} = 1$ and the residue is
  unchanged.

  \emph{Homomorphism.} For $\sigma,\tau \in I$ put $a = \tau\pi/\pi \in \mathcal{O}^\times$, so
  $\tau\pi = a\pi$ and $\sigma\tau(\pi) = \sigma(a)\,\sigma(\pi)$. Reducing modulo $\mathfrak{m}$
  and using $\overline{\sigma(a)} = \bar a$ ($\sigma$ trivial on $k$),
  \[
    \theta(\sigma\tau) = \overline{\sigma\tau(\pi)/\pi}
      = \overline{\sigma(a)}\cdot\overline{\sigma(\pi)/\pi}
      = \bar a\cdot\theta(\sigma) = \theta(\tau)\,\theta(\sigma) = \theta(\sigma)\,\theta(\tau),
  \]
  the last step since $k^\times$ is abelian.
\end{proof}

\begin{lemma}[Rigidity]
  \label{lem:tame_character_rigidity}
  \lean{Ideal.eq_one_of_tameCharacter_eq_one}
  \uses{def:tame_character}
  If $\tau \in I$ has order coprime to $\ell$ and $\theta(\tau) = 1$, then $\tau = \mathrm{id}$.
\end{lemma}

\begin{proof}
  Let $m = \operatorname{ord}(\tau)$, coprime to $\ell$. We show by induction on $n \geq 1$ that
  $\tau$ acts trivially on $\mathcal{O}/\mathfrak{m}^n$; since $\bigcap_n \mathfrak{m}^n = 0$
  (Krull intersection), this forces $\tau = \mathrm{id}$ on $\mathcal{O}$, hence on $L$.

  For $n = 1$, $\tau$ is trivial on $\mathcal{O}/\mathfrak{m} = k$ ($\tau \in I$). For $n = 2$,
  $\theta(\tau) = 1$ means $\tau(\pi) \equiv \pi \pmod{\mathfrak{m}^2}$; as $\mathcal{O}/\mathfrak{m}^2$
  is generated as a ring by $k$ and $\bar\pi$, $\tau$ is trivial on $\mathcal{O}/\mathfrak{m}^2$.

  Assume $n \geq 3$ and $\tau \equiv \mathrm{id}$ on $\mathcal{O}/\mathfrak{m}^{n-1}$. Set $R =
  \mathcal{O}/\mathfrak{m}^n$ and $J = \mathfrak{m}^{n-1}/\mathfrak{m}^n \subseteq R$, a
  one-dimensional $k$-vector space (spanned by $\bar\pi^{\,n-1}$) with $J^2 = 0$. The hypothesis
  gives $\delta := \tau - \mathrm{id} : R \to J$. As $\tau$ is a ring homomorphism and $\delta$
  takes values in the square-zero ideal $J$,
  \[
    \delta(xy) = \tau(x)\tau(y) - xy = x\,\delta(y) + y\,\delta(x) + \delta(x)\delta(y)
      = x\,\delta(y) + y\,\delta(x),
  \]
  so $\delta$ is a derivation. Moreover $\tau$ acts on $J$ by multiplication by $\theta(\tau)^{n-1}
  = 1$ (since $\tau(\pi^{n-1}) = (\tau\pi)^{n-1} \equiv \theta(\tau)^{n-1}\pi^{n-1}
  \pmod{\mathfrak{m}^n}$), so $\delta|_J = 0$, whence $\delta^2 = 0$. Therefore $\tau^m =
  (\mathrm{id} + \delta)^m = \mathrm{id} + m\,\delta$. From $\tau^m = \mathrm{id}$ we get
  $m\,\delta = 0$; as $J$ is a $k$-vector space and $m$ is invertible in $k$ ($m$ coprime to
  $\ell = \operatorname{char} k$), $\delta = 0$, i.e.\ $\tau \equiv \mathrm{id}$ on
  $\mathcal{O}/\mathfrak{m}^n$.
\end{proof}

\begin{lemma}
  \label{lem:tame_inertia_cyclic}
  \lean{Ideal.isCyclic_inertia}
  \uses{lem:tame_character_hom, lem:tame_character_rigidity}
  If $\mathfrak{Q}\mid\mathfrak{p}$ is tamely ramified (i.e.\ $\ell \nmid e$), then $\theta$ is
  injective; consequently $I(\mathfrak{Q}\mid\mathfrak{p})$ is cyclic of order $e$.
\end{lemma}

\begin{proof}
  By \ref{lem:tame_character_hom}, $\theta$ is a homomorphism. Since $\ell \nmid e = |I|$, every
  $\tau \in I$ has order dividing $e$, hence coprime to $\ell$; so if $\theta(\tau) = 1$ then
  $\tau = \mathrm{id}$ by \ref{lem:tame_character_rigidity}. Thus $\theta$ is injective and $I$
  embeds in $k^\times$. A finite subgroup of the multiplicative group of a field is cyclic, so
  $I \cong \theta(I)$ is cyclic of order $|I| = e$ (with $\theta(I) = \mu_e(k)$).
\end{proof}

\begin{lemma}
  \label{lem:tame_character_frob}
  \lean{Ideal.tameCharacter_conj}
  \uses{lem:tame_character_hom}
  Let $\sigma$ lie in the decomposition group $D(\mathfrak{Q}\mid\mathfrak{p})$ and reduce to the
  Frobenius $x \mapsto x^{N\mathfrak{p}}$ of $k/k(\mathfrak{p})$. Then for all $\tau \in I$,
  $\theta(\sigma\tau\sigma^{-1}) = \theta(\tau)^{N\mathfrak{p}}$.
\end{lemma}

\begin{proof}
  Since $\sigma$ preserves $\mathcal{O}$ and $\mathfrak{m}$, $\pi' = \sigma^{-1}(\pi)$ is again a
  uniformizer; by \ref{lem:tame_character_hom} we may compute $\theta(\tau)$ with $\pi'$, so
  $\tau(\pi') = a'\pi'$ with $\overline{a'} = \theta(\tau)$. Then $\sigma\tau\sigma^{-1}(\pi) =
  \sigma(\tau(\pi')) = \sigma(a')\,\sigma(\pi')= \sigma(a')\,\pi$, so $\theta(\sigma\tau\sigma^{-1})
  = \overline{\sigma(a')} = \overline{a'}^{\,N\mathfrak{p}} = \theta(\tau)^{N\mathfrak{p}}$, using
  that $\sigma$ induces the $N\mathfrak{p}$-power Frobenius on $k$.
\end{proof}

\begin{lemma}
  \label{lem:abelian_tame_ramificationIdx_dvd}
  \lean{Ideal.card_inertia_dvd_card_sub_one}
  \uses{lem:tame_inertia_cyclic, lem:tame_character_frob}
  Let $L/K$ be a finite \emph{abelian} extension of number fields and
  $\mathfrak{Q}\mid\mathfrak{p}$ a tamely ramified prime. Then $e(\mathfrak{Q}\mid\mathfrak{p})$
  divides $N\mathfrak{p} - 1 = \#k(\mathfrak{p}) - 1$.
\end{lemma}

\begin{proof}
  Let $\tau$ generate $I$ (cyclic by \ref{lem:tame_inertia_cyclic}) and let $\sigma \in
  D(\mathfrak{Q}\mid\mathfrak{p})$ reduce to Frobenius. As $\sigma$ normalizes $I$,
  $\sigma\tau\sigma^{-1} \in I$, and by \ref{lem:tame_character_frob} together with injectivity
  of $\theta$ (\ref{lem:tame_inertia_cyclic}), $\theta(\sigma\tau\sigma^{-1}) =
  \theta(\tau)^{N\mathfrak{p}} = \theta(\tau^{N\mathfrak{p}})$ gives $\sigma\tau\sigma^{-1} =
  \tau^{N\mathfrak{p}}$. Since $L/K$ is abelian, $\sigma\tau\sigma^{-1} = \tau$, so
  $\tau^{N\mathfrak{p}-1} = 1$ and $e = \operatorname{ord}(\tau)$ divides $N\mathfrak{p} - 1$.
\end{proof}

\begin{lemma}
  \label{lem:kw_ramification_reduction}
  \uses{lem:tame_inertia_cyclic, lem:abelian_tame_ramificationIdx_dvd}
  Let $K/\QQ$ be a cyclic extension of $p$-power degree. If $q \neq p$ is ramified in $K/\QQ$,
  then there exist a cyclic cyclotomic extension $\Lambda/\QQ$ and a cyclic extension $F/\QQ$
  of $p$-power degree in which $q$ is unramified, such that $K\Lambda = F\Lambda$ and every
  prime ramified in $F$ is ramified in $K$.
\end{lemma}

\begin{proof}
  Write $[K:\QQ] = p^m$. As $q \neq p$, the prime $q$ is tamely ramified in $K/\QQ$; let
  $p^s = e_q(K)$ be its ramification index, so by \ref{lem:tame_inertia_cyclic} the inertia
  $I_q(K) \subseteq \Gal(K/\QQ) \cong \ZZ/p^m\ZZ$ is cyclic of order $p^s$. Since $K/\QQ$ is
  abelian, \ref{lem:abelian_tame_ramificationIdx_dvd} (with $N(q) = q$) gives $p^s \mid q - 1$.
  As $\Gal(\QQ(\zeta_q)/\QQ) \cong \ZZ/(q-1)\ZZ$ is cyclic, it has a unique subfield $\Lambda$ of
  degree $p^s$; since $\QQ(\zeta_q)/\QQ$ is totally ramified at $q$, so is $\Lambda$, whence
  $I_q(\Lambda) = \Gal(\Lambda/\QQ)$.

  Put $A = K\Lambda$ and fix a prime of $A$ above $q$ with inertia $I := I_q(A) \subseteq
  \Gal(A/\QQ)$. Restriction embeds $\Gal(A/\QQ) \hookrightarrow \Gal(K/\QQ) \times
  \Gal(\Lambda/\QQ)$ and carries $I$ onto $I_q(K)$ and onto $I_q(\Lambda) = \Gal(\Lambda/\QQ)$
  (inertia surjects onto the inertia of a sub-extension). So $I$, cyclic by
  \ref{lem:tame_inertia_cyclic}, embeds into $I_q(K) \times I_q(\Lambda) \cong \ZZ/p^s\ZZ \times
  \ZZ/p^s\ZZ$ while surjecting onto each factor; a cyclic subgroup of $\ZZ/p^s\ZZ \times
  \ZZ/p^s\ZZ$ with that property has order exactly $p^s$, so $|I| = p^s$.

  Let $F := A^{I}$ be the inertia field of $q$ in $A/\QQ$; by the defining property of the
  inertia field, $q$ is unramified in $F/\QQ$. Because $\Lambda$ is totally ramified at $q$, the
  restriction $I \to \Gal(\Lambda/\QQ)$ is onto, and since $|I| = p^s = |\Gal(\Lambda/\QQ)|$ it
  is an isomorphism. Hence $I \cap \Gal(A/\Lambda) = 1$ and $|I|\cdot|\Gal(A/\Lambda)| =
  |\Gal(A/\QQ)|$, so
  \[
    \Gal(A/\QQ) = I \times \Gal(A/\Lambda).
  \]
  Therefore:
  \begin{itemize}
    \item $F\Lambda = A$ (from $I \cap \Gal(A/\Lambda) = 1$), i.e.\ $F\Lambda = K\Lambda$;
    \item $\Gal(F/\QQ) = \Gal(A/\QQ)/I \cong \Gal(A/\Lambda) \cong \Gal\!\big(K/(K\cap\Lambda)\big)$,
      which is cyclic of $p$-power degree; so $F/\QQ$ is cyclic of $p$-power degree.
  \end{itemize}
  Finally $F \subseteq A = K\Lambda$ and $\Lambda$ is ramified only at $q$, so every prime
  ramified in $F$ is ramified in $K$, while $q$ is unramified in $F$.
  (When $K \cap \Lambda = \QQ$ — automatic if $s < m$, since the only subfield of $K$ totally
  ramified at $q$ is $\QQ$ — one gets $[F:\QQ] = p^m$, recovering Lemmermeyer's statement.)
\end{proof}

\begin{proposition}
  \label{prop:kw_2_quadratic}
  The only quadratic extensions of $\QQ$ unramified outside $2$ are
  $\QQ(i)$, $\QQ(\sqrt{-2})$, and $\QQ(\sqrt{2})$. In particular, the
  maximal real abelian $2$-extension of $\QQ$ with exponent $2$ and
  unramified outside $2$ is $\QQ(\sqrt{2})$, which is the subfield of
  $\QQ(\zeta_8)$ of degree $2$.
\end{proposition}

\begin{proof}
  Write a quadratic field as $\QQ(\sqrt{d})$ with $d \neq 1$ squarefree. A prime $\ell$
  ramifies in $\QQ(\sqrt d)$ iff $\ell \mid \operatorname{disc}$, where $\operatorname{disc} = d$
  if $d \equiv 1 \pmod 4$ and $\operatorname{disc} = 4d$ otherwise. Hence $\QQ(\sqrt d)$ is
  unramified outside $2$ iff $\operatorname{disc}$ is
  (up to sign) a power of $2$, i.e.\ iff $d$ has no odd prime factor, i.e.\ $d \in \{-1, 2, -2\}$.
  This gives exactly $\QQ(i)$ ($d = -1$), $\QQ(\sqrt 2)$ ($d = 2$), and $\QQ(\sqrt{-2})$
  ($d = -2$); of these only $\QQ(\sqrt 2)$ is real. Since $\zeta_8 + \zeta_8^{-1} = \sqrt 2$,
  we have $\QQ(\sqrt 2) = \QQ(\zeta_8)^+$, the degree-$2$ subfield of $\QQ(\zeta_8)$; being the
  only real quadratic field unramified outside $2$, it is the maximal real abelian extension
  of exponent $2$ unramified outside $2$.
\end{proof}

\begin{proposition}
  \label{prop:kw_2_power}
  \uses{prop:kw_2_quadratic, lem:kw_cyclic_compositum, prop:kw_exponent_p}
  Let $K/\QQ$ be a cyclic extension of degree $2^m$ unramified outside $2$.
  Then $K$ is cyclotomic.
\end{proposition}

\begin{proof}
  We proceed by induction on $m$. The base case $m = 1$ is \ref{prop:kw_2_quadratic}: each
  of $\QQ(i) = \QQ(\zeta_4)$, $\QQ(\sqrt 2)$, $\QQ(\sqrt{-2})$ lies in $\QQ(\zeta_8)$. Assume
  $m \geq 2$ and the result for degree $2^{m-1}$.

  \emph{Real case.} Suppose $K$ is totally real. Let $K' = \QQ(\zeta_{2^{m+2}})^+$ be the
  maximal real subfield of $\QQ(\zeta_{2^{m+2}})$; since $\Gal(\QQ(\zeta_{2^{m+2}})/\QQ) \cong
  \ZZ/2\ZZ \times \ZZ/2^m\ZZ$ with complex conjugation generating the $\ZZ/2\ZZ$ factor, the
  quotient $\Gal(K'/\QQ) \cong \ZZ/2^m\ZZ$ is cyclic, and $K'$ is unramified outside $2$. If
  $K'K/\QQ$ were not cyclic, then $\Gal(K'K/\QQ)$ — an abelian $2$-group, as $K$ and $K'$ are
  abelian — would have $(\ZZ/2\ZZ)^2$ as a quotient, hence three distinct quadratic subfields
  of $K'K$; all are unramified outside $2$, and all are real since $K'K$ is totally real. This
  contradicts \ref{prop:kw_2_quadratic} (a unique real quadratic field unramified outside $2$).
  Hence $K'K/\QQ$ is cyclic, and \ref{lem:kw_cyclic_compositum} gives
  $K \subseteq K' \subseteq \QQ(\zeta_{2^{m+2}})$.

  \emph{Nonreal case.} Suppose $K$ is not totally real, and let $M$ be the maximal real subfield
  of $K(i)$. Then $K(i) = K \cdot \QQ(i) = M(i)$, so $K$ is cyclotomic $\iff K(i)$ is cyclotomic
  $\iff M$ is cyclotomic. Now $K(i)/\QQ$ is abelian (a compositum of the abelian extensions $K$
  and $\QQ(i)$) and unramified outside $2$, hence so is $M$; moreover $M$ is cyclic: if
  $i \notin K$ then $\Gal(K(i)/\QQ) \cong \ZZ/2^m\ZZ \times \ZZ/2\ZZ$ with complex conjugation
  equal to $(2^{m-1}, 1)$, and the quotient by $\langle(2^{m-1},1)\rangle$ is cyclic of order
  $2^m$; if $i \in K$ then $M = K^+$ is a subfield of the cyclic field $K$. Thus $M$ is real,
  cyclic of $2$-power degree, and unramified outside $2$, so by the real case $M$ — and hence
  $K$ — is cyclotomic. (This is Lemmermeyer's argument, Prop.~1.5.)
\end{proof}

\begin{proposition}
  \label{prop:kw_odd_prime_power}
  \uses{prop:kw_exponent_p, lem:kw_cyclic_compositum, lem:kw_minkowski}
  Let $K/\QQ$ be a cyclic extension of odd prime power degree $p^m$
  unramified outside $p$. Then $K$ is cyclotomic.
\end{proposition}

\begin{proof}
  Let $K'$ be the unique subfield of degree $p^m$ in $\QQ(\zeta_{p^{m+1}})$;
  it is cyclic of degree $p^m$ and unramified outside $p$. Suppose $K'K/\QQ$
  is not cyclic. Then $\Gal(K'K/\QQ)$ is a non-cyclic $p$-group, so it has
  $(\ZZ/p\ZZ)^2$ as a quotient, giving a subfield $E \subseteq K'K$ with
  $\Gal(E/\QQ) \cong (\ZZ/p\ZZ)^2$. Since $K$ and $K'$ are unramified
  outside $p$, so is $E$. But by \ref{prop:kw_exponent_p} the maximal
  abelian extension of exponent $p$ unramified outside $p$ is cyclic of
  degree $p$, so no extension with Galois group $(\ZZ/p\ZZ)^2$ exists.
  This contradiction shows $K'K/\QQ$ is cyclic, so \ref{lem:kw_cyclic_compositum}
  gives $K \subseteq K' \subseteq \QQ(\zeta_{p^{m+1}})$.
\end{proof}

\section{The prime degree case}

\begin{proposition}
  \label{prop:kw_exponent_p}
  \lean{prop_kw_exponent_p}
  \uses{thm:stickelberger, lem:kw_kummer, lem:kw_abelian_kummer,
        lem:kw_split_prime, lem:kw_mu_val_p, lem:kw_mu_pth_power_ideal, lem:kw_class_trivial,
        lem:kw_mu_unit, lem:kw_unit_root_of_unity, lem:kw_ramification_reduction}
  Let $p$ be an odd prime, $A$ a field of characteristic $0$, and $\xi \in A$ a primitive
  $p^2$-th root of unity. Let $K \subseteq A$ be a subextension of $A/\QQ$ that is Galois and
  cyclic of degree $p$ over $\QQ$ and unramified outside $p$. Then $K \subseteq \QQ(\xi)$
  (the cyclotomic subfield $\QQ(\zeta_{p^2})$ of $A$). In particular $K$ is the unique subfield
  of degree $p$ of $\QQ(\xi)$, and the maximal abelian extension of $\QQ$ of exponent $p$
  unramified outside $p$ is cyclic of degree $p$.
\end{proposition}

\begin{proof}
  Set $\zeta = \xi^p$, a primitive $p$-th root of unity, $F = \QQ(\zeta) \subseteq A$, and
  $L = K \cdot F \subseteq A$ the compositum.

  \begin{enumerate}
    \item By \ref{lem:kw_kummer}, $L = F(\sqrt[p]{\mu})$ for some $\mu \in
      \cO_F$, $\mu \neq 0$, with $X^p - \mu$ irreducible over $F$.
    \item By \ref{lem:kw_mu_val_p}, $p \mid v_{\mathfrak{q}}(\mu)$ for every prime
      $\mathfrak{q}$ of $\cO_F$.
    \item By \ref{lem:kw_mu_pth_power_ideal}, $(\mu) = \mathfrak{a}^p$ for
      some ideal $\mathfrak{a}$ of $\cO_F$.
    \item By \ref{lem:kw_class_trivial}, $\mathfrak{a}$ is principal.
    \item By \ref{lem:kw_mu_unit}, $\mu = \alpha^p \eta$ for some $\alpha \in
      \cO_F$ and unit $\eta \in \cO_F^\times$.
    \item By \ref{lem:kw_unit_root_of_unity}, $L = \QQ(\xi)$ (the $p^2$-cyclotomic subfield).
  \end{enumerate}

  Therefore $K \subseteq L = \QQ(\xi)$. The Galois group
  $\Gal(\QQ(\xi)/\QQ) \cong (\ZZ/p^2\ZZ)^\times \cong \ZZ/p\ZZ \times
  \ZZ/(p-1)\ZZ$ has a unique subgroup of index $p$, giving a unique subfield
  of degree $p$ over $\QQ$ in $\QQ(\xi)$. Since $K \cap F = \QQ$
  (as $\gcd([K:\QQ], [F:\QQ]) = \gcd(p, p-1) = 1$) and $[\QQ(\xi):F] = p$,
  $K$ is exactly this unique subfield. The same applies to any other cyclic
  extension of degree $p$ unramified outside $p$, so the maximal such abelian
  extension of exponent $p$ is cyclic of degree $p$.
\end{proof}

The proof of \ref{prop:kw_exponent_p} proceeds via the following lemmas.
Let $K/\QQ$ be a cyclic extension of prime degree $p$ unramified outside $p$.
Set $F = \QQ(\zeta_p)$ and $L = KF$.

\begin{lemma}
  \label{lem:kw_kummer}
  \uses{lem:kw_abelian_kummer}
  \lean{kw_kummer, kw_kummer'}
  \leanok
  With the above notation, $L = F(\sqrt[p]{\mu})$ for some nonzero
  $\mu \in \cO_F$, and $L/F$ is unramified outside $p$.
\end{lemma}

\begin{proof}
  \leanok
  Since $[K:\QQ] = p$ and $[F:\QQ] = p-1$ are coprime, $K \cap F = \QQ$,
  so $[L:F] = [KF:F] = [K:\QQ] = p$. The extension $L/F$ is Galois with
  cyclic Galois group of order $p$, and $F$ contains the $p$-th roots of
  unity $\mu_p$. By Kummer theory (\texttt{IsCyclic} + \texttt{Kummer}),
  $L = F(\sqrt[p]{\mu})$ for some $\mu \in F^\times$. Multiplying by a
  $p$-th power if necessary, we may assume $\mu \in \cO_F$.

  For the ramification: let $q \neq p$ be a prime. Since $K/\QQ$ is
  unramified at $q$ and $F/\QQ$ is unramified at $q$ (as $F = \QQ(\zeta_p)$
  is only ramified at $p$), the compositum $L = KF$ is also unramified
  at $q$.
\end{proof}

\begin{lemma}
  \label{lem:kw_abelian_kummer}
  \lean{kw_abelian_kummer}
  \leanok
  Since $L/\QQ$ is Galois, for every $\sigma_a \in \Gal(F/\QQ)$ there exists
  $\xi \in F^\times$ and $a \in \ZZ$ such that $\sigma_a(\mu) = \xi^p \mu^a$.
\end{lemma}

\begin{proof}
  \leanok
  This is the backward direction of \texttt{isGalois\_iff\_forall\_apply\_eq\_pow\_mul\_zpow}:
  since $L/\QQ$ is Galois (which holds because $L = KF$ is the compositum of

\end{proof}


\begin{lemma}
  \label{lem:kw_split_prime}
  \lean{kw_split_prime}
  \leanok
  \uses{lem:kw_abelian_kummer}
  Let $\mathfrak{q}$ be a prime ideal of $F$ with $(\mu) = \mathfrak{q}^r \mathfrak{a}$
  with $\mathfrak{q} \nmid \mathfrak{a}$. If $p \nmid r$ and $L/\QQ$ is
  abelian, then $\mathfrak{q}$ splits completely in $F/\QQ$.
\end{lemma}

\begin{proof}
  \leanok
  Let $\sigma_a$ be an element of the decomposition group of $\mathfrak{q}$
  in $\Gal(F/\QQ)$, so $\sigma_a(\mathfrak{q}) = \mathfrak{q}$.
  Since $L/\QQ$ is abelian, by \ref{lem:kw_abelian_kummer} there exists
  $\xi \in F^\times$ such that $\sigma_a(\mu) = \xi^p \mu^a$. Since
  $\sigma_a(\mathfrak{q}) = \mathfrak{q}$, applying $\sigma_a$ to the
  factorization $(\mu) = \mathfrak{q}^r \mathfrak{a}$ gives
  $(\xi^p \mu^a) = \mathfrak{q}^r \sigma_a(\mathfrak{a})$, hence
  $\mathfrak{q}^r \| \xi^p \mu^a$. But also $(\mu^a) = \mathfrak{q}^{ar}\mathfrak{a}^a$,
  so $\mathfrak{q}^r \| \xi^p \mu^a$ implies $r \equiv ar \pmod{p}$.
  Since $p \nmid r$, this forces $a \equiv 1 \pmod{p}$,
  so $\sigma_a = 1$ and $\mathfrak{q}$ splits completely in $F/\QQ$.
\end{proof}

\begin{lemma}
  \label{lem:kw_mu_val_p}
  \uses{lem:kw_kummer, lem:kw_split_prime}
  \lean{kw_mu_val_p}
  \leanok
  For any prime $\mathfrak{q}$ of $\cO_F$, $p \mid v_{\mathfrak{q}}(\mu)$.
\end{lemma}

\begin{proof}
  \leanok
  Write $r = v_{\mathfrak{q}}(\mu)$.

  \textbf{Case 1: $\mathfrak{q} \nmid p$.}
  By \ref{lem:kw_kummer}, $L/F$ is unramified at $\mathfrak{q}$, so
  $e(\mathfrak{Q}|\mathfrak{q}) = 1$ for any prime $\mathfrak{Q}$ of $L$
  above $\mathfrak{q}$. Let $\alpha \in L$ satisfy $\alpha^p = \mu$. Then
  \[
    p \cdot v_{\mathfrak{Q}}(\alpha) = v_{\mathfrak{Q}}(\alpha^p)
    = v_{\mathfrak{Q}}(\mu) = e(\mathfrak{Q}|\mathfrak{q})\cdot v_{\mathfrak{q}}(\mu)
    = v_{\mathfrak{q}}(\mu) = r,
  \]
  so $p \mid r$.

  \textbf{Case 2: $\mathfrak{q} = \mathfrak{p} = (1-\zeta_p)$.}
  Since $F/\QQ$ is totally ramified at $p$, $\mathfrak{p}$ is the unique prime
  of $F$ above $p$, so its decomposition group in $\Gal(F/\QQ)$ is all of
  $\Gal(F/\QQ)$. Since $|\Gal(F/\QQ)| = p-1 > 1$, $\mathfrak{p}$ does not
  split completely in $F/\QQ$. By the contrapositive of
  \ref{lem:kw_split_prime}, $p \mid r$.
\end{proof}

\begin{lemma}
  \label{lem:kw_not_dvd_nu}
  \uses{lem:kw_kummer, lem:kw_mu_val_p}
  \lean{kw_not_dvd_nu}
  \leanok
  There exists $\nu \in \cO_F$, $\nu \neq 0$, such that $L = F(\sqrt[p]{\nu})$,
  $X^p - \nu$ is irreducible over $F$, and $(1-\zeta_p) \nmid \nu$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:kw_mu_val_p}, $p \mid v_{\mathfrak{p}}(\mu)$; write
  $v_{\mathfrak{p}}(\mu) = pk$. Set $\nu = \mu / (1-\zeta_p)^{pk} \in \cO_F$.
  Since $(1-\zeta_p)^k \in F^\times$, we have $F(\sqrt[p]{\nu}) =
  F(\sqrt[p]{\mu}/(1-\zeta_p)^k) = F(\sqrt[p]{\mu}) = L$
  (by \ref{lem:kw_kummer}). By construction $v_{\mathfrak{p}}(\nu) = 0$,
  i.e. $(1-\zeta_p) \nmid \nu$.
\end{proof}

\begin{lemma}
  \label{lem:kw_mu_pth_power_ideal}
  \uses{lem:kw_mu_val_p}
  \lean{kw_mu_pth_power_ideal}
  \leanok
  $(\mu) = \mathfrak{a}^p$ for some ideal $\mathfrak{a}$ of $\cO_F$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:kw_mu_val_p}, $p \mid v_{\mathfrak{q}}(\mu)$ for every prime
  $\mathfrak{q}$. Therefore $(\mu) = \mathfrak{a}^p$ where
  $\mathfrak{a} = \prod_{\mathfrak{q}} \mathfrak{q}^{v_{\mathfrak{q}}(\mu)/p}$.
\end{proof}

\begin{lemma}
  \label{lem:kw_class_trivial}
  \lean{kw_class_trivial}
  \leanok
  \uses{lem:kw_mu_pth_power_ideal, cor:stickelberger_annihilates, lem:kw_abelian_kummer}
  The ideal class $[\mathfrak{a}]$ is trivial, i.e., $\mathfrak{a}$ is principal.
\end{lemma}

\begin{proof}
  \leanok
  From $(\mu) = \mathfrak{a}^p$ and $\sigma_a(\mu) = \xi^p \mu^a$
  (by \ref{lem:kw_abelian_kummer}), we get
  $\sigma_a(\mathfrak{a})^p = \sigma_a(\mathfrak{a}^p) = \sigma_a((\mu)) =
  (\xi^p \mu^a) = (\xi)^p \cdot \mathfrak{a}^{pa}$. Since the ideal monoid is
  torsion-free, $\sigma_a(\mathfrak{a}) = (\xi) \cdot \mathfrak{a}^a$, hence
  $\sigma_a(c) = c^a$ for the ideal class $c = [\mathfrak{a}]$ and every
  $1 \leq a < p$.

  Note that we work throughout with the \emph{class} $c$: unlike in the per-ideal
  form of Stickelberger, we never need $\mathfrak{a}$ itself to be coprime to $p$
  or to have odd norm. By \ref{cor:stickelberger_annihilates} applied with
  $m = p$, the class $c$ is annihilated by the Stickelberger element $\Theta =
  \sum_{a=1}^{p-1} a\sigma_a^{-1} \in \ZZ[G]$. We compute $c^{\Theta}$:
  \[
    1 = c^{\Theta} = \prod_{a=1}^{p-1} \sigma_a^{-1}(c)^a
    = \prod_{a=1}^{p-1} c^{a^{-1} \cdot a} = c^{p-1},
  \]
  where we used $\sigma_a^{-1}(c) = \sigma_{a^{-1}}(c) = c^{a^{-1}}$, with
  $a^{-1}$ the inverse of $a$ modulo $p$. Since $(\mu) = \mathfrak{a}^p$, the
  ideal $\mathfrak{a}^p$ is principal, so $c^p = 1$ and hence
  $c^{p-1} = c^{-1}$. Combining with $c^{\Theta} = 1$ gives $c^{-1} = 1$, i.e.\
  $c = [\mathfrak{a}] = 1$ and $\mathfrak{a}$ is principal.
\end{proof}

\begin{lemma}
  \label{lem:kw_mu_unit}
  \lean{kw_mu_unit}
  \leanok
  \uses{lem:kw_class_trivial}
  There exists a unit $\eta \in \cO_F^\times$ such that $\mu = \alpha^p \eta$
  for some $\alpha \in \cO_F$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:kw_class_trivial}, $\mathfrak{a} = (\alpha)$ for some
  $\alpha \in \cO_F$. Then $(\mu) = \mathfrak{a}^p = (\alpha^p)$, so
  $\mu/\alpha^p \in \cO_F^\times$. Setting $\eta = \mu/\alpha^p$ gives
  $\mu = \alpha^p \eta$.
\end{proof}

\begin{lemma}
  \label{lem:kw_conj_mul_eq_pow}
  \lean{kw_conj_mul_eq_pow}
  \leanok
  \uses{lem:kw_abelian_kummer}
  Let $\sigma_{-1}$ denote complex conjugation on $F$. With $\mu = \alpha^p \eta$, the product
  $\eta\,\sigma_{-1}(\eta)$ is a $p$-th power in $F^\times$.
\end{lemma}

\begin{proof}
  \leanok
  Since $L/\QQ$ is abelian, \ref{lem:kw_abelian_kummer} applied to $\sigma_{-1}$ gives
  $\xi \in F^\times$ with $\sigma_{-1}(\mu) = \xi^p\, \mu^{-1}$. Substituting $\mu = \alpha^p \eta$
  and using that $\sigma_{-1}$ is a ring homomorphism,
  \[
    \sigma_{-1}(\alpha)^p\, \sigma_{-1}(\eta) = \xi^p\, \alpha^{-p}\, \eta^{-1},
  \]
  whence
  \[
    \eta\, \sigma_{-1}(\eta)
      = \xi^p\, \alpha^{-p}\, \sigma_{-1}(\alpha)^{-p}
      = \bigl(\xi/(\alpha\, \sigma_{-1}(\alpha))\bigr)^p .
  \]
\end{proof}

\begin{lemma}
  \label{lem:kw_exists_realUnit_torsion}
  \lean{kw_exists_realUnit_torsion}
  \leanok
  \uses{lem:kw_conj_mul_eq_pow}
  The fourth power $\eta^4$ factors as $\eta^4 = \zeta\, \delta^p$ with $\zeta$ a $p$-th root of
  unity and $\delta \in F$.
\end{lemma}

\begin{proof}
  \leanok
  Let $\sigma_{-1}$ be complex conjugation on $F = \QQ(\zeta_p)$. For any unit
  $\eta \in \cO_F^\times$, the ratio $\eta/\sigma_{-1}(\eta)$ has absolute value $1$ at every
  archimedean place, hence is a root of unity, while $\eta\,\sigma_{-1}(\eta)$ is a real unit;
  their product being $\eta^2$, we get $\eta^2 = \zeta_0\,\varepsilon_0$ with $\zeta_0$ a root of
  unity and $\sigma_{-1}(\varepsilon_0) = \varepsilon_0$. Squaring,
  \[
    \eta^4 = \zeta\, \varepsilon, \qquad \zeta = \zeta_0^2,\ \varepsilon = \varepsilon_0^2 .
  \]
  The roots of unity in $F$ form $\mu_{2p}$, so $\zeta_0^{2p} = 1$ and hence $\zeta^p =
  \zeta_0^{2p} = 1$: $\zeta$ is a $p$-th root of unity. (Squaring twice lands the torsion factor
  in $\mu_p$, sidestepping any sign normalization.)

  Now $\sigma_{-1}$ inverts the root of unity $\zeta$ and fixes the real unit $\varepsilon$, so
  applying it to $\eta^4 = \zeta\,\varepsilon$ gives $\sigma_{-1}(\eta)^4 = \zeta^{-1}\varepsilon$;
  multiplying the two relations the $\zeta$ cancels:
  \[
    \varepsilon^2 = \bigl(\eta\,\sigma_{-1}(\eta)\bigr)^4 .
  \]
  By \ref{lem:kw_conj_mul_eq_pow}, $\eta\,\sigma_{-1}(\eta) = w^p$, so $\varepsilon^2 = (w^4)^p$.
  As $p$ is odd, $\gcd(2,p) = 1$, hence $\varepsilon = \delta^p$ for some $\delta \in F$.
  Therefore $\eta^4 = \zeta\, \delta^p$.
\end{proof}

\begin{lemma}
  \label{lem:kw_unit_root_of_unity}
  \lean{kw_unit_root_of_unity}
  \leanok
  \uses{lem:kw_mu_unit, lem:kw_exists_realUnit_torsion}
  The unit $\eta$ is a $p$-th power times a root of unity. Therefore $L = \QQ(\zeta_{p^2})$.
\end{lemma}

\begin{proof}
  \leanok
  By \ref{lem:kw_exists_realUnit_torsion}, $\eta^4 = \zeta\, \delta^p$ with $\zeta$ a $p$-th root
  of unity and $\delta \in F$. With $\mu = \alpha^p \eta$ (\ref{lem:kw_mu_unit}),
  \[
    \mu^4 = \alpha^{4p}\, \zeta\, \delta^p = \zeta\,(\alpha^4 \delta)^p \in
    \zeta_p^{\ZZ} (F^\times)^p .
  \]
  As $\gcd(4,p) = 1$ we have $F(\sqrt[p]{\mu}) = F(\sqrt[p]{\mu^4})$, and $\mu^4 \in
  \zeta_p^{\ZZ}(F^\times)^p$ gives $F(\sqrt[p]{\mu^4}) = F(\sqrt[p]{\zeta}) \subseteq
  \QQ(\zeta_{p^2})$. Since $[L:F] = p = [\QQ(\zeta_{p^2}):F]$, we conclude $L = \QQ(\zeta_{p^2})$.
\end{proof}
